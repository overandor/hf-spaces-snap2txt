# HF Repo Scanner

Scan every line of code in every file across all Hugging Face repos for a given username.

## Features

- **Security Pattern Detection**: Scans for hardcoded secrets, private keys, SQL injection risks, eval/exec usage, and more
- **Multi-Repo Support**: Scans models, datasets, and spaces
- **Real-time Progress**: Background scanning with status polling
- **Detailed Findings**: Line-by-line findings with severity levels (critical, warning, info)

## Usage

1. Enter a Hugging Face username
2. Select repo type (All, Spaces only, Models only, Datasets only)
3. Click "Start Scan"
4. View results as they are discovered

## Environment Variables

- `HF_TOKEN`: Optional Hugging Face API token for private repos

# ── Helpers ─────────────────────────────────────────────────────
def _get_api():
    token = os.environ.get("HF_TOKEN") or os.environ.get("HUGGING_FACE_TOKEN")
    return HfApi(token=token)


def _fetch_file_content(repo_id: str, path: str, repo_type: str, api: HfApi) -> str | None:
    """Download and decode a file from HF. Returns None for binary or too large."""
    try:
        # Skip binary / media / archive files
        skip_ext = {".png", ".jpg", ".jpeg", ".gif", ".mp4", ".mp3", ".wav", ".zip", ".tar", ".gz", ".tgz", ".bz2", ".7z", ".rar", ".exe", ".dll", ".so", ".dylib", ".bin", ".onnx", ".pt", ".pth", ".safetensors", ".ckpt", ".h5", ".pb", ".pbtxt", ".pbtxt", ".npy", ".npz", ".parquet", ".arrow", ".orc", ".avro", ".msgpack", ".pickle", ".pkl", ".joblib", ".pkl", ".feather", ".db", ".sqlite", ".sqlite3", ".woff", ".woff2", ".ttf", ".eot", ".ico", ".pdf", ".doc", ".docx", ".xls", ".xlsx", ".ppt", ".pptx"}
        if Path(path).suffix.lower() in skip_ext:
            return None
        if Path(path).name.lower() in {"license", "license.txt", "copying", "authors", "changelog", "news", "readme", "readme.md", ".gitignore", ".gitattributes", ".dockerignore"}:
            return None

        local = hf_hub_download(repo_id=repo_id, filename=path, repo_type=repo_type, local_dir=None)
        # hf_hub_download returns path — but for spaces with no token may fail. Fallback to API.
        if local and Path(local).exists():
            try:
                return Path(local).read_text(encoding="utf-8", errors="ignore")
            except Exception:
                return None
    except EntryNotFoundError:
        return None
    except Exception:
        # Fallback: try raw download via API
        try:
            from huggingface_hub import hf_hub_url, cached_download
            url = hf_hub_url(repo_id=repo_id, filename=path, repo_type=repo_type)
            data = cached_download(url, force_download=False)
            return Path(data).read_text(encoding="utf-8", errors="ignore")
        except Exception:
            return None
    return None


def _scan_text(text: str, checks: list[dict]) -> list[dict]:
    findings = []
    for line_no, line in enumerate(text.splitlines(), 1):
        for check in checks:
            if re.search(check["pattern"], line):
                findings.append({
                    "line_number": line_no,
                    "line_text": line[:500],
                    "check_id": check["id"],
                    "check_name": check["name"],
                    "severity": check["severity"],
                })
    return findings


def _scan_worker(scan_id: str, username: str, repo_filter: str):
    """Background thread that performs the scan."""
    api = _get_api()
    result = {
        "scan_id": scan_id,
        "username": username,
        "started_at": datetime.utcnow().isoformat(),
        "status": "running",
        "total_repos": 0,
        "total_files": 0,
        "total_lines": 0,
        "findings": [],
        "error": None,
    }
    SCAN_RESULTS[scan_id] = result

    try:
        # Collect repo list
        repos = []
        if repo_filter in ("all", "model"):
            repos += [(m.id, "model") for m in api.list_models(author=username)]
        if repo_filter in ("all", "dataset"):
            repos += [(d.id, "dataset") for d in api.list_datasets(author=username)]
        if repo_filter in ("all", "space"):
            repos += [(s.id, "space") for s in api.list_spaces(author=username)]

        result["total_repos"] = len(repos)

        for repo_id, rtype in repos:
            try:
                files = list_repo_tree(repo_id=repo_id, repo_type=rtype)
                for entry in files:
                    if entry.type != "file":
                        continue
                    path = entry.path
                    content = _fetch_file_content(repo_id, path, rtype, api)
                    if content is None:
                        continue
                    result["total_files"] += 1
                    lines = content.splitlines()
                    result["total_lines"] += len(lines)
                    findings = _scan_text(content, DEFAULT_CHECKS)
                    for f in findings:
                        f["repo_id"] = repo_id
                        f["repo_type"] = rtype
                        f["file_path"] = path
                        result["findings"].append(f)
            except Exception as e:
                # Log per-repo errors but continue
                result.setdefault("repo_errors", []).append({"repo": repo_id, "error": str(e)})

        result["status"] = "done"
        result["finished_at"] = datetime.utcnow().isoformat()
    except Exception as e:
        result["status"] = "error"
        result["error"] = str(e)
        traceback.print_exc()
    finally:
        ACTIVE_SCANS.discard(scan_id)

@app.route("/")
def index():
    scan_id = request.args.get("scan_id")
    username = request.args.get("username", "")
    results = SCAN_RESULTS.get(scan_id) if scan_id else None
    return render_template_string(PAGE, scan_id=scan_id, username=username, results=results)


@app.route("/api/scan", methods=["POST"])
def api_scan():
    data = request.get_json(force=True) or {}
    username = (data.get("username") or "").strip().lower()
    repo_type = (data.get("repo_type") or "all").strip().lower()
    if not username:
        return jsonify({"error": "username required"}), 400

    scan_id = f"{username}_{int(time.time() * 1000)}"
    ACTIVE_SCANS.add(scan_id)
    t = threading.Thread(target=_scan_worker, args=(scan_id, username, repo_type), daemon=True)
    t.start()
    return jsonify({"scan_id": scan_id, "status": "started"})


@app.route("/api/scan/<scan_id>")
def api_scan_status(scan_id: str):
    res = SCAN_RESULTS.get(scan_id)
    if not res:
        return jsonify({"scan_id": scan_id, "status": "not_found"}), 404
    return jsonify({"scan_id": scan_id, "status": res["status"], "error": res.get("error")})


@app.route("/health")
def health():
    return jsonify({"status": "ok"})


if __name__ == "__main__":
    port = int(os.environ.get("PORT", 7860))
    logger.info(f"Starting HF Repo Scanner on port {port}")
    app.run(host="0.0.0.0", port=port, debug=False, threaded=True)

"""
HF MCP Client — Minimal SSE-based MCP client for Hugging Face.

Endpoints (from .env):
  HF_MCP_LOGIN_URL=https://huggingface.co/mcp?login
  HF_MCP_ENDPOINT=https://huggingface.co/mcp
"""
import os
import json
import uuid
import logging
import requests
from typing import Any, Optional

logger = logging.getLogger(__name__)

DEFAULT_MCP_URL = os.environ.get("HF_MCP_ENDPOINT", "https://huggingface.co/mcp")
DEFAULT_TOKEN = os.environ.get("HF_TOKEN") or os.environ.get("HUGGING_FACE_TOKEN")


class HfMcpClient:
    """Minimal MCP client using SSE + HTTP POST (Hugging Face MCP)."""

    def __init__(self, endpoint: Optional[str] = None, token: Optional[str] = None):
        self.endpoint = endpoint or DEFAULT_MCP_URL
        self.token = token or DEFAULT_TOKEN
        self.session = requests.Session()
        self.message_endpoint: Optional[str] = None
        self._headers = {"Accept": "text/event-stream"}
        if self.token:
            self._headers["Authorization"] = f"Bearer {self.token}"

    def connect(self, timeout: int = 30) -> str:
        """
        Connect via SSE and capture the message POST endpoint.
        Returns the discovered message endpoint URL.
        """
        resp = self.session.get(
            self.endpoint,
            headers=self._headers,
            stream=True,
            timeout=timeout,
        )
        resp.raise_for_status()

        # Parse first SSE events to find endpoint
        for line in resp.iter_lines(decode_unicode=True):
            if line is None:
                continue
            if line.startswith("event: endpoint"):
                # Next line should be data: <url>
                continue
            if line.startswith("data: ") and self.message_endpoint is None:
                data = line[6:].strip()
                if data.startswith("http"):
                    self.message_endpoint = data
                    logger.info("MCP message endpoint: %s", self.message_endpoint)
                    break

        if not self.message_endpoint:
            # Fallback: assume same endpoint for POST
            self.message_endpoint = self.endpoint
        return self.message_endpoint

    def call_tool(self, name: str, arguments: dict[str, Any]) -> dict[str, Any]:
        """Call an MCP tool by name with JSON arguments."""
        if not self.message_endpoint:
            self.connect()

        payload = {
            "jsonrpc": "2.0",
            "id": str(uuid.uuid4()),
            "method": "tools/call",
            "params": {"name": name, "arguments": arguments},
        }
        headers = {"Content-Type": "application/json"}
        if self.token:
            headers["Authorization"] = f"Bearer {self.token}"

        resp = self.session.post(
            self.message_endpoint,
            headers=headers,
            json=payload,
            timeout=60,
        )
        resp.raise_for_status()
        return resp.json()

    def list_tools(self) -> list[dict[str, Any]]:
        """List available MCP tools."""
        if not self.message_endpoint:
            self.connect()

        payload = {
            "jsonrpc": "2.0",
            "id": str(uuid.uuid4()),
            "method": "tools/list",
            "params": {},
        }
        headers = {"Content-Type": "application/json"}
        if self.token:
            headers["Authorization"] = f"Bearer {self.token}"

        resp = self.session.post(
            self.message_endpoint,
            headers=headers,
            json=payload,
            timeout=30,
        )
        resp.raise_for_status()
        result = resp.json()
        return result.get("result", {}).get("tools", [])


def get_client() -> HfMcpClient:
    """Factory: returns a configured HfMcpClient."""
    return HfMcpClient()


if __name__ == "__main__":
    logging.basicConfig(level=logging.INFO)
    client = get_client()
    try:
        tools = client.list_tools()
        print("Available tools:", json.dumps(tools, indent=2))
    except Exception as e:
        print("Error:", e)
