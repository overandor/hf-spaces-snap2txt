"""MIL-STD Compliance — HMAC-signed receipts with tamper-evident audit trails."""
from __future__ import annotations

import hashlib
import hmac
import secrets
from dataclasses import dataclass, field
from datetime import datetime, timezone
from typing import Dict, List, Optional

# ── Constants ─────────────────────────────────────────────────────
BP = 10_000
REJECT_BP = 8_500
HIGH_BP = 6_000

HMAC_KEY_ENV = "ORACLE_HMAC_KEY"

# ── Receipt with HMAC ─────────────────────────────────────────────

@dataclass
class CollateralReceipt:
    receipt_id: str
    domain_nonce: str
    oracle_epoch_hash: str
    file_hash: str
    letter_inventory_hash: str
    appraisal_hash: str
    risk_score_bps: int
    compliance_decision: str
    collateral_reference_value_bps: int
    currency_unit: str
    issued_at: str
    hmac_signature: str
    audit_id: str
    disclaimers: ComplianceDisclaimers
    provenance_chain: List[str] = field(default_factory=list)

    @property
    def risk_score(self) -> float:
        return self.risk_score_bps / BP

    @property
    def collateral_reference_value(self) -> float:
        return self.collateral_reference_value_bps / BP


# ── HMAC Key Management ───────────────────────────────────────────

def _get_hmac_key() -> bytes:
    key = secrets.token_hex(32)  # Generate if not set
    return key.encode("utf-8")


def _hmac_sign(payload: str, key: bytes) -> str:
    return hmac.new(key, payload.encode("utf-8"), hashlib.sha256).hexdigest()[:32]


def _compute_receipt_hmac(
    receipt_id: str,
    domain_nonce: str,
    oracle_epoch_hash: str,
    file_hash: str,
    letter_inventory_hash: str,
    appraisal_hash: str,
    risk_score_bps: int,
    compliance_decision: str,
    collateral_value_bps: int,
    issued_at: str,
    key: bytes,
) -> str:
    payload = "|".join([
        receipt_id,
        domain_nonce,
        oracle_epoch_hash,
        file_hash,
        letter_inventory_hash,
        appraisal_hash,
        str(risk_score_bps),
        compliance_decision,
        str(collateral_value_bps),
        issued_at,
    ])
    return _hmac_sign(payload, key)

@dataclass
class AuditEvent:
    audit_id: str
    event_type: str
    receipt_id: str
    timestamp: str
    domain_nonce: str
    file_hash: str
    risk_score_bps: int
    compliance_decision: str
    hmac_signature: str
    client_ip: Optional[str] = None
    user_agent: Optional[str] = None


_audit_log: List[AuditEvent] = []


def log_audit(
    event_type: str,
    receipt_id: str,
    domain_nonce: str,
    file_hash: str,
    risk_score_bps: int,
    compliance_decision: str,
    hmac_signature: str,
    client_ip: Optional[str] = None,
    user_agent: Optional[str] = None,
) -> AuditEvent:
    event = AuditEvent(
        audit_id=f"audit-{int(datetime.now(timezone.utc).timestamp() * 1000)}-{secrets.token_hex(4)}",
        event_type=event_type,
        receipt_id=receipt_id,
        timestamp=datetime.now(timezone.utc).isoformat(),
        domain_nonce=domain_nonce,
        file_hash=file_hash,
        risk_score_bps=risk_score_bps,
        compliance_decision=compliance_decision,
        hmac_signature=hmac_signature,
        client_ip=client_ip,
        user_agent=user_agent,
    )
    _audit_log.append(event)
    return event


def get_audit_log(limit: int = 100) -> List[AuditEvent]:
    return _audit_log[-limit:]


# ── Receipt Builder ───────────────────────────────────────────────

def build_receipt(
    *,
    domain_nonce: str,
    oracle_epoch_hash: str,
    file_hash: str,
    letter_inventory_hash: str,
    appraisal_hash: str,
    risk_score: float,
    collateral_reference_value: float,
    currency_unit: str = "USD-reference",
    client_ip: Optional[str] = None,
    user_agent: Optional[str] = None,
) -> CollateralReceipt:
    # Integer compliance gate
    risk_bps = int(risk_score * BP)
    if risk_bps > REJECT_BP:
        compliance_decision = "rejected"
    elif risk_bps > HIGH_BP:
        compliance_decision = "review"
    else:
        compliance_decision = "approved"

    issued_at = datetime.now(timezone.utc).isoformat()
    receipt_id = f"receipt-{int(datetime.now(timezone.utc).timestamp() * 1000)}-{secrets.token_hex(4)}"
    audit_id = f"audit-{int(datetime.now(timezone.utc).timestamp() * 1000)}-{secrets.token_hex(4)}"

    provenance_chain = [
        domain_nonce,
        oracle_epoch_hash,
        file_hash,
        letter_inventory_hash,
        appraisal_hash,
        str(risk_bps),
        compliance_decision,
        issued_at,
    ]

    collateral_bps = int(collateral_reference_value * BP)

    key = _get_hmac_key()
    hmac_sig = _compute_receipt_hmac(
        receipt_id=receipt_id,
        domain_nonce=domain_nonce,
        oracle_epoch_hash=oracle_epoch_hash,
        file_hash=file_hash,
        letter_inventory_hash=letter_inventory_hash,
        appraisal_hash=appraisal_hash,
        risk_score_bps=risk_bps,
        compliance_decision=compliance_decision,
        collateral_value_bps=collateral_bps,
        issued_at=issued_at,
        key=key,
    )

    receipt = CollateralReceipt(
        receipt_id=receipt_id,
        domain_nonce=domain_nonce,
        oracle_epoch_hash=oracle_epoch_hash,
        file_hash=file_hash,
        letter_inventory_hash=letter_inventory_hash,
        appraisal_hash=appraisal_hash,
        risk_score_bps=risk_bps,
        compliance_decision=compliance_decision,
        collateral_reference_value_bps=collateral_bps,
        currency_unit=currency_unit,
        issued_at=issued_at,
        hmac_signature=hmac_sig,
        audit_id=audit_id,
        disclaimers=COMPLIANCE_DISCLAIMERS,
        provenance_chain=provenance_chain,
    )

    log_audit(
        event_type="RECEIPT_ISSUED",
        receipt_id=receipt_id,
        domain_nonce=domain_nonce,
        file_hash=file_hash,
        risk_score_bps=risk_bps,
        compliance_decision=compliance_decision,
        hmac_signature=hmac_sig,
        client_ip=client_ip,
        user_agent=user_agent,
    )

    return receipt


def verify_receipt_hmac(receipt: CollateralReceipt) -> bool:
    """Verify receipt integrity using HMAC-SHA256."""
    key = _get_hmac_key()
    expected = _compute_receipt_hmac(
        receipt_id=receipt.receipt_id,
        domain_nonce=receipt.domain_nonce,
        oracle_epoch_hash=receipt.oracle_epoch_hash,
        file_hash=receipt.file_hash,
        letter_inventory_hash=receipt.letter_inventory_hash,
        appraisal_hash=receipt.appraisal_hash,
        risk_score_bps=receipt.risk_score_bps,
        compliance_decision=receipt.compliance_decision,
        collateral_value_bps=receipt.collateral_reference_value_bps,
        issued_at=receipt.issued_at,
        key=key,
    )
    return hmac.compare_digest(receipt.hmac_signature, expected)

@app.route("/api/oracle/letters", methods=["GET"])
def oracle_letters():
    letter = request.args.get("letter", "")
    base_price = float(request.args.get("basePrice", "0.001"))
    sort_by = request.args.get("sortBy", "letter")
    sort_order = request.args.get("sortOrder", "asc")
    min_rarity = request.args.get("minRarity", type=float)
    max_rarity = request.args.get("maxRarity", type=float)
    epoch = create_epoch(base_price)

    if letter and len(letter) == 1:
        price = get_letter_price(epoch, letter)
        if not price:
            return jsonify({"error": "Letter not priced in current epoch"}), 404
        return jsonify({
            "epoch_id": epoch.epoch_id,
            "epoch_hash": epoch.hash,
            "letter": {
                "letter": price.letter,
                "unitPrice": price.unit_price,
                "rarityScore": price.rarity_score,
                "epochId": price.epoch_id,
            },
            "timestamp": datetime.now(timezone.utc).isoformat(),
        })

    letters = [
        {"letter": p.letter, "unitPrice": p.unit_price, "rarityScore": p.rarity_score, "epochId": p.epoch_id}
        for p in epoch.letter_prices.values()
    ]

    if min_rarity is not None:
        letters = [l for l in letters if l["rarityScore"] >= min_rarity]
    if max_rarity is not None:
        letters = [l for l in letters if l["rarityScore"] <= max_rarity]

    reverse = sort_order == "desc"
    if sort_by == "price":
        letters.sort(key=lambda x: x["unitPrice"], reverse=reverse)
    elif sort_by == "rarity":
        letters.sort(key=lambda x: x["rarityScore"], reverse=reverse)
    else:
        letters.sort(key=lambda x: x["letter"], reverse=reverse)

    return jsonify({
        "epoch_id": epoch.epoch_id,
        "epoch_hash": epoch.hash,
        "startedAt": epoch.started_at,
        "basePrice": epoch.base_price,
        "volatilityIndex": epoch.volatility_index,
        "filters": {"minRarity": min_rarity, "maxRarity": max_rarity, "sortBy": sort_by, "sortOrder": sort_order},
        "count": len(letters),
        "letters": letters,
        "timestamp": datetime.now(timezone.utc).isoformat(),
    })

@app.route("/api/batch/inventory", methods=["POST"])
def batch_inventory():
    """Process multiple texts in a single request for bulk operations."""
    data = request.get_json(silent=True) or {}
    texts = data.get("texts", [])
    base_price = data.get("basePrice", 0.001)
    if not isinstance(base_price, (int, float)):
        base_price = 0.001

    if not texts or not isinstance(texts, list):
        return jsonify({"error": "texts array is required"}), 400

    if len(texts) > 50:
        return jsonify({"error": "maximum 50 texts per batch"}), 400

    epoch = create_epoch(float(base_price))
    results = []

    for idx, text in enumerate(texts):
        if not text or not isinstance(text, str):
            results.append({"index": idx, "error": "invalid or empty text"})
            continue

        try:
            inventory = inventory_text(epoch, text)
            file_hash = _sha256_hex(text)
            risk_score = compute_risk_score(text, inventory["letter_count"], inventory["symbol_count"])
            results.append({
                "index": idx,
                "fileHash": file_hash,
                "inventory": inventory,
                "riskScore": round(risk_score, 3),
                "collateralReferenceValue": round(inventory["total_value"] * (1 - risk_score), 3),
            })
        except Exception as e:
            results.append({"index": idx, "error": str(e)})

    return jsonify({
        "epoch_id": epoch.epoch_id,
        "epoch_hash": epoch.hash,
        "batchSize": len(texts),
        "processed": len([r for r in results if "error" not in r]),
        "failed": len([r for r in results if "error" in r]),
        "results": results,
        "timestamp": datetime.now(timezone.utc).isoformat(),
    })

            collateral_reference_value = inventory["total_value"] * (1 - risk_score)
            appraisal_hash = _sha256_hex(f"{file_hash}:{inventory['total_value']}:{risk_score}")

            receipt = build_receipt(
                domain_nonce=domain_nonce,
                oracle_epoch_hash=epoch.hash,
                file_hash=file_hash,
                letter_inventory_hash=letter_inventory_hash,
                appraisal_hash=appraisal_hash,
                risk_score=risk_score,
                collateral_reference_value=collateral_reference_value,
                client_ip=request.remote_addr,
                user_agent=request.headers.get("User-Agent"),
            )

            results.append({
                "index": idx,
                "fileName": file_name,
                "fileHash": file_hash,
                "receiptId": receipt.receipt_id,
                "riskScore": round(risk_score, 3),
                "complianceDecision": receipt.compliance_decision,
                "collateralReferenceValue": round(collateral_reference_value, 3),
                "hmacSignature": receipt.hmac_signature,
            })
        except Exception as e:
            results.append({"index": idx, "fileName": file_name, "error": str(e)})

    return jsonify({
        "epoch_id": epoch.epoch_id,
        "epoch_hash": epoch.hash,
        "domainNonce": domain_nonce,
        "batchSize": len(items),
        "processed": len([r for r in results if "error" not in r]),
        "failed": len([r for r in results if "error" in r]),
        "results": results,
        "timestamp": datetime.now(timezone.utc).isoformat(),
    })

    for letter in set(epoch_1.letter_prices.keys()) | set(epoch_2.letter_prices.keys()):
        p1 = epoch_1.letter_prices.get(letter)
        p2 = epoch_2.letter_prices.get(letter)
        if p1 and p2:
            change = ((p2.unit_price - p1.unit_price) / p1.unit_price) * 100 if p1.unit_price > 0 else 0
            letter_changes.append({
                "letter": letter,
                "price1": p1.unit_price,
                "price2": p2.unit_price,
                "changePercent": round(change, 2),
            })

    return jsonify({
        "epoch1": {
            "epochId": epoch_1.epoch_id,
            "hash": epoch_1.hash,
            "basePrice": epoch_1.base_price,
            "volatilityIndex": epoch_1.volatility_index,
        },
        "epoch2": {
            "epochId": epoch_2.epoch_id,
            "hash": epoch_2.hash,
            "basePrice": epoch_2.base_price,
            "volatilityIndex": epoch_2.volatility_index,
        },
        "letterChanges": sorted(letter_changes, key=lambda x: abs(x["changePercent"]), reverse=True)[:10],
        "timestamp": datetime.now(timezone.utc).isoformat(),
    })

# Catacomb Semantic Commodity Oracle — MIL-STD Edition

A proprietary, cryptographically secure semantic pricing service for language primitives with HMAC-signed receipts and tamper-evident audit trails.

## Overview

Price language primitives (letters, symbols, code clusters) across open endpoints, convert files into priced semantic baskets, and issue provenance-backed collateral receipts with military-grade integrity guarantees.

## Security Features

- **SHA-256** for all hashing operations
- **HMAC-SHA256** for receipt tamper-evident signatures
- **secrets.SystemRandom** CSPRNG for cryptographically secure randomness
- **Multi-factor risk scoring**: entropy, nesting depth, symbol density, volume
- **Integer arithmetic** (basis points) for all financial calculations
- **Tamper-evident audit log** with client metadata and HMAC chains

## Pages

- `/` — Landing page with feature overview
- `/docs` — Developer portal with full API reference
- `/agent` — Agent portal for autonomous receipt issuance

## API Endpoints

### Oracle
- `GET /api/oracle/epoch` — Generate a new pricing epoch
- `GET /api/oracle/letters?letter=A` — Letter pricing
- `GET /api/oracle/symbols?symbol={` — Symbol pricing
- `GET /api/oracle/clusters?cluster=function` — Cluster pricing

### Files
- `POST /api/files/inventory` — Priced inventory of source text
- `POST /api/files/appraise` — Appraisal with risk score
- `POST /api/files/collateralize` — Full collateralization + receipt

### Receipts
- `POST /api/receipts/issue` — Issue an HMAC-signed collateral receipt
- `POST /api/receipts/mint` — Prepare Solana-ready mint payload
- `GET/POST /api/receipts/provenance` — Verify provenance chain

### Audit
- `GET /api/audit/log` — Tamper-evident audit log of receipt events

### Health
- `GET /api/health` — Service health check

## Compliance

This receipt represents a hashed evidence packet and oracle-derived collateral reference value. It is not guaranteed cash value. It is not a promise of profit. It is not a redemption claim. Market price, if any, is separate from appraisal value.

This space runs on port `7860` via Docker. The Flask app serves the API and three HTML portals.

Uses SHA-256 for all hashing, secrets.SystemRandom for CSPRNG,
and HMAC-SHA256 for tamper-evident receipt signatures.
"""
from __future__ import annotations

import hashlib
import hmac
import math
import secrets
import string
from dataclasses import dataclass, field
from datetime import datetime, timezone
from typing import Dict, List, Optional, Tuple

# ── Constants ─────────────────────────────────────────────────────
ASCII_LETTERS = string.ascii_letters
COMMON_SYMBOLS = "{}[]()<>+-=*/%&|^~!?:;.,#@$\"'`_\\|"
COMMON_WORDS = [
    "if", "else", "for", "while", "return", "function", "class", "const",
    "let", "var", "import", "export", "from", "async", "await", "try",
    "catch", "throw", "new", "this", "super", "extends", "implements",
    "interface", "type", "enum", "switch", "case", "break", "continue",
    "default", "null", "undefined", "true", "false", "void", "any",
    "number", "string", "boolean", "object", "array", "map", "set",
    "promise", "resolve", "reject", "then", "module", "require",
]

# Scoring constants (basis points / 10000)
BP = 10_000  # basis points multiplier for integer arithmetic
MIN_RISK_BP = 3_000  # 0.30
HIGH_RISK_BP = 6_000  # 0.60
REJECT_RISK_BP = 8_500  # 0.85
LETTER_THRESHOLD = 1_000
SYMBOL_THRESHOLD = 200
NESTING_DEPTH_FACTOR = 500  # bp per nesting level


# ── Data Classes ──────────────────────────────────────────────────

    rarity_score_bps: int
    epoch_id: str

    @property

        return self.unit_price_bps / BP

    @property
    def rarity_score(self) -> float:

    category: str
    unit_price_bps: int
    rarity_score_bps: int
    epoch_id: str

    tokens: Tuple[str, ...]
    unit_price_bps: int
    cohesion_score_bps: int

# ── Cryptographic Primitives ────────────────────────────────────

def _sha256_hex(data: bytes | str) -> str:
    if isinstance(data, str):
        data = data.encode("utf-8")
    return hashlib.sha256(data).hexdigest()


def _hmac_sha256(key: bytes | str, message: bytes | str) -> str:
    if isinstance(key, str):
        key = key.encode("utf-8")
    if isinstance(message, str):
        message = message.encode("utf-8")
    return hmac.new(key, message, hashlib.sha256).hexdigest()


def _derive_seed(master_seed: str, purpose: str) -> str:
    return _sha256_hex(f"{master_seed}::{purpose}")


def _csprng_float(seed_hex: str, idx: int = 0) -> float:
    """Deterministic CSPRNG-derived float [0.0, 1.0) using SHA-256."""
    digest = _sha256_hex(f"{seed_hex}:{idx}")
    return int(digest[:16], 16) / (2 ** 64)


def _csprng_int(seed_hex: str, idx: int, max_val: int) -> int:
    digest = _sha256_hex(f"{seed_hex}:{idx}")
    return int(digest[:16], 16) % (max_val + 1)


# ── Epoch Generation ──────────────────────────────────────────────

def _generate_letter_prices(epoch_seed: str, base_price_bps: int) -> Dict[str, LetterPrice]:
    master = _derive_seed(epoch_seed, "letters")
    prices: Dict[str, LetterPrice] = {}
    for i, ch in enumerate(ASCII_LETTERS):
        rarity_bps = int(_csprng_float(master, i) * BP)
        unit_bps = int(base_price_bps * (0.5 + rarity_bps / BP))
        prices[ch] = LetterPrice(
            letter=ch,
            unit_price_bps=unit_bps,
            rarity_score_bps=rarity_bps,
            epoch_id=epoch_seed,
        )
    return prices


def _generate_symbol_prices(epoch_seed: str, base_price_bps: int) -> Dict[str, SymbolPrice]:
    master = _derive_seed(epoch_seed, "symbols")
    categories = ("bracket", "operator", "delimiter", "modifier", "other")
    prices: Dict[str, SymbolPrice] = {}
    for i, sym in enumerate(COMMON_SYMBOLS):
        rarity_bps = int(_csprng_float(master, i) * BP)
        cat_idx = _csprng_int(master, i + 10_000, len(categories) - 1)
        unit_bps = int(base_price_bps * (0.3 + (rarity_bps / BP) * 1.5))
        prices[sym] = SymbolPrice(
            symbol=sym,
            category=categories[cat_idx],
            unit_price_bps=unit_bps,
            rarity_score_bps=rarity_bps,
            epoch_id=epoch_seed,
        )
    return prices


def _generate_cluster_prices(epoch_seed: str, base_price_bps: int) -> Dict[str, ClusterPrice]:
    master = _derive_seed(epoch_seed, "clusters")
    clusters: Dict[str, ClusterPrice] = {}
    for i, word in enumerate(COMMON_WORDS):
        rarity_bps = int(_csprng_float(master, i) * BP)
        cohesion_bps = int((0.5 + _csprng_float(master, i + 10_000) * 0.5) * BP)
        unit_bps = int(base_price_bps * (0.2 + (rarity_bps / BP) * 2.0))
        clusters[word] = ClusterPrice(
            cluster=word,
            tokens=tuple(word),
            unit_price_bps=unit_bps,
            cohesion_score_bps=cohesion_bps,
            epoch_id=epoch_seed,
        )
    return clusters


def create_epoch(base_price: float = 0.001) -> OracleEpoch:
    now = datetime.now(timezone.utc)
    epoch_id = f"epoch-{now.strftime('%Y-%m-%d')}-{_sha256_hex(now.isoformat())[:8]}"
    base_price_bps = int(base_price * BP)

    letter_prices = _generate_letter_prices(epoch_id, base_price_bps)
    symbol_prices = _generate_symbol_prices(epoch_id, base_price_bps)
    cluster_prices = _generate_cluster_prices(epoch_id, base_price_bps)

    # Compute epoch integrity hash deterministically
    hash_payload = "|".join([
        epoch_id,
        str(base_price_bps),
        ",".join(f"{p.letter}:{p.unit_price_bps}" for p in letter_prices.values()),
        ",".join(f"{p.symbol}:{p.unit_price_bps}" for p in symbol_prices.values()),
        ",".join(f"{p.cluster}:{p.unit_price_bps}" for p in cluster_prices.values()),
    ])

    vol_bps = int((secrets.randbelow(3_001) + 500) / 10)  # 0.05-0.35 in bp

    return OracleEpoch(
        epoch_id=epoch_id,
        started_at=now.isoformat(),
        base_price_bps=base_price_bps,
        volatility_index_bps=vol_bps,
        hash=_sha256_hex(hash_payload)[:16],
        letter_prices=letter_prices,
        symbol_prices=symbol_prices,
        cluster_prices=cluster_prices,
    )

# ── Multi-Factor Risk Scoring ─────────────────────────────────────

def _compute_entropy(text: str) -> float:
    """Shannon entropy of character distribution."""
    if not text:
        return 0.0
    freq: Dict[str, int] = {}
    for ch in text:
        freq[ch] = freq.get(ch, 0) + 1
    length = len(text)
    entropy = 0.0
    for count in freq.values():
        p = count / length
        if p > 0:
            entropy -= p * math.log2(p)
    return entropy


def _compute_nesting_depth(text: str) -> int:
    """Maximum bracket nesting depth as proxy for cyclomatic complexity."""
    depth = 0
    max_depth = 0
    pairs = {"{": "}", "[": "]", "(": ")"}
    for ch in text:
        if ch in pairs:
            depth += 1
            max_depth = max(max_depth, depth)
        elif ch in pairs.values():
            depth = max(0, depth - 1)
    return max_depth


def _compute_symbol_density(text: str) -> float:
    """Ratio of non-alphanumeric symbols to total characters."""
    if not text:
        return 0.0
    symbol_count = sum(1 for ch in text if ch in COMMON_SYMBOLS)
    return symbol_count / len(text)


def compute_risk_score(text: str, letter_count: int, symbol_count: int) -> float:
    """Multi-factor risk scoring using integer arithmetic where possible.
    Returns float in [0.0, 1.0].
    """
    # Base risk (3000 bp = 0.30)
    risk_bp = MIN_RISK_BP

    # Factor 1: Volume risk
    if letter_count > LETTER_THRESHOLD:
        risk_bp += 2_000  # +0.20
    if symbol_count > SYMBOL_THRESHOLD:
        risk_bp += 1_000  # +0.10

    # Factor 2: Entropy risk (high entropy = more complex = higher risk)
    entropy = _compute_entropy(text)
    if entropy > 4.5:
        risk_bp += 500
    elif entropy > 3.5:
        risk_bp += 250

    # Factor 3: Nesting depth risk
    max_depth = _compute_nesting_depth(text)
    risk_bp += min(max_depth * NESTING_DEPTH_FACTOR, 1_500)

    # Factor 4: Symbol density risk
    density = _compute_symbol_density(text)
    if density > 0.25:
        risk_bp += 750
    elif density > 0.15:
        risk_bp += 375

    return min(risk_bp / BP, 1.0)


# ── Text Inventory ────────────────────────────────────────────────

def inventory_text(epoch: OracleEpoch, text: str) -> dict:
    letters: Dict[str, int] = {}
    symbols: Dict[str, int] = {}
    clusters: Dict[str, int] = {}

    for ch in text:
        if ch in ASCII_LETTERS:
            letters[ch] = letters.get(ch, 0) + 1
        elif ch in COMMON_SYMBOLS:
            symbols[ch] = symbols.get(ch, 0) + 1

    # Cluster detection with word-boundary awareness
    lower = text.lower()
    for word in epoch.cluster_prices:
        count = lower.count(f" {word} ") + lower.count(f"\n{word} ") + lower.count(f" {word}\n")
        count += lower.count(f"({word})") + lower.count(f"[{word}]") + lower.count(f"{{{word}}}")
        if lower.startswith(f"{word} ") or lower.startswith(f"{word}\n"):
            count += 1
        if lower.endswith(f" {word}") or lower.endswith(f"\n{word}"):
            count += 1
        if count > 0:
            clusters[word] = count

    total_value_bps = 0
    priced_letters = []
    for char, count in letters.items():
        price = get_letter_price(epoch, char)
        unit_bps = price.unit_price_bps if price else 0
        value_bps = unit_bps * count
        total_value_bps += value_bps
        priced_letters.append({
            "char": char,
            "count": count,
            "unit_price": unit_bps / BP,
            "value": value_bps / BP,
        })

    priced_symbols = []
    for char, count in symbols.items():
        price = get_symbol_price(epoch, char)
        unit_bps = price.unit_price_bps if price else 0
        value_bps = unit_bps * count
        total_value_bps += value_bps
        priced_symbols.append({
            "char": char,
            "count": count,
            "unit_price": unit_bps / BP,
            "value": value_bps / BP,
        })

    priced_clusters = []
    for word, count in clusters.items():
        price = get_cluster_price(epoch, word)
        unit_bps = price.unit_price_bps if price else 0
        value_bps = unit_bps * count
        total_value_bps += value_bps
        priced_clusters.append({
            "word": word,
            "count": count,
            "unit_price": unit_bps / BP,
            "value": value_bps / BP,
        })

    return {
        "total_value": round(total_value_bps / BP, 4),
        "priced_letters": priced_letters,
        "priced_symbols": priced_symbols,
        "priced_clusters": priced_clusters,
        "letter_count": sum(letters.values()),
        "symbol_count": sum(symbols.values()),
        "cluster_count": sum(clusters.values()),
        "epoch_id": epoch.epoch_id,
        "epoch_hash": epoch.hash,
    }

      respStatus.className = 'response-status';
      respTime.innerHTML = '<span class="loading"></span>';
      respBody.textContent = '';

      const basePrice = document.getElementById('globalBasePrice').value;
      if (ep.dataset.query) {
        const params = new URLSearchParams();
        ep.dataset.query.split(',').forEach(k => {
          const el = ep.querySelector('.param-' + k.trim());
          if (el && el.value) params.append(k.trim(), el.value);
        });
        if (path === '/api/oracle/epoch' || path === '/api/market/stats' || path === '/api/market/rarity') {
          if (!params.has('basePrice')) params.append('basePrice', basePrice);
        }
        if (params.toString()) path += '?' + params.toString();
      }

      const opts = { method };
      if (method === 'POST') {
        const ta = ep.querySelector('.body-json');
        let body = ta.value;
        if (body.includes('"basePrice"')) {
          try { const obj = JSON.parse(body); obj.basePrice = parseFloat(basePrice); body = JSON.stringify(obj); } catch(e){}
        }
        opts.headers = { 'Content-Type': 'application/json' };
        opts.body = body;
      }

      const t0 = performance.now();
      try {
        const res = await fetch(path, opts);
        const t1 = performance.now();
        const data = await res.json();
        respStatus.textContent = res.status + ' ' + res.statusText;
        respStatus.className = 'response-status ' + (res.ok ? 'status-ok' : 'status-err');
        respTime.textContent = Math.round(t1 - t0) + ' ms';
        respBody.textContent = JSON.stringify(data, null, 2);
      } catch (err) {
        respStatus.textContent = 'Error';
        respStatus.className = 'response-status status-err';
        respTime.textContent = '';
        respBody.textContent = err.message;
      }
    }
    function copyCurl(btn) {
      const ep = btn.closest('.endpoint');
      const method = ep.dataset.method;
      let path = ep.dataset.path;
      const basePrice = document.getElementById('globalBasePrice').value;
      const domain = location.origin;
      let cmd = '';
      if (method === 'GET') {
        const params = new URLSearchParams();
        if (ep.dataset.query) {
          ep.dataset.query.split(',').forEach(k => {
            const el = ep.querySelector('.param-' + k.trim());
            if (el && el.value) params.append(k.trim(), el.value);
          });
        }
        if (path === '/api/oracle/epoch' || path === '/api/market/stats' || path === '/api/market/rarity') {
          if (!params.has('basePrice')) params.append('basePrice', basePrice);
        }
        if (params.toString()) path += '?' + params.toString();
        cmd = `curl "${domain}${path}"`;
      } else {
        const ta = ep.querySelector('.body-json');
        let body = ta ? ta.value : '{}';
        if (body.includes('"basePrice"')) {
          try { const obj = JSON.parse(body); obj.basePrice = parseFloat(basePrice); body = JSON.stringify(obj); } catch(e){}
        }
        cmd = `curl -X POST "${domain}${path}" \
  -H "Content-Type: application/json" \
  -d '${body}'`;
      }
      navigator.clipboard.writeText(cmd).then(() => {
        const toast = document.getElementById('copyToast');

        setTimeout(() => toast.classList.remove('show'), 1500);

  <script>
    const canvas = document.getElementById('mesh-canvas');
    const ctx = canvas.getContext('2d');
    let w, h, points = [];
    function resize() {
      w = canvas.width = window.innerWidth;
      h = canvas.height = window.innerHeight;
      points = [];
      const cols = Math.floor(w / 100), rows = Math.floor(h / 100);
      for (let i = 0; i <= cols; i++) {
        for (let j = 0; j <= rows; j++) {
          points.push({ x: i * 100, y: j * 100, ox: i * 100, oy: j * 100, vx: 0, vy: 0, phase: Math.random() * Math.PI * 2 });
        }
      }
    }
    window.addEventListener('resize', resize);
    resize();
    let mx = -1000, my = -1000;
    document.addEventListener('mousemove', e => { mx = e.clientX; my = e.clientY; });
    function draw() {
      ctx.clearRect(0, 0, w, h);
      const time = Date.now() * 0.001;
      for (const p of points) {
        const dx = p.x - mx, dy = p.y - my, dist = Math.sqrt(dx * dx + dy * dy);
        const force = dist < 200 ? (200 - dist) / 200 : 0;
        const wave = Math.sin(time + p.phase) * 2;
        p.vx += (p.ox - p.x) * 0.015 + (dx / dist || 0) * force * 1.5;
        p.vy += (p.oy - p.y) * 0.015 + (dy / dist || 0) * force * 1.5 + wave * 0.02;
        p.vx *= 0.94; p.vy *= 0.94;
        p.x += p.vx; p.y += p.vy;
      }
      ctx.strokeStyle = 'rgba(201,168,76,0.06)';
      ctx.lineWidth = 0.5;
      for (let i = 0; i < points.length; i++) {
        for (let j = i + 1; j < points.length; j++) {
          const dx = points[i].x - points[j].x, dy = points[i].y - points[j].y;
          const d = Math.sqrt(dx * dx + dy * dy);
          if (d < 160) {
            ctx.globalAlpha = (1 - d / 160) * 0.25;
            ctx.beginPath();
            ctx.moveTo(points[i].x, points[i].y);
            ctx.lineTo(points[j].x, points[j].y);
            ctx.stroke();
          }
        }
      }
      ctx.globalAlpha = 1;
      requestAnimationFrame(draw);
    }
    draw();

    const observer = new IntersectionObserver((entries) => {
      entries.forEach(entry => {
        if (entry.isIntersecting) {
          entry.target.classList.add('visible');
          if (entry.target.querySelector('.metric-value')) {
            animateCounters(entry.target);
          }
        }
      });
    }, { threshold: 0.1, rootMargin: '0px 0px -50px 0px' });

    function animateCounters(section) {
      const counters = section.querySelectorAll('.metric-value[data-count]');
      counters.forEach(counter => {
        const target = parseInt(counter.dataset.count);
        const duration = 1500;
        const start = performance.now();
        function update(currentTime) {
          const elapsed = currentTime - start;
          const progress = Math.min(elapsed / duration, 1);
          const eased = 1 - Math.pow(1 - progress, 3);
          counter.textContent = Math.floor(eased * target);
          if (progress < 1) requestAnimationFrame(update);
        }
        requestAnimationFrame(update);
      });
    }

    document.querySelectorAll('.reveal-on-scroll').forEach(el => observer.observe(el));
  </script>
