FROM ubuntu:22.04
ENV DEBIAN_FRONTEND=noninteractive
ENV APP_HOME=/app
ENV PORT=7860
RUN apt-get update && apt-get install -y \
    nginx \
    php8.1-fpm \
    php8.1-cli \
    php8.1-curl \
    php8.1-mbstring \
    php8.1-zip \
    ffmpeg \
    tesseract-ocr \
    build-essential \
    ca-certificates \
    curl \
    zip \
    unzip \
    python3 \
    python3-pip \
    && pip3 install --no-cache-dir yt-dlp \
    && rm -rf /var/lib/apt/lists/*
WORKDIR /app
COPY nginx.conf /etc/nginx/sites-available/default
COPY c/Makefile /app/Makefile
COPY c/videoextract.c /app/videoextract.c
COPY public /app/public
COPY api /app/api
RUN make
RUN mkdir -p /app/storage/uploads /app/storage/receipts /app/storage/generated /app/storage/dmgs
RUN chown -R www-data:www-data /app/storage /app/public /app/api
RUN mkdir -p /run/php && \
    sed -i 's|;listen.owner = www-data|listen.owner = www-data|' /etc/php/8.1/fpm/pool.d/www.conf && \
    sed -i 's|;listen.group = www-data|listen.group = www-data|' /etc/php/8.1/fpm/pool.d/www.conf
RUN rm -f /etc/nginx/sites-enabled/default && \
    ln -s /etc/nginx/sites-available/default /etc/nginx/sites-enabled/default
EXPOSE 7860
CMD service php8.1-fpm start && nginx -g "daemon off;"

local_extract.sh:
```
#!/bin/bash
# local_extract.sh — Extracts video evidence LOCALLY on your Mac
# Video never leaves your machine. Only the receipt JSON is sent to HF Space.
#
# Usage:
#   ./local_extract.sh /path/to/video.mp4 [format]
#   ./local_extract.sh "https://youtube.com/watch?v=XXX" [format]
#
# Requires: ffmpeg, tesseract (brew install ffmpeg tesseract)
# Optional: yt-dlp for YouTube (brew install yt-dlp)

set -e

HF_SPACE="https://josephrw-videoapp-compiler.hf.space"
FORMAT="${2:-evidence}"
WORKDIR="/tmp/videoapp_local_$$"

# Colors
ORANGE='\033[1;33m'
GREEN='\033[0;32m'
RED='\033[0;31m'
NC='\033[0m'

echo -e "${ORANGE}◉ VIDEOAPP LOCAL EXTRACTOR${NC}"
echo -e "${GREEN}◆ source video exposed: false · ⧖ receipt-only${NC}"
echo ""

mkdir -p "$WORKDIR/keyframes"
trap "rm -rf $WORKDIR" EXIT

INPUT="$1"
VIDEO_PATH=""

# ─── Check if YouTube URL ─────────────────────────────
if [[ "$INPUT" == *"youtube.com"* ]] || [[ "$INPUT" == *"youtu.be"* ]] || [[ "$INPUT" == *"yt://"* ]]; then
    echo -e "${ORANGE}⌁ YouTube URL detected — downloading locally...${NC}"

    if ! command -v yt-dlp &>/dev/null; then
        echo -e "${RED}✕ yt-dlp not found. Install: brew install yt-dlp${NC}"
        exit 1
    fi

    # Clean URL
    URL="${INPUT/yt\///https://www.youtube.com/watch?v=}"

    # Download best quality mp4, max 1080p
    yt-dlp -f "bestvideo[height<=1080][ext=mp4]+bestaudio[ext=m4a]/best[height<=1080][ext=mp4]/best" \
        --merge-output-format mp4 \
        -o "$WORKDIR/video.mp4" \
        "$URL" 2>&1 | tail -5

    VIDEO_PATH="$WORKDIR/video.mp4"
    VIDEO_NAME=$(yt-dlp --get-title "$URL" 2>/dev/null | head -1)
    if [ -z "$VIDEO_NAME" ]; then
        VIDEO_NAME="youtube_video"
    fi

    echo -e "${GREEN}◆ Downloaded: $VIDEO_NAME${NC}"

elif [[ -f "$INPUT" ]]; then
    VIDEO_PATH="$INPUT"
    VIDEO_NAME=$(basename "$INPUT")

else
    echo -e "${RED}✕ Input must be a file path or YouTube URL${NC}"
    echo "Usage: $0 /path/to/video.mp4 [format]"
    echo "       $0 \"https://youtube.com/watch?v=XXX\" [format]"
    exit 1
fi

# ─── Check dependencies ───────────────────────────────
for cmd in ffmpeg ffprobe tesseract; do
    if ! command -v $cmd &>/dev/null; then
        echo -e "${RED}✕ $cmd not found. Install: brew install ffmpeg tesseract${NC}"
        exit 1
    fi
done

# ─── Probe video ──────────────────────────────────────
echo -e "${ORANGE}⌁ Probing video metadata...${NC}"
DURATION=$(ffprobe -v quiet -show_entries format=duration -of csv=p=0 "$VIDEO_PATH" 2>/dev/null)
WIDTH=$(ffprobe -v quiet -select_streams v:0 -show_entries stream=width -of csv=p=0 "$VIDEO_PATH" 2>/dev/null)
HEIGHT=$(ffprobe -v quiet -select_streams v:0 -show_entries stream=height -of csv=p=0 "$VIDEO_PATH" 2>/dev/null)
FPS=$(ffprobe -v quiet -select_streams v:0 -show_entries stream=r_frame_rate -of csv=p=0 "$VIDEO_PATH" 2>/dev/null | awk -F/ '{if($2>0) printf "%.1f", $1/$2; else print "0"}')

echo "  Duration: ${DURATION}s | ${WIDTH}x${HEIGHT} | ${FPS}fps"

# ─── Extract keyframes ────────────────────────────────
echo -e "${ORANGE}⌁ Extracting keyframes (every 5s)...${NC}"
INTERVAL=5
MAX_FRAMES=50
FRAME_COUNT=0
TS=0

# JSON arrays
KEYFRAMES_JSON="["
OCR_JSON="["
ACTIONS_JSON="["
PROOF_JSON="["
FIRST_KF=true
FIRST_OCR=true
FIRST_PROOF=true

PREV_TYPE=""
PREV_TS=0
PREV_TEXT=""

while (( $(echo "$TS < $DURATION" | bc -l) )) && [ $FRAME_COUNT -lt $MAX_FRAMES ]; do
    IMG="$WORKDIR/keyframes/kf_${FRAME_COUNT}_ts${TS}.jpg"
    ffmpeg -ss "$TS" -i "$VIDEO_PATH" -frames:v 1 -q:v 2 -y "$IMG" 2>/dev/null

    if [ -f "$IMG" ]; then
        KF_NAME="kf_${FRAME_COUNT}_ts${TS}.jpg"

        if [ "$FIRST_KF" = true ]; then
            KEYFRAMES_JSON+="{\"path\":\"$KF_NAME\",\"timestamp\":$TS}"
            FIRST_KF=false
        else
            KEYFRAMES_JSON+=",{\"path\":\"$KF_NAME\",\"timestamp\":$TS}"
        fi

        # OCR
        OCR_TEXT=$(tesseract "$IMG" stdout 2>/dev/null | tr '\n' ' ' | sed 's/[[:space:]]*$//' | sed 's/"/\\"/g')

        if [ ${#OCR_TEXT} -gt 2 ]; then
            # Detect type
            TYPE="info"
            if echo "$OCR_TEXT" | grep -qiE "error|fail|exception|crash"; then TYPE="error"; fi
            if echo "$OCR_TEXT" | grep -qiE "success|pass|complete|done"; then TYPE="pass"; fi
            if echo "$OCR_TEXT" | grep -qiE "click|tap|select|open|close"; then TYPE="action"; fi

            if [ "$FIRST_OCR" = true ]; then
                OCR_JSON+="{\"timestamp\":$TS,\"text\":\"$OCR_TEXT\",\"type\":\"$TYPE\"}"
                ACTIONS_JSON+="{\"timestamp\":$TS,\"type\":\"$TYPE\",\"text\":\"$OCR_TEXT\"}"
                FIRST_OCR=false
            else
                OCR_JSON+=",{\"timestamp\":$TS,\"text\":\"$OCR_TEXT\",\"type\":\"$TYPE\"}"
                ACTIONS_JSON+=",{\"timestamp\":$TS,\"type\":\"$TYPE\",\"text\":\"$OCR_TEXT\"}"
            fi

            # Proof moment detection (error followed by pass)
            if [ "$PREV_TYPE" = "error" ] && [ "$TYPE" = "pass" ]; then
                if [ "$FIRST_PROOF" = true ]; then
                    PROOF_JSON+="{\"error_time\":$PREV_TS,\"success_time\":$TS,\"error_text\":\"$PREV_TEXT\",\"success_text\":\"$OCR_TEXT\"}"
                    FIRST_PROOF=false
                else
                    PROOF_JSON+=",{\"error_time\":$PREV_TS,\"success_time\":$TS,\"error_text\":\"$PREV_TEXT\",\"success_text\":\"$OCR_TEXT\"}"
                fi
            fi

            PREV_TYPE="$TYPE"
            PREV_TS="$TS"
            PREV_TEXT="$OCR_TEXT"
        fi

        FRAME_COUNT=$((FRAME_COUNT + 1))
        echo -ne "\r  Keyframes: $FRAME_COUNT / $MAX_FRAMES"
    fi

    TS=$(echo "$TS + $INTERVAL" | bc -l)
done

echo ""
KEYFRAMES_JSON+="]"
OCR_JSON+="]"
ACTIONS_JSON+="]"
PROOF_JSON+="]"

# ─── Extract audio ────────────────────────────────────
echo -e "${ORANGE}⌁ Extracting audio...${NC}"
ffmpeg -i "$VIDEO_PATH" -vn -acodec pcm_s16le -ar 16000 -ac 1 -y "$WORKDIR/audio.wav" 2>/dev/null
echo "  Audio extracted"

# ─── Transcribe audio (local whisper if available) ────
echo -e "${ORANGE}⟡ Transcribing audio...${NC}"
TRANSCRIPT=""
TRANSCRIPT_SOURCE="none"

if command -v whisper &>/dev/null; then
    echo "  Using local whisper..."
    whisper "$WORKDIR/audio.wav" --model tiny --output_format txt --output_dir "$WORKDIR" --language English 2>/dev/null
    if [ -f "$WORKDIR/audio.txt" ]; then
        TRANSCRIPT=$(cat "$WORKDIR/audio.txt" | tr '\n' ' ' | sed 's/[[:space:]]*$//')
        TRANSCRIPT_SOURCE="local_whisper"
    fi
elif [ -f "$HOME/.local/bin/whisper-ctranslate2" ]; then
    echo "  Using whisper-ctranslate2..."
    "$HOME/.local/bin/whisper-ctranslate2" "$WORKDIR/audio.wav" --model tiny --output_format txt --output_dir "$WORKDIR" 2>/dev/null
    if [ -f "$WORKDIR/audio.txt" ]; then
        TRANSCRIPT=$(cat "$WORKDIR/audio.txt" | tr '\n' ' ' | sed 's/[[:space:]]*$//')
        TRANSCRIPT_SOURCE="local_whisper_ct2"
    fi
fi

if [ -n "$TRANSCRIPT" ]; then
    echo "  Transcript: ${#TRANSCRIPT} chars"
    echo "  Preview: ${TRANSCRIPT:0:120}..."
else
    echo "  No local whisper found — HF Space will transcribe via HF Whisper API"
    echo "  Install: pip install openai-whisper (for local transcription)"
fi

# ─── Build receipt JSON ───────────────────────────────
echo -e "${ORANGE}⟡ Building receipt...${NC}"
VIDEO_ID=$(echo "$VIDEO_NAME" | md5 2>/dev/null | cut -c1-16 || echo "$(date +%s)" | md5 | cut -c1-16)

# Escape video name for JSON
VIDEO_NAME_ESC=$(echo "$VIDEO_NAME" | sed 's/"/\\"/g')

cat > "$WORKDIR/receipt.json" << RECEIPT
{
  "videoId": "$VIDEO_ID",
  "videoName": "$VIDEO_NAME_ESC",
  "videoPath": "LOCAL — not uploaded",
  "duration": $DURATION,
  "width": $WIDTH,
  "height": $HEIGHT,
  "fps": $FPS,
  "keyframe_count": $FRAME_COUNT,
  "keyframes": $KEYFRAMES_JSON,
  "ocr_samples": $OCR_JSON,
  "actions": $ACTIONS_JSON,
  "proof_moments": $PROOF_JSON,
  "files_detected": [],
  "urls_detected": [],
  "transcript": "$TRANSCRIPT",
  "transcript_source": "$TRANSCRIPT_SOURCE",
  "source_video_exposed": false,
  "created_at": $(date +%s)
}
RECEIPT

echo "  Receipt: $(wc -c < "$WORKDIR/receipt.json") bytes"
echo "  Keyframes: $FRAME_COUNT"
echo "  OCR blocks: $(echo "$OCR_JSON" | grep -o '"timestamp"' | wc -l | tr -d ' ')"
echo "  Actions: $(echo "$ACTIONS_JSON" | grep -o '"timestamp"' | wc -l | tr -d ' ')"
if [ -n "$TRANSCRIPT" ]; then
    echo "  Transcript: ${#TRANSCRIPT} chars ($TRANSCRIPT_SOURCE)"
fi

# ─── Send receipt to HF Space ─────────────────────────
echo ""
echo -e "${ORANGE}◉ Sending receipt to HF Space (video stays local)...${NC}"

RESPONSE=$(curl -s -X POST "$HF_SPACE/api/analyze_receipt.php" \
    -H "Content-Type: application/json" \
    -d "{\"receipt\": $(cat "$WORKDIR/receipt.json"), \"format\": \"$FORMAT\"}")

HF_ID=$(echo "$RESPONSE" | python3 -c "import sys,json; print(json.load(sys.stdin).get('id',''))" 2>/dev/null)

if [ -z "$HF_ID" ]; then
    echo -e "${RED}✕ Failed to send receipt to HF Space${NC}"
    echo "$RESPONSE"
    exit 1
fi

echo -e "${GREEN}◆ Receipt sent. HF ID: $HF_ID${NC}"

# ─── AI Analysis ──────────────────────────────────────
echo -e "${ORANGE}⟡ AI analysis on HF Space...${NC}"
ANALYZE_RESP=$(curl -s -X POST "$HF_SPACE/api/analyze.php" \
    -H "Content-Type: application/x-www-form-urlencoded" \
    -d "id=$HF_ID&format=$FORMAT")

echo "  Analysis complete"

# ─── Generate App ─────────────────────────────────────
echo -e "${ORANGE}◉ Generating app on HF Space...${NC}"
GEN_RESP=$(curl -s -X POST "$HF_SPACE/api/generate.php" \
    -H "Content-Type: application/x-www-form-urlencoded" \
    -d "id=$HF_ID&format=$FORMAT")

PREVIEW_URL=$(echo "$GEN_RESP" | python3 -c "import sys,json; print(json.load(sys.stdin).get('preview_url',''))" 2>/dev/null)

# ─── Package ──────────────────────────────────────────
echo -e "${ORANGE}⬇ Packaging...${NC}"
PKG_RESP=$(curl -s -X POST "$HF_SPACE/api/package.php" \
    -H "Content-Type: application/x-www-form-urlencoded" \
    -d "id=$HF_ID")

# ─── Download ─────────────────────────────────────────
echo -e "${ORANGE}⬇ Downloading generated app...${NC}"
DOWNLOAD_DIR="$HOME/Downloads/videoapp_generated"
mkdir -p "$DOWNLOAD_DIR"
ZIP_PATH="$DOWNLOAD_DIR/${HF_ID}.videoapp.zip"

curl -s -o "$ZIP_PATH" "$HF_SPACE/api/download.php?id=$HF_ID"

echo ""
echo -e "${GREEN}◆ DONE!${NC}"
echo ""
echo "  App zip:   $ZIP_PATH"
echo "  Preview:   $HF_SPACE$PREVIEW_URL"
echo "  Gallery:   $HF_SPACE"
echo ""
echo -e "${GREEN}◆ source video exposed: false · ⧖ receipt-only · ⟡ redacted${NC}"

# VideoGame From Video
Turn any video into a unique playable browser micro-game.
No Gradio. No Streamlit. No Python runtime.
Stack:
- C extraction binary
- PHP orchestration backend
- Vanilla JavaScript game engine
- Nginx on port 7860
- FFmpeg for frames/audio metadata
- Tesseract OCR
- Local Whisper for transcription
- Hugging Face Inference API for game design
Source video exposed: false.
Derived receipt only.

nginx.conf:
```
server {
    listen 7860;
    server_name _;
    root /app/public;
    index index.html;
    client_max_body_size 8000M;
    proxy_read_timeout 3600;
    fastcgi_read_timeout 3600;

    location / {
        try_files $uri $uri/ =404;
    }

    location /api/ {
        root /app;
        index index.php;
        location ~ \.php$ {
            include fastcgi_params;
            fastcgi_param SCRIPT_FILENAME $document_root$fastcgi_script_name;
            fastcgi_pass unix:/run/php/php8.1-fpm.sock;
        }
    }

    location /generated/ {
        alias /app/storage/generated/;
        autoindex off;
    }
}

// No frameworks. No build step. No dependencies.

(function () {
    'use strict';

    // ─── STATE ───

        { id: 'action', name: 'Action', glyph: '⌁', desc: 'Dodge enemies, collect targets. Mouse-driven survival.' },
        { id: 'strategy', name: 'Strategy', glyph: '⟡', desc: 'Plan moves, manage resources. Slower pace, deeper decisions.' },
        { id: 'simulation', name: 'Simulation', glyph: '◍', desc: 'Simulate what the video shows. Interactive sandbox.' },

    // ─── NAV ───
    document.querySelectorAll('.nav-btn').forEach(btn => {
        btn.addEventListener('click', () => {
            document.querySelectorAll('.nav-btn').forEach(b => b.classList.remove('active'));

            if (view === 'gallery') loadGallery();

        });

    function showView(name) {
        document.querySelectorAll('.view').forEach(v => v.classList.remove('active'));
        const el = document.getElementById('view-' + name);
        if (el) el.classList.add('active');
    }

    // ─── DROP ZONE ───
    const dropZone = document.getElementById('dropZone');
    const fileInput = document.getElementById('fileInput');

    dropZone.addEventListener('click', () => fileInput.click());

    dropZone.addEventListener('dragover', e => {
        e.preventDefault();
        dropZone.classList.add('drag-over');
    });
    dropZone.addEventListener('dragleave', () => {
        dropZone.classList.remove('drag-over');
    });
    dropZone.addEventListener('drop', e => {
        e.preventDefault();
        dropZone.classList.remove('drag-over');
        const file = e.dataTransfer.files[0];
        if (file) handleVideoFile(file);
    });

    fileInput.addEventListener('change', e => {
        const file = e.target.files[0];
        if (file) handleVideoFile(file);
    });

    // ─── YOUTUBE URL HANDLER ───
    const urlInput = document.getElementById('urlInput');
    const urlBtn = document.getElementById('urlBtn');

    urlBtn.addEventListener('click', () => {
        const url = urlInput.value.trim();
        if (!url) return;
        handleVideoURL(url);
    });

    urlInput.addEventListener('keypress', e => {
        if (e.key === 'Enter') {
            const url = urlInput.value.trim();
            if (url) handleVideoURL(url);
        }
    });

    function handleVideoURL(url) {
        if (!url.match(/^https?:\/\/(youtube\.com|youtu\.be|www\.youtube\.com)/)) {
            alert('Please enter a valid YouTube URL');
            return;
        }

        state.currentVideoName = url;
        document.getElementById('formatVideoName').textContent = '\u25B6 ' + url.substring(0, 60) + '...';

        showView('progress');
        setProgress('\u25C9 YOUTUBE', 'Sending URL to HF Space...', 'Starting download via yt-dlp', 5);

        fetch('/api/upload.php', {
            method: 'POST',
            headers: { 'Content-Type': 'application/x-www-form-urlencoded' },
            body: 'youtube_url=' + encodeURIComponent(url)
        })
            .then(r => r.json())
            .then(data => {
                if (!data.ok) throw new Error(data.error || 'URL submission failed');
                state.currentVideoId = data.id;
                pollYouTubeStatus(data.id);
            })
            .catch(err => {
                alert('Failed: ' + err.message);
                showView('drop');
            });
    }

    async function pollYouTubeStatus(id) {
        let attempts = 0;
        const maxAttempts = 120;

        while (attempts < maxAttempts) {
            await new Promise(r => setTimeout(r, 2000));
            attempts++;

            try {
                const r = await fetch('/api/status.php?id=' + id);
                const data = await r.json();

                if (data.status === 'downloaded' || data.ready) {
                    state.currentVideoName = (data.job && data.job.original_name) || 'youtube_video';
                    document.getElementById('formatVideoName').textContent = '\u25B6 ' + state.currentVideoName;
                    setProgress('\u25C6', 'Download complete!', 'Video ready for extraction', 100);
                    await new Promise(r => setTimeout(r, 500));
                    showFormatChooser();
                    return;
                }

                if (data.status === 'download_failed') {
                    alert('YouTube download failed. Try local mode: ./local_extract.sh "URL"');
                    showView('drop');
                    return;
                }

                setProgress('\u25C9 YOUTUBE', 'Downloading video...', 'Attempt ' + attempts + ' \u2014 waiting for yt-dlp', Math.min(attempts * 2, 50));
            } catch (e) {
                // keep trying
            }
        }

        alert('Download timed out. Try local mode: ./local_extract.sh "URL"');
        showView('drop');
    }

    function handleVideoFile(file) {
        const ext = file.name.split('.').pop().toLowerCase();
        const allowed = ['mp4', 'mov', 'm4v', 'webm', 'mkv'];
        if (!allowed.includes(ext)) {
            alert('Unsupported format. Use: ' + allowed.join(', '));
            return;
        }

        state.currentVideoName = file.name;
        document.getElementById('formatVideoName').textContent = '▶ ' + file.name;

        // Upload
        const formData = new FormData();
        formData.append('video', file);

        showView('progress');
        setProgress('◉ UPLOADING', file.name, 'Uploading to HF Space...', 10);

        fetch('/api/upload.php', { method: 'POST', body: formData })
            .then(r => r.json())
            .then(data => {
                if (!data.ok) throw new Error(data.error || 'Upload failed');
                state.currentVideoId = data.id;
                showFormatChooser();
            })
            .catch(err => {
                alert('Upload failed: ' + err.message);
                showView('drop');
            });
    }

    function showFormatChooser() {

            <div class="format-name">${f.name}</div>
            <div class="format-desc">${f.desc}</div>

            card.addEventListener('click', () => {
                state.selectedFormat = f.id;
                document.querySelectorAll('.format-card').forEach(c => c.classList.remove('selected'));
                card.classList.add('selected');
            });
            grid.appendChild(card);
        });
    }

    document.getElementById('generateBtn').addEventListener('click', runPipeline);
    document.getElementById('cancelFormatBtn').addEventListener('click', () => showView('drop'));

    // ─── PIPELINE ───
    async function runPipeline() {
        const id = state.currentVideoId;
        if (!id) return;

        const steps = [
            { label: 'Extract', glyph: '⌁', action: () => extractStep(id) },
            { label: 'Analyze', glyph: '⟡', action: () => analyzeStep(id) },
            { label: 'Generate', glyph: '◉', action: () => generateStep(id) },
            { label: 'Package', glyph: '⬇', action: () => packageStep(id) }
        ];

        showView('progress');
        renderSteps(steps, 0);

        for (let i = 0; i < steps.length; i++) {
            renderSteps(steps, i);
            setProgress(steps[i].glyph + ' ' + steps[i].label.toUpperCase(), 'Step ' + (i + 1) + ' of ' + steps.length, 'Processing...', ((i) / steps.length) * 100);
            try {
                const result = await steps[i].action();
                if (!result.ok) throw new Error(result.error || steps[i].label + ' failed');
                steps[i].result = result;
            } catch (err) {
                alert(steps[i].label + ' failed: ' + err.message);
                showView('format');
                return;
            }
        }

        renderSteps(steps, steps.length);
        setProgress('◆', 'COMPLETE', 'App generated successfully', 100);
        showResult(steps);
    }

    function renderSteps(steps, current) {

/*
 * videoextract.c — C video evidence extractor for VideoApp Compiler
 * Uses ffmpeg subprocess for frame extraction, tesseract for OCR.
 * No Python. No external runtime. Pure C.
 *
 * Usage:
 *   videoextract extract <video_path> <output_dir>
 *
 * Output: writes receipt.json + keyframe images to <output_dir>
 */

#include <stdio.h>
#include <stdlib.h>
#include <string.h>
#include <dirent.h>
#include <sys/stat.h>
#include <sys/types.h>
#include <unistd.h>
#include <math.h>
#include <time.h>

#define MAX_CMD 8192
#define MAX_OUTPUT 131072
#define MAX_KEYFRAMES 50
#define INTERVAL_SEC 5.0

static char* exec_cmd(const char* cmd) {
    FILE* fp = popen(cmd, "r");
    if (!fp) return strdup("");
    char* output = malloc(MAX_OUTPUT);
    if (!output) { pclose(fp); return strdup(""); }
    output[0] = '\0';
    char buf[4096];
    while (fgets(buf, sizeof(buf), fp)) {
        if (strlen(output) + strlen(buf) < MAX_OUTPUT - 1) {
            strcat(output, buf);
        }
    }
    pclose(fp);
    return output;
}

static void mkdir_p(const char* path) {
    char cmd[MAX_CMD];
    snprintf(cmd, sizeof(cmd), "mkdir -p '%s'", path);
    system(cmd);
}

/* Get video duration via ffprobe */
static double get_duration(const char* video_path) {
    char cmd[MAX_CMD];
    snprintf(cmd, sizeof(cmd),
        "ffprobe -v quiet -show_entries format=duration -of csv=p=0 '%s' 2>&1",
        video_path);
    char* out = exec_cmd(cmd);
    double d = atof(out);
    free(out);
    return d;
}

/* Get video dimensions via ffprobe */
static void get_dimensions(const char* video_path, int* w, int* h, double* fps) {
    char cmd[MAX_CMD];
    snprintf(cmd, sizeof(cmd),
        "ffprobe -v quiet -select_streams v:0 -show_entries stream=width,height,r_frame_rate -of csv=p=0 '%s' 2>&1",
        video_path);
    char* out = exec_cmd(cmd);
    *w = 0; *h = 0; *fps = 0;
    /* Parse "1920 1080 30/1" or similar */
    sscanf(out, "%d %d %lf/%*d", w, h, fps);
    if (*fps == 0) {
        /* Try alternate parse */
        char* tok = strtok(out, " ,\n");
        if (tok) *w = atoi(tok);
        tok = strtok(NULL, " ,\n");
        if (tok) *h = atoi(tok);
        tok = strtok(NULL, " ,\n");
        if (tok) {
            int num, den;
            if (sscanf(tok, "%d/%d", &num, &den) == 2 && den > 0)
                *fps = (double)num / den;
            else
                *fps = atof(tok);
        }
    }
    free(out);
}

/* Extract keyframes at regular intervals */
static int extract_keyframes(const char* video_path, const char* out_dir,
                             double duration, char kf_names[][256],
                             double kf_times[], int* kf_count) {
    mkdir_p(out_dir);

    int count = 0;
    double ts = 0.0;
    char img_path[MAX_CMD];

    while (ts < duration && count < MAX_KEYFRAMES) {
        snprintf(img_path, sizeof(img_path), "%s/keyframe_%04d_ts%.1f.jpg",
                 out_dir, count, ts);

        char cmd[MAX_CMD];
        snprintf(cmd, sizeof(cmd),
            "ffmpeg -ss %.2f -i '%s' -frames:v 1 -q:v 2 -y '%s' 2>/dev/null",
            ts, video_path, img_path);
        system(cmd);

        struct stat st;
        if (stat(img_path, &st) == 0 && st.st_size > 0) {
            snprintf(kf_names[count], 256, "keyframe_%04d_ts%.1f.jpg", count, ts);
            kf_times[count] = ts;
            count++;
        }
        ts += INTERVAL_SEC;
    }

    *kf_count = count;
    return 0;
}

/* OCR a single image using tesseract */
static char* ocr_image(const char* image_path) {
    char cmd[MAX_CMD];
    snprintf(cmd, sizeof(cmd), "tesseract '%s' stdout 2>/dev/null", image_path);
    char* out = exec_cmd(cmd);
    /* Trim trailing whitespace */
    char* end = out + strlen(out) - 1;
    while (end > out && (*end == '\n' || *end == '\r' || *end == ' ' || *end == '\t')) {
        *end = '\0';
        end--;
    }
    return out;
}

/* Detect event type from text */
static const char* detect_type(const char* text) {
    if (strstr(text, "error") || strstr(text, "Error") || strstr(text, "ERROR") ||
        strstr(text, "fail") || strstr(text, "Fail") || strstr(text, "FAIL") ||
        strstr(text, "exception") || strstr(text, "crash"))
        return "error";
    if (strstr(text, "success") || strstr(text, "Success") || strstr(text, "SUCCESS") ||
        strstr(text, "pass") || strstr(text, "Pass") || strstr(text, "PASS") ||
        strstr(text, "complete") || strstr(text, "done") || strstr(text, "Done"))
        return "pass";
    if (strstr(text, "click") || strstr(text, "Click") || strstr(text, "tap") ||
        strstr(text, "select") || strstr(text, "Select") || strstr(text, "open") ||
        strstr(text, "Open") || strstr(text, "close") || strstr(text, "Close"))
        return "action";
    return "info";
}

/* Write JSON-escaped string to file */
static void json_write_string(FILE* f, const char* s) {
    fputc('"', f);
    for (; *s; s++) {
        switch (*s) {
            case '"': fputs("\\\"", f); break;
            case '\\': fputs("\\\\", f); break;
            case '\n': fputs("\\n", f); break;
            case '\r': fputs("\\r", f); break;
            case '\t': fputs("\\t", f); break;
            default:
                if ((unsigned char)*s < 0x20) {
                    fprintf(f, "\\u%04x", (unsigned char)*s);
                } else {
                    fputc(*s, f);
                }
        }
    }
    fputc('"', f);
}

int main(int argc, char** argv) {
    if (argc < 4) {
        fprintf(stderr, "Usage: videoextract extract <video_path> <output_dir>\n");
        return 1;
    }

    const char* video_path = argv[2];
    const char* out_dir = argv[3];

    /* Create output dir */
    mkdir_p(out_dir);
    char kf_dir[MAX_CMD];
    snprintf(kf_dir, sizeof(kf_dir), "%s/keyframes", out_dir);

    /* Probe */
    double duration = get_duration(video_path);
    int width = 0, height = 0;
    double fps = 0;
    get_dimensions(video_path, &width, &height, &fps);

    /* Extract keyframes */
    char kf_names[MAX_KEYFRAMES][256];
    double kf_times[MAX_KEYFRAMES];
    int kf_count = 0;
    extract_keyframes(video_path, kf_dir, duration, kf_names, kf_times, &kf_count);

    /* OCR each keyframe */
    char ocr_texts[MAX_KEYFRAMES][4096];
    int ocr_count = 0;

    for (int i = 0; i < kf_count; i++) {
        char img_path[MAX_CMD];
        snprintf(img_path, sizeof(img_path), "%s/%s", kf_dir, kf_names[i]);
        char* text = ocr_image(img_path);
        if (text && strlen(text) > 2) {
            strncpy(ocr_texts[ocr_count], text, 4095);
            ocr_texts[ocr_count][4095] = '\0';
            ocr_count++;
        }
        free(text);
    }

    /* Extract audio for potential transcription */
    char audio_path[MAX_CMD];
    snprintf(audio_path, sizeof(audio_path), "%s/audio.wav", out_dir);
    char cmd[MAX_CMD];
    snprintf(cmd, sizeof(cmd),
        "ffmpeg -i '%s' -vn -acodec pcm_s16le -ar 16000 -ac 1 -y '%s' 2>/dev/null",
        video_path, audio_path);
    system(cmd);

    /* Write receipt.json */
    char receipt_path[MAX_CMD];
    snprintf(receipt_path, sizeof(receipt_path), "%s/receipt.json", out_dir);
    FILE* f = fopen(receipt_path, "w");
    if (!f) {
        fprintf(stderr, "Cannot write receipt.json\n");
        return 1;
    }

    fprintf(f, "{\n");
    fprintf(f, "  \"videoId\": \"%s\",\n", out_dir); /* will be overwritten by PHP */
    fprintf(f, "  \"duration\": %.2f,\n", duration);
    fprintf(f, "  \"width\": %d,\n", width);
    fprintf(f, "  \"height\": %d,\n", height);
    fprintf(f, "  \"fps\": %.2f,\n", fps);
    fprintf(f, "  \"keyframe_count\": %d,\n", kf_count);
    fprintf(f, "  \"ocr_count\": %d,\n", ocr_count);

    /* Keyframes array */
    fprintf(f, "  \"keyframes\": [\n");
    for (int i = 0; i < kf_count; i++) {
        fprintf(f, "    {\"path\": ");
        json_write_string(f, kf_names[i]);
        fprintf(f, ", \"timestamp\": %.1f}%s\n", kf_times[i], i < kf_count - 1 ? "," : "");
    }
    fprintf(f, "  ],\n");

    /* OCR samples array */
    fprintf(f, "  \"ocr_samples\": [\n");
    for (int i = 0; i < ocr_count; i++) {
        const char* type = detect_type(ocr_texts[i]);
        fprintf(f, "    {\"timestamp\": %.1f, \"text\": ", kf_times[i]);
        json_write_string(f, ocr_texts[i]);
        fprintf(f, ", \"type\": \"%s\"}%s\n", type, i < ocr_count - 1 ? "," : "");
    }
    fprintf(f, "  ],\n");

    /* Actions array (derived from OCR) */
    fprintf(f, "  \"actions\": [\n");
    for (int i = 0; i < ocr_count; i++) {
        const char* type = detect_type(ocr_texts[i]);
        fprintf(f, "    {\"timestamp\": %.1f, \"type\": \"%s\", \"text\": ", kf_times[i], type);
        json_write_string(f, ocr_texts[i]);
        fprintf(f, "}%s\n", i < ocr_count - 1 ? "," : "");
    }
    fprintf(f, "  ],\n");

    /* Proof moments (error→pass pairs) */
    fprintf(f, "  \"proof_moments\": [\n");
    int proof_count = 0;
    for (int i = 0; i < ocr_count - 1; i++) {
        const char* t1 = detect_type(ocr_texts[i]);
        if (strcmp(t1, "error") == 0) {
            /* Look for a pass within next 5 OCR blocks */
            for (int j = i + 1; j < ocr_count && j < i + 6; j++) {
                const char* t2 = detect_type(ocr_texts[j]);
                if (strcmp(t2, "pass") == 0) {
                    if (proof_count > 0) fprintf(f, ",\n");
                    fprintf(f, "    {\"error_time\": %.1f, \"success_time\": %.1f, \"error_text\": ",
                            kf_times[i], kf_times[j]);
                    json_write_string(f, ocr_texts[i]);
                    fprintf(f, ", \"success_text\": ");
                    json_write_string(f, ocr_texts[j]);
                    fprintf(f, "}");
                    proof_count++;
                    break;
                }
            }
        }
    }
    fprintf(f, "\n  ],\n");

    /* Files and URLs detected */
    fprintf(f, "  \"files_detected\": [\n");
    /* Simple regex-like scan for file patterns in OCR text */
    int file_count = 0;
    for (int i = 0; i < ocr_count; i++) {
        char* p = ocr_texts[i];
        while (*p) {
            /* Look for word.ext pattern */
            char* start = p;
            while (*p && (isalnum(*p) || *p == '-' || *p == '_')) p++;
            if (*p == '.' && isalpha(*(p+1))) {
                char* ext_start = p + 1;
                while (isalpha(*p)) p++;
                int ext_len = p - ext_start;
                if (ext_len >= 2 && ext_len <= 4) {
                    if (file_count > 0) fprintf(f, ",\n    ");
                    fprintf(f, "    ");
                    json_write_string(f, start);
                    file_count++;
                }
            }
            while (*p && !isalnum(*p) && *p != '-' && *p != '_') p++;
        }
    }
    fprintf(f, "\n  ],\n");

    fprintf(f, "  \"urls_detected\": [\n");
    int url_count = 0;
    for (int i = 0; i < ocr_count; i++) {
        char* p = strstr(ocr_texts[i], "http");
        while (p) {
            char* end = p;
            while (*end && !isspace(*end) && *end != '"' && *end != '\'' && *end != '<') end++;
            if (url_count > 0) fprintf(f, ",\n    ");
            fprintf(f, "    ");
            /* Write substring */
            fputc('"', f);
            for (char* c = p; c < end; c++) {
                if (*c == '"') fputs("\\\"", f);
                else if (*c == '\\') fputs("\\\\", f);
                else fputc(*c, f);
            }
            fputc('"', f);
            url_count++;
            p = strstr(end, "http");
        }
    }
    fprintf(f, "\n  ],\n");

    fprintf(f, "  \"source_video_exposed\": false,\n");
    fprintf(f, "  \"created_at\": %ld\n", (long)time(NULL));
    fprintf(f, "}\n");

    fclose(f);

    /* Print summary to stdout for PHP to capture */
    printf("EXTRACT_OK|keyframes=%d|ocr=%d|duration=%.1f\n", kf_count, ocr_count, duration);

    return 0;
}
