# Chat Room

Browse and view conversation history in a ChatGPT/Claude-style interface. Full access to all conversations without restrictions.

## Features

- **Browse Conversations** - View conversation list with summaries
- **Full Access** - No paywall or restrictions - view all conversations freely
- **ChatGPT/Claude UI** - Modern chat interface with message bubbles
- **Large File Support** - Optimized for large JSON files (47MB+) using streaming parser
- **Clean Design** - Dark theme inspired by popular AI chat interfaces

## How It Works

1. **Browse** - View all available conversations from the archive
2. **Read** - Click any conversation to view the full chat history
3. **Navigate** - Easy back navigation to the conversation list

## Technical Details

- **Flask** - Web framework
- **ijson** - Streaming JSON parser for large files
- **JSON Backend** - Conversations stored in conversations.json
- **Chat UI** - Message bubbles with avatars for human/assistant

## Deployment

This Space is configured for Docker deployment on Hugging Face Spaces.

def load_conversations_index():
    """Load just the index of conversations for listing with valuations."""
    conversations = []
    try:
        with open(CONVERSATIONS_FILE, 'r', encoding='utf-8') as f:
            for conv in ijson.items(f, 'item'):
                messages = conv.get('chat_messages', [])
                
                # Skip empty conversations
                if not messages or len(messages) == 0:
                    continue
                
                valuation = None
                topics = None
                
                if prompt_valuation:
                    valuation = prompt_valuation.calculate_total_valuation(messages)
                
                if topic_extractor:
                    topics = topic_extractor.extract_topics(messages)
                
                # Extract files from messages
                files = []
                for msg in messages:
                    text = msg.get('text', '')
                    # Look for file references in code blocks
                    import re
                    file_matches = re.findall(r'```[\w]*\n(?:#|//|/\*).*?file:\s*([^\s\n]+)', text, re.IGNORECASE)
                    files.extend(file_matches)
                    # Also look for common file patterns
                    file_matches = re.findall(r'[\w./-]+\.(py|js|ts|rs|go|java|cpp|c|h|html|css|json|yaml|yml|md|txt)', text)
                    files.extend(file_matches)
                
                # Remove duplicates
                files = list(set(files))[:10]  # Limit to 10 files
                
                # Classify systems
                system_analysis = None
                if system_classifier:
                    system_analysis = system_classifier.analyze_conversation(messages)
                
                conversations.append({
                    'uuid': conv.get('uuid'),
                    'name': conv.get('name', 'Untitled'),
                    'summary': conv.get('summary', ''),
                    'created_at': conv.get('created_at'),
                    'message_count': len(messages),
                    'valuation': valuation,
                    'topics': topics,
                    'files': files,
                    'system_analysis': system_analysis
                })
    except Exception as e:
        print(f"Error loading conversations: {e}")
    
    # Sort by valuation (highest first), then by message count
    conversations.sort(key=lambda x: (
        (x['valuation']['total_value'] if x['valuation'] else 0),
        x['message_count']
    ), reverse=True)
    
    return conversations

def load_conversation(conv_uuid):
    """Load a specific conversation by UUID."""
    try:
        with open(CONVERSATIONS_FILE, 'r', encoding='utf-8') as f:
            for conv in ijson.items(f, 'item'):
                if conv.get('uuid') == conv_uuid:
                    return conv
    except Exception as e:
        print(f"Error loading conversation {conv_uuid}: {e}")
    return None

def format_message(msg):
    """Format a chat message for display."""
    text = msg.get('text', '')
    # Apply code blurring if available
    if code_blur:
        text = code_blur.blur_text_with_code(text)
    
    return {
        'sender': msg.get('sender', 'unknown'),
        'text': text,
        'created_at': msg.get('created_at')
    }

@app.route('/conversation/<conv_uuid>')
def view_conversation(conv_uuid):
    """View a specific conversation."""
    start = time.time()
    conversation = load_conversation(conv_uuid)
    if not conversation:
        return render_template('error.html', message='Conversation not found'), 404
    
    messages = []
    for msg in conversation.get('chat_messages', []):
        formatted = format_message(msg)
        messages.append(formatted)
    
    # Reverse to show newest first (descending order)
    messages = messages[::-1]
    
    # Calculate total valuation
    total_valuation = None
    if prompt_valuation:
        total_valuation = prompt_valuation.calculate_total_valuation(conversation.get('chat_messages', []))
    
    # Extract topics
    topics = None
    if topic_extractor:
        topics = topic_extractor.extract_topics(conversation.get('chat_messages', []))
    
    if kpi_tracker:
        kpi_tracker.record_response_time(time.time() - start)
        kpi_tracker.record_interaction('view_conversation')
    
    return render_template('conversation.html', 
                          conversation=conversation,
                          messages=messages,
                          total_valuation=total_valuation,
                          topics=topics)

@app.route('/api/analyze', methods=['POST'])
def api_analyze():
    """Analyze prompt or code."""
    if not MODULES_AVAILABLE:
        return jsonify({'error': 'Analysis modules not available'})
    
    data = request.get_json()
    text = data.get('text', '')
    analysis_type = data.get('type', 'prompt')
    
    start = time.time()
    
    if analysis_type == 'prompt':
        result = prompt_analyzer.analyze_prompt(text)
    elif analysis_type == 'token_score':
        code = data.get('code', '')
        discussion = data.get('discussion', '')
        result = token_scorer.calculate_total_score(code, discussion)
    elif analysis_type == 'markov':
        markov = MarkovChain()
        result = markov.analyze_patterns(text)
    elif analysis_type == 'liquidity':
        code = data.get('code', '')
        discussion = data.get('discussion', '')
        result = liquidity_analyzer.calculate_liquidity_score(code, discussion)
    elif analysis_type == 'dialogue':
        messages = data.get('messages', [])
        result = dialogue_analyzer.analyze_dialogue(messages)
    elif analysis_type == 'token_suggestions':
        quality_score = data.get('quality_score', {})
        liquidity_score = data.get('liquidity_score', {})
        result = token_suggestions.generate_suggestions(quality_score, liquidity_score)
    elif analysis_type == 'valuation':
        sender = data.get('sender', 'human')
        result = prompt_valuation.appraise_prompt(text, sender)
    elif analysis_type == 'total_valuation':
        messages = data.get('messages', [])
        result = prompt_valuation.calculate_total_valuation(messages)
    else:
        result = {'error': 'Unknown analysis type'}
    
    if kpi_tracker:
        kpi_tracker.record_response_time(time.time() - start)
    if cost_tracker:
        cost_tracker.end_operation(tokens=len(text.split()))
    
    return jsonify(result)

@app.route('/api/ollama/chat', methods=['POST'])
def api_ollama_chat():
    """Chat with Ollama with code execution support and token tracking."""
    if not ollama_client:
        return jsonify({'error': 'Ollama client not available'})
    
    data = request.get_json()
    messages = data.get('messages', [])
    model = data.get('model', 'llama2')
    
    result = ollama_client.chat(messages, model)
    
    # Calculate cost based on real token usage
    # Pricing: $0.001 per 1K tokens (adjust based on actual pricing)
    TOKEN_PRICE_PER_1K = 0.001
    total_tokens = result.get('total_tokens', 0)
    cost = (total_tokens / 1000) * TOKEN_PRICE_PER_1K
    result['cost'] = round(cost, 6)
    result['tokens'] = total_tokens
    
    # Track in KPI tracker if available
    if kpi_tracker and total_tokens > 0:
        kpi_tracker.record_tokens(total_tokens)
        kpi_tracker.record_cost(cost)
    
    # Check if response contains code to execute
    if result.get('message') and result['message'].get('content'):
        content = result['message']['content']
        
        # Look for code blocks
        import re
        code_blocks = re.findall(r'```python\n(.*?)```', content, re.DOTALL)
        
        if code_blocks and local_executor:
            # Execute the first code block
            code = code_blocks[0]
            execution_result = local_executor.execute_code(code, {})
            
            # Append execution result to response
            if execution_result.get('stdout'):
                result['message']['content'] += f"\n\n**Execution Output:**\n```\n{execution_result['stdout']}\n```"
            if execution_result.get('stderr'):
                result['message']['content'] += f"\n\n**Execution Errors:**\n```\n{execution_result['stderr']}\n```"
    
    return jsonify(result)

            particle.style.left = Math.random() * 100 + '%';

            threshold: 0.1,
            rootMargin: '0px 0px -50px 0px'
        };

        const observer = new IntersectionObserver((entries) => {
            entries.forEach(entry => {
                if (entry.isIntersecting) {
                    entry.target.classList.add('active');
                }
            });
        }, observerOptions);

        document.querySelectorAll('.reveal').forEach(el => observer.observe(el));

        // Nav scroll effect
        window.addEventListener('scroll', () => {
            const nav = document.querySelector('nav');
            if (window.scrollY > 50) {
                nav.classList.add('scrolled');
            } else {
                nav.classList.remove('scrolled');
            }

        function appendToTerminal(text, type = 'normal') {
            const terminal = document.getElementById('terminal');

            line.className = 'terminal-line ' + type;
            line.textContent = text;
            terminal.appendChild(line);
            terminal.scrollTop = terminal.scrollHeight;

        async function executeStrategy() {
            const code = document.getElementById('strategyCode').value;
            const marketDataText = document.getElementById('marketData').value;
            
            let marketData = null;
            if (marketDataText.trim()) {
                try {
                    marketData = JSON.parse(marketDataText);
                } catch (e) {
                    appendToTerminal('Error parsing market data: ' + e.message, 'error');
                    return;
                }
            }

            appendToTerminal('Executing strategy...', 'normal');

            try {
                const response = await fetch('/api/execute', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ code, data: marketData })
                });

                const result = await response.json();

                if (result.success) {
                    appendToTerminal('✓ Execution successful', 'success');
                    if (result.stdout) {
                        result.stdout.split('\n').forEach(line => {
                            if (line.trim()) appendToTerminal(line);
                        });
                    }
                } else {
                    appendToTerminal('✗ Execution failed', 'error');
                    if (result.stderr) {
                        result.stderr.split('\n').forEach(line => {
                            if (line.trim()) appendToTerminal(line, 'error');
                        });
                    }
                }
            } catch (e) {
                appendToTerminal('Error: ' + e.message, 'error');
            }
        }

        async function runBacktest() {
            const code = document.getElementById('strategyCode').value;
            const marketDataText = document.getElementById('marketData').value;
            
            if (!marketDataText.trim()) {
                appendToTerminal('Please provide market data for backtesting', 'error');
                return;
            }

            try {
                const marketData = JSON.parse(marketDataText);
                const prices = marketData.map(d => d.price);
                
                // First execute to get signals
                appendToTerminal('Generating signals...', 'normal');
                
                const execResponse = await fetch('/api/execute', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ code, data: marketData })
                });

                const execResult = await execResponse.json();

                if (!execResult.success) {
                    appendToTerminal('Failed to generate signals', 'error');
                    return;
                }

                // Parse signals from output
                let signals = [];
                try {
                    const outputLines = execResult.stdout.split('\n');
                    for (const line of outputLines) {
                        if (line.includes('[')) {
                            signals = JSON.parse(line.match(/\[.*\]/)[0]);
                            break;
                        }
                    }
                } catch (e) {
                    appendToTerminal('Could not parse signals from output', 'error');
                    return;
                }

                // Run backtest
                appendToTerminal('Running backtest...', 'normal');
                
                const backtestResponse = await fetch('/api/backtest', {
                    method: 'POST',
                    headers: { 'Content-Type': 'application/json' },
                    body: JSON.stringify({ signals, prices })
                });

                const backtestResult = await backtestResponse.json();

                if (backtestResult.error) {
                    appendToTerminal('Backtest error: ' + backtestResult.error, 'error');
                    return;
                }

                // Display results
                appendToTerminal('✓ Backtest complete', 'success');
                appendToTerminal('Total Return: ' + backtestResult.total_return + '%');

        // Load KPI dashboard
        async function loadKPI() {
            try {
                const response = await fetch('/api/kpi');
                const data = await response.json();

                document.getElementById('kpi-response-time').textContent =
                    data.avg_response_time.toFixed(2) + 's';
                document.getElementById('kpi-tokens').textContent =
                    data.total_tokens.toLocaleString();
                document.getElementById('kpi-interactions').textContent =
                    data.total_interactions.toLocaleString();
                document.getElementById('kpi-cost').textContent =
                    '$' + data.total_cost.toFixed(4);
            } catch (e) {
                console.error('Failed to load KPI:', e);
            }
        }

        // Audio recording
        let mediaRecorder;
        let audioChunks = [];
        let isRecording = false;

        async function toggleRecording() {
            const btn = document.getElementById('recordBtn');
            const status = document.getElementById('recordingStatus');

            if (!isRecording) {
                try {
                    const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
                    mediaRecorder = new MediaRecorder(stream);
                    audioChunks = [];

                    mediaRecorder.ondataavailable = (event) => {
                        audioChunks.push(event.data);
                    };

                    mediaRecorder.onstop = () => {
                        const audioBlob = new Blob(audioChunks, { type: 'audio/wav' });
                        status.textContent = 'Recording saved! Size: ' + (audioBlob.size / 1024).toFixed(2) + ' KB';
                    };

                    mediaRecorder.start();
                    isRecording = true;
                    btn.textContent = 'Stop Recording';
                    btn.classList.add('recording');
                    status.textContent = 'Recording...';
                } catch (e) {
                    status.textContent = 'Error: ' + e.message;
                }
            } else {
                mediaRecorder.stop();
                isRecording = false;
                btn.textContent = 'Start Recording';
                btn.classList.remove('recording');
            }
        }

        // Load KPI on page load
        loadKPI();
        setInterval(loadKPI, 5000);

        // Load market data
        async function loadMarketData() {
            try {
                const response = await fetch('/api/market/prices');
                const data = await response.json();

                if (data.bitcoin) {
                    document.getElementById('btc-price').textContent = '$' + data.bitcoin.usd.toLocaleString();
                    const btcChange = data.bitcoin.usd_24h_change;
                    document.getElementById('btc-change').textContent = (btcChange > 0 ? '+' : '') + btcChange.toFixed(2) + '%';
                    document.getElementById('btc-change').className = 'change ' + (btcChange >= 0 ? 'positive' : 'negative');
                }

                if (data.ethereum) {
                    document.getElementById('eth-price').textContent = '$' + data.ethereum.usd.toLocaleString();
                    const ethChange = data.ethereum.usd_24h_change;
                    document.getElementById('eth-change').textContent = (ethChange > 0 ? '+' : '') + ethChange.toFixed(2) + '%';
                    document.getElementById('eth-change').className = 'change ' + (ethChange >= 0 ? 'positive' : 'negative');
                }

                if (data.solana) {
                    document.getElementById('sol-price').textContent = '$' + data.solana.usd.toLocaleString();
                    const solChange = data.solana.usd_24h_change;
                    document.getElementById('sol-change').textContent = (solChange > 0 ? '+' : '') + solChange.toFixed(2) + '%';
                    document.getElementById('sol-change').className = 'change ' + (solChange >= 0 ? 'positive' : 'negative');
                }
            } catch (e) {
                console.error('Failed to load market data:', e);
            }

            // Load sentiment
            try {
                const sentimentResponse = await fetch('/api/market/sentiment');
                const sentimentData = await sentimentResponse.json();

                if (sentimentData.fear_greed_index) {
                    document.getElementById('sentiment-value').textContent = sentimentData.fear_greed_index;
                    document.getElementById('sentiment-fill').style.width = sentimentData.fear_greed_index + '%';
                }
            } catch (e) {
                console.error('Failed to load sentiment:', e);
            }
        }

        loadMarketData();
        setInterval(loadMarketData, 30000); // Update every 5 seconds
    </script>

        function updateTime() {
            const now = new Date();

            document.getElementById('currentTime').textContent = timeStr;
            document.getElementById('currentTime2').textContent = timeStr;
        }

        updateTime();

        document.getElementById('quickAsk').addEventListener('keypress', function (e) {
            if (e.key === 'Enter') {
                const question = this.value;
                if (question.trim()) {
                    alert('Quick ask: ' + question);
                    this.value = '';
                }
            }
        });

        // Edit message functionality
        document.querySelectorAll('.edit-btn').forEach(btn => {
            btn.addEventListener('click', function (e) {
                e.stopPropagation();
                const messageId = this.getAttribute('data-message-id');
                editMessage(messageId);
            });
        });

        // TTS functionality
        let synthesis = window.speechSynthesis;
        let currentUtterance = null;

        document.querySelectorAll('.tts-btn').forEach(btn => {
            btn.addEventListener('click', function (e) {
                e.stopPropagation();
                const messageId = this.getAttribute('data-message-id');
                speakMessage(messageId);
            });
        });

        function speakMessage(messageId) {
            const messageText = document.getElementById(`message-text-${messageId}`);
            if (!messageText) return;

            const text = messageText.innerText || messageText.textContent;

            if (synthesis.speaking) {
                synthesis.cancel();
            }

            currentUtterance = new SpeechSynthesisUtterance(text);
            currentUtterance.rate = 1;
            currentUtterance.pitch = 1;
            currentUtterance.volume = 1;

            synthesis.speak(currentUtterance);
        }

        function editMessage(messageId) {
            const messageText = document.getElementById(`message-text-${messageId}`);
            const currentText = messageText.innerHTML;

            // Create textarea with current content
            const textarea = document.createElement('textarea');
            textarea.className = 'edit-textarea';
            textarea.value = currentText;

            // Create buttons
            const saveBtn = document.createElement('button');
            saveBtn.className = 'save-btn';
            saveBtn.textContent = 'Save';

            const cancelBtn = document.createElement('button');
            cancelBtn.className = 'cancel-btn';
            cancelBtn.textContent = 'Cancel';

            // Replace content with editor
            messageText.innerHTML = '';
            messageText.appendChild(textarea);
            messageText.appendChild(saveBtn);
            messageText.appendChild(cancelBtn);

            // Handle save
            saveBtn.onclick = function () {
                const newText = textarea.value;
                messageText.innerHTML = newText;
                alert('Message updated! (Note: Changes are not persisted to the JSON file)');
            };

            // Handle cancel
            cancelBtn.onclick = function () {
                messageText.innerHTML = currentText;
            };
        }
    </script>

        // Load KPI dashboard
        async function loadKPI() {
            try {
                const response = await fetch('/api/kpi');
                const data = await response.json();

                document.getElementById('kpi-response-time').textContent =
                    data.avg_response_time.toFixed(2) + 's';
                document.getElementById('kpi-tokens').textContent =
                    data.total_tokens.toLocaleString();
                document.getElementById('kpi-interactions').textContent =
                    data.total_interactions.toLocaleString();
                document.getElementById('kpi-cost').textContent =
                    '$' + data.total_cost.toFixed(4);
            } catch (e) {
                console.error('Failed to load KPI:', e);
            }
        }

        // Audio recording
        let mediaRecorder;
        let audioChunks = [];
        let isRecording = false;

        async function toggleRecording() {
            const btn = document.getElementById('recordBtn');
            const status = document.getElementById('recordingStatus');

            if (!isRecording) {
                try {
                    const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
                    mediaRecorder = new MediaRecorder(stream);
                    audioChunks = [];

                    mediaRecorder.ondataavailable = (event) => {
                        audioChunks.push(event.data);
                    };

                    mediaRecorder.onstop = () => {
                        const audioBlob = new Blob(audioChunks, { type: 'audio/wav' });
                        status.textContent = 'Recording saved! Size: ' + (audioBlob.size / 1024).toFixed(2) + ' KB';
                    };

                    mediaRecorder.start();
                    isRecording = true;
                    btn.textContent = 'Stop Recording';
                    btn.classList.add('recording');
                    status.textContent = 'Recording...';
                } catch (e) {
                    status.textContent = 'Error: ' + e.message;
                }
            } else {
                mediaRecorder.stop();
                isRecording = false;
                btn.textContent = 'Start Recording';
                btn.classList.remove('recording');
            }
        }

        // Load KPI on page load
        loadKPI();
        setInterval(loadKPI, 5000);

        // Load market data
        async function loadMarketData() {
            try {
                const response = await fetch('/api/market/prices');
                const data = await response.json();

                if (data.bitcoin) {
                    document.getElementById('btc-price').textContent = '$' + data.bitcoin.usd.toLocaleString();
                    const btcChange = data.bitcoin.usd_24h_change;
                    document.getElementById('btc-change').textContent = (btcChange > 0 ? '+' : '') + btcChange.toFixed(2) + '%';
                    document.getElementById('btc-change').className = 'change ' + (btcChange >= 0 ? 'positive' : 'negative');
                }

                if (data.ethereum) {
                    document.getElementById('eth-price').textContent = '$' + data.ethereum.usd.toLocaleString();
                    const ethChange = data.ethereum.usd_24h_change;
                    document.getElementById('eth-change').textContent = (ethChange > 0 ? '+' : '') + ethChange.toFixed(2) + '%';
                    document.getElementById('eth-change').className = 'change ' + (ethChange >= 0 ? 'positive' : 'negative');
                }

                if (data.solana) {
                    document.getElementById('sol-price').textContent = '$' + data.solana.usd.toLocaleString();
                    const solChange = data.solana.usd_24h_change;
                    document.getElementById('sol-change').textContent = (solChange > 0 ? '+' : '') + solChange.toFixed(2) + '%';
                    document.getElementById('sol-change').className = 'change ' + (solChange >= 0 ? 'positive' : 'negative');
                }
            } catch (e) {
                console.error('Failed to load market data:', e);
            }

            // Load sentiment
            try {
                const sentimentResponse = await fetch('/api/market/sentiment');
                const sentimentData = await sentimentResponse.json();

                if (sentimentData.fear_greed_index) {
                    document.getElementById('sentiment-value').textContent = sentimentData.fear_greed_index;
                    document.getElementById('sentiment-fill').style.width = sentimentData.fear_greed_index + '%';
                }
            } catch (e) {
                console.error('Failed to load sentiment:', e);
            }
        }

        loadMarketData();
        setInterval(loadMarketData, 30000); // Update every 5 seconds
    </script>

"""
Liquidity analysis for token launch potential
"""
import re
from typing import Dict, List
from datetime import datetime

class LiquidityAnalyzer:
    """Analyze liquidity potential from code and discussion"""
    
    def __init__(self):
        self.factors = {
            'market_interest': 0.3,
            'utility': 0.25,
            'community': 0.2,
            'partnerships': 0.15,
            'technical_quality': 0.1
        }
    
    def analyze_market_interest(self, discussion: str) -> float:
        """Analyze market interest indicators (0-10)"""
        score = 5.0
        
        # Trading indicators
        if re.search(r'trading|volume|liquidity|market cap', discussion, re.IGNORECASE):
            score += 2.0
        
        # Demand indicators
        if re.search(r'demand|supply|scarcity|rare', discussion, re.IGNORECASE):
            score += 1.5
        
        # Price indicators
        if re.search(r'price|value|worth|investment', discussion, re.IGNORECASE):
            score += 1.0
        
        # Exchange mentions
        if re.search(r'exchange|listing|dex|cex', discussion, re.IGNORECASE):
            score += 0.5
        
        return min(10.0, score)
    
    def analyze_utility(self, code: str, discussion: str) -> float:
        """Analyze utility/use case (0-10)"""
        score = 5.0
        
        # Functionality indicators
        if re.search(r'function|feature|capability|service', code, re.IGNORECASE):
            score += 1.5
        
        # Use case mentions
        if re.search(r'use case|application|solve|problem', discussion, re.IGNORECASE):
            score += 2.0
        
        # Integration potential
        if re.search(r'integration|api|sdk|plugin', code, re.IGNORECASE):
            score += 1.0
        
        # Practical application
        if re.search(r'practical|real.world|production', discussion, re.IGNORECASE):
            score += 0.5
        
        return min(10.0, score)
    
    def analyze_community(self, discussion: str) -> float:
        """Analyze community indicators (0-10)"""
        score = 5.0
        
        # Community mentions
        if re.search(r'community|users|adopters|followers', discussion, re.IGNORECASE):
            score += 2.0
        
        # Social indicators
        if re.search(r'twitter|discord|telegram|reddit', discussion, re.IGNORECASE):
            score += 1.5
        
        # Engagement
        if re.search(r'engagement|active|participate', discussion, re.IGNORECASE):
            score += 1.0
        
        # Growth
        if re.search(r'growth|expanding|scaling', discussion, re.IGNORECASE):
            score += 0.5
        
        return min(10.0, score)
    
    def analyze_partnerships(self, discussion: str) -> float:
        """Analyze partnership indicators (0-10)"""
        score = 5.0
        
        # Partnership mentions
        if re.search(r'partner|collaboration|joint', discussion, re.IGNORECASE):
            score += 2.0
        
        # Integration mentions
        if re.search(r'integration|ecosystem|network', discussion, re.IGNORECASE):
            score += 1.5
        
        # Institutional mentions
        if re.search(r'institution|enterprise|business', discussion, re.IGNORECASE):
            score += 1.0
        
        return min(10.0, score)
    
    def analyze_technical_quality(self, code: str) -> float:
        """Analyze technical quality (0-10)"""
        score = 5.0
        
        # Code organization
        if re.search(r'class|module|function', code, re.IGNORECASE):
            score += 1.0
        
        # Error handling
        if re.search(r'try|except|error|handle', code, re.IGNORECASE):
            score += 1.5
        
        # Testing
        if re.search(r'test|spec|assert', code, re.IGNORECASE):
            score += 1.0
        
        # Documentation
        if re.search(r'doc|comment|"""', code, re.IGNORECASE):
            score += 0.5
        
        return min(10.0, score)
    
    def calculate_liquidity_score(self, code: str, discussion: str) -> Dict:
        """Calculate overall liquidity score"""
        scores = {
            'market_interest': self.analyze_market_interest(discussion),
            'utility': self.analyze_utility(code, discussion),
            'community': self.analyze_community(discussion),
            'partnerships': self.analyze_partnerships(discussion),
            'technical_quality': self.analyze_technical_quality(code)
        }
        
        total = sum(scores[k] * self.factors[k] for k in scores)
        
        return {
            'total_score': total,
            'breakdown': scores,
            'liquidity_tier': self._get_liquidity_tier(total),
            'estimated_liquidity': self._estimate_liquidity(total),
            'timestamp': datetime.utcnow().isoformat()
        }
    
    def _get_liquidity_tier(self, score: float) -> str:
        """Get liquidity tier based on score"""
        if score >= 8.0:
            return "High Liquidity Potential"
        elif score >= 6.0:
            return "Moderate Liquidity Potential"
        elif score >= 4.0:
            return "Low Liquidity Potential"
        else:
            return "Very Low Liquidity Potential"
    
    def _estimate_liquidity(self, score: float) -> Dict:
        """Estimate liquidity metrics"""
        # Simplified estimation model
        base_liquidity = score * 10000  # Base in USD
        volatility = max(0.1, 1.0 - (score / 10))
        
        return {
            'estimated_daily_volume': int(base_liquidity),
            'estimated_market_depth': int(base_liquidity * 5),
            'volatility_index': round(volatility, 2),
            'confidence': 'medium' if 4 <= score <= 7 else 'high' if score > 7 else 'low'
        }

"""
Trading strategy backtesting module
"""
import numpy as np
from typing import Dict, List, Tuple
from datetime import datetime

class Backtester:
    """Simple backtesting engine for trading strategies"""
    
    def __init__(self, initial_capital: float = 100000):
        self.initial_capital = initial_capital
        self.trades = []
        self.equity_curve = []
    
    def run_backtest(self, signals: List[int], prices: List[float], 
                     fees: float = 0.001) -> Dict:
        """
        Run backtest with buy/sell signals
        signals: 1 = buy, -1 = sell, 0 = hold
        """
        capital = self.initial_capital
        position = 0
        equity = [capital]
        
        for i in range(1, len(prices)):
            signal = signals[i-1]
            price = prices[i]
            
            if signal == 1 and position == 0:  # Buy
                shares = (capital * (1 - fees)) / price
                position = shares
                capital = 0
                self.trades.append({
                    'type': 'buy',
                    'price': price,
                    'shares': shares,
                    'timestamp': i
                })
            elif signal == -1 and position > 0:  # Sell
                capital = position * price * (1 - fees)
                position = 0
                self.trades.append({
                    'type': 'sell',
                    'price': price,
                    'shares': self.trades[-1]['shares'],
                    'timestamp': i
                })
            
            # Calculate current equity
            current_equity = capital + (position * price if position > 0 else 0)
            equity.append(current_equity)
        
        # Close final position
        if position > 0:
            capital = position * prices[-1] * (1 - fees)
            equity[-1] = capital
        
        self.equity_curve = equity
        
        return self.calculate_metrics(equity, prices)
    
    def calculate_metrics(self, equity: List[float], prices: List[float]) -> Dict:
        """Calculate performance metrics"""
        returns = np.diff(equity) / equity[:-1]
        
        # Total return
        total_return = (equity[-1] - self.initial_capital) / self.initial_capital
        
        # Sharpe ratio (simplified)
        if len(returns) > 0:
            sharpe_ratio = np.mean(returns) / (np.std(returns) + 1e-10) * np.sqrt(252)
        else:
            sharpe_ratio = 0
        
        # Max drawdown
        peak = np.maximum.accumulate(equity)
        drawdown = (peak - equity) / peak
        max_drawdown = np.max(drawdown) if len(drawdown) > 0 else 0
        
        # Win rate
        winning_trades = [t for t in self.trades if t['type'] == 'sell']
        win_rate = 0
        if winning_trades:
            # Simplified win rate calculation
            win_rate = len(winning_trades) / max(len(self.trades), 1)
        
        # Profit factor
        gross_profit = sum(t['price'] * t['shares'] for t in self.trades if t['type'] == 'sell')
        gross_loss = sum(t['price'] * t['shares'] for t in self.trades if t['type'] == 'buy')
        profit_factor = gross_profit / (gross_loss + 1e-10)
        
        return {
            'total_return': round(total_return * 100, 2),
            'sharpe_ratio': round(sharpe_ratio, 2),
            'max_drawdown': round(max_drawdown * 100, 2),
            'win_rate': round(win_rate * 100, 2),
            'profit_factor': round(profit_factor, 2),
            'total_trades': len(self.trades),
            'final_equity': round(equity[-1], 2),
            'equity_curve': equity,
            'timestamp': datetime.utcnow().isoformat()
        }
    
    def generate_report(self) -> str:

        if re.search(r'(router|endpoint|@app\.route|express|fastapi|flask)', text):
            real_score += 2
        
        classification = 'toy'
        confidence = 0.5
        
        if real_score > toy_score:
            classification = 'real'
            confidence = min(0.95, 0.5 + (real_score - toy_score) * 0.1)
        elif toy_score > real_score:
            classification = 'toy'
            confidence = min(0.95, 0.5 + (toy_score - real_score) * 0.1)

            return f"def {func_name}(...):\n    # ... implementation ...\n"
        elif self.blur_level == 'medium':
            return f"def {func_name}(...):\n    ████ REDACTED ████\n"
        else:  # high
            return f"def {func_name}(...):\n    ████████████████████████████\n"

        result = code
        for pattern, replacement in self.SENSITIVE_PATTERNS:
            result = re.sub(pattern, replacement, result, flags=re.IGNORECASE)
        return result
    
    def blur_important_logic(self, code: str) -> str:
        """Blur important logic like function bodies"""
        result = code
        for pattern, replacement in self.LOGIC_PATTERNS:
            result = re.sub(pattern, replacement, result, flags=re.DOTALL)
        return result

        result = self.blur_sensitive_data(code)
        
        if self.blur_level in ['medium', 'high']:

class CostTracker:
    """Track compute and runtime costs"""
    
    # Approximate costs (USD per unit)
    COSTS = {
        'cpu_per_second': 0.00001,
        'memory_per_mb': 0.000001,
        'llm_per_1k_tokens': 0.002,
        'storage_per_gb': 0.0001
    }

        self.operations = []
        self.current_operation = None

        """Start tracking an operation"""

            'name': name,
            'start_time': time.time(),
            'start_datetime': datetime.utcnow().isoformat(),
            'metadata': metadata or {}
        }

        """End tracking and calculate cost"""
        if not self.current_operation:
            return None
        
        duration = time.time() - self.current_operation['start_time']
        
        cost = (
            duration * self.COSTS['cpu_per_second'] +
            memory_mb * self.COSTS['memory_per_mb'] +
            (tokens / 1000) * self.COSTS['llm_per_1k_tokens']
        )
        
        operation = {
            **self.current_operation,
            'end_time': time.time(),
            'end_datetime': datetime.utcnow().isoformat(),
            'duration': duration,
            'cpu_time': cpu_time,
            'memory_mb': memory_mb,
            'tokens': tokens,
            'cost': cost
        }
        
        self.operations.append(operation)
        self.current_operation = None
        return operation
    
    def get_total_cost(self) -> float:
        """Get total cost of all operations"""
        return sum(op['cost'] for op in self.operations)
    
    def get_cost_breakdown(self) -> Dict:
        """Get cost breakdown by operation type"""

        for op in self.operations:

            if name not in breakdown:
                breakdown[name] = {'count': 0, 'total_cost': 0, 'total_duration': 0}
            breakdown[name]['count'] += 1
            breakdown[name]['total_cost'] += op['cost']

"""
Merkle tree implementation for data integrity verification
"""
import hashlib
from typing import List, Optional

class MerkleNode:
    """Node in Merkle tree"""
    def __init__(self, hash: str, left: 'MerkleNode' = None, right: 'MerkleNode' = None):
        self.hash = hash
        self.left = left
        self.right = right

class MerkleTree:
    """Merkle tree for data integrity"""
    
    def __init__(self, data: List[str]):
        self.leaves = [self._hash(item) for item in data]
        self.root = self._build_tree(self.leaves)
    
    def _hash(self, data: str) -> str:
        """SHA-256 hash"""
        return hashlib.sha256(data.encode()).hexdigest()
    
    def _build_tree(self, leaves: List[str]) -> Optional[MerkleNode]:
        """Build Merkle tree from leaves"""
        if not leaves:
            return None
        if len(leaves) == 1:
            return MerkleNode(leaves[0])
        
        new_level = []
        for i in range(0, len(leaves), 2):
            left = leaves[i]
            right = leaves[i + 1] if i + 1 < len(leaves) else left
            combined = left + right
            new_level.append(self._hash(combined))
        
        return self._build_tree(new_level)
    
    def get_root_hash(self) -> Optional[str]:
        """Get root hash of the tree"""
        return self.root.hash if self.root else None
    
    def verify(self, data: str, proof: List[str], index: int) -> bool:
        """Verify data with Merkle proof"""
        current_hash = self._hash(data)
        for sibling_hash in proof:
            if index % 2 == 0:
                current_hash = self._hash(current_hash + sibling_hash)
            else:
                current_hash = self._hash(sibling_hash + current_hash)
            index //= 2
        return current_hash == self.get_root_hash()

"""
Markov chain analysis for pattern detection
"""
import random
from collections import defaultdict
from typing import Dict, List

class MarkovChain:
    """Simple Markov chain for text analysis"""
    
    def __init__(self, order: int = 2):
        self.order = order
        self.chain = defaultdict(lambda: defaultdict(int))
        self.starts = []
    
    def train(self, text: str):
        """Train the Markov chain on text"""
        words = text.split()
        if len(words) < self.order:
            return
        
        for i in range(len(words) - self.order):
            state = tuple(words[i:i + self.order])
            next_word = words[i + self.order]
            self.chain[state][next_word] += 1
            if i == 0:
                self.starts.append(state)
    
    def generate(self, length: int = 50) -> str:
        """Generate text from the chain"""
        if not self.starts:
            return ""
        
        current = random.choice(self.starts)
        result = list(current)
        
        for _ in range(length - self.order):
            if current not in self.chain:
                break
            next_words = list(self.chain[current].keys())
            weights = list(self.chain[current].values())
            next_word = random.choices(next_words, weights=weights)[0]
            result.append(next_word)
            current = tuple(result[-self.order:])
        
        return " ".join(result)
    
    def analyze_patterns(self, text: str) -> Dict:
        """Analyze patterns in text"""
        self.train(text)
        
        return {
            'unique_states': len(self.chain),
            'total_transitions': sum(sum(transitions.values()) for transitions in self.chain.values()),
            'avg_branching_factor': sum(len(transitions) for transitions in self.chain.values()) / len(self.chain) if self.chain else 0
        }

                ],

            )

                'timestamp': datetime.utcnow().isoformat(),
                'message_count': len(messages)

            print(f"LLM dialogue analysis failed: {e}")
            return self._basic_dialogue_analysis(messages)

        """Basic dialogue analysis without LLM"""

            return {'error': 'No messages to analyze'}

        human_messages = [m for m in messages if m.get('sender') == 'human']
        assistant_messages = [m for m in messages if m.get('sender') == 'assistant']
        
        avg_human_length = sum(len(m.get('text', '')) for m in human_messages) / len(human_messages) if human_messages else 0
        avg_assistant_length = sum(len(m.get('text', '')) for m in assistant_messages) / len(assistant_messages) if assistant_messages else 0
        
        # Simple coherence check
        coherence = 5.0
        if len(messages) > 2:
            coherence = min(10.0, 5.0 + (len(messages) / 10))
        
        return {
            'coherence': coherence,
            'relevance': 5.0,
            'clarity': 5.0,
            'engagement': min(10.0, len(messages) / 2),
            'overall_quality': 5.0,
            'human_message_count': len(human_messages),
            'assistant_message_count': len(assistant_messages),
            'avg_human_length': avg_human_length,
            'avg_assistant_length': avg_assistant_length,
            'timestamp': datetime.utcnow().isoformat(),
            'message_count': len(messages)
        }
    
    def suggest_improvements(self, messages: List[Dict]) -> List[str]:
        """Suggest improvements for the dialogue"""

        if len(human_messages) < len(assistant_messages):

        if avg_human_length < 50:
            suggestions.append("Human messages are brief - consider adding more detail")

"""

import numpy as np
from typing import Dict, List
import json

class AudioClassifier:
    """Classify audio sounds and transcribe speech"""
    
    def __init__(self):
        self.sound_types = {
            'speech': ['human', 'voice', 'speaking', 'talking', 'words'],
            'typing': ['keyboard', 'typing', 'click', 'clack'],
            'ambient': ['noise', 'background', 'silence', 'quiet'],
            'music': ['music', 'song', 'melody', 'rhythm'],
            'alert': ['alarm', 'notification', 'beep', 'ding']
        }
    
    def classify_audio(self, audio_data: np.ndarray, sample_rate: int) -> Dict:
        """Classify audio type (simplified - would need ML model in production)"""
        # Calculate basic audio features
        if len(audio_data) == 0:
            return {'type': 'silence', 'confidence': 0.9}
        
        # RMS energy
        rms = np.sqrt(np.mean(audio_data**2))
        
        # Zero crossing rate (speech has high ZCR)
        zero_crossings = np.sum(np.diff(np.sign(audio_data)) != 0)
        zcr = zero_crossings / len(audio_data)
        
        # Spectral centroid (speech has lower centroid than music)
        fft = np.fft.fft(audio_data)
        freqs = np.fft.fftfreq(len(audio_data), 1/sample_rate)
        magnitude = np.abs(fft)
        spectral_centroid = np.sum(freqs * magnitude) / np.sum(magnitude)
        
        # Classification based on features
        audio_type = 'ambient'
        confidence = 0.5
        
        if rms > 0.01 and zcr > 0.1:
            audio_type = 'speech'
            confidence = 0.7
        elif rms > 0.05 and zcr < 0.05:
            audio_type = 'music'
            confidence = 0.6
        elif rms > 0.02 and zcr > 0.15:
            audio_type = 'typing'
            confidence = 0.5
        elif rms < 0.005:
            audio_type = 'silence'
            confidence = 0.9
        
        return {
            'type': audio_type,
            'confidence': round(confidence, 2),
            'features': {
                'rms': round(rms, 4),
                'zcr': round(zcr, 4),
                'spectral_centroid': round(spectral_centroid, 2)
            }
        }
    
    def detect_speech_segments(self, audio_data: np.ndarray, sample_rate: int) -> List[Dict]:
        """Detect speech segments in audio"""
        segments = []
        chunk_size = sample_rate  # 1 second chunks
        is_speaking = False
        segment_start = 0
        
        for i in range(0, len(audio_data), chunk_size):
            chunk = audio_data[i:i+chunk_size]
            if len(chunk) < chunk_size:
                break
            
            classification = self.classify_audio(chunk, sample_rate)
            
            if classification['type'] == 'speech' and classification['confidence'] > 0.6:
                if not is_speaking:
                    is_speaking = True
                    segment_start = i
            else:
                if is_speaking:
                    is_speaking = False
                    segments.append({
                        'start': segment_start / sample_rate,
                        'end': i / sample_rate,
                        'duration': (i - segment_start) / sample_rate
                    })
        
        if is_speaking:
            segments.append({
                'start': segment_start / sample_rate,
                'end': len(audio_data) / sample_rate,
                'duration': (len(audio_data) - segment_start) / sample_rate
            })
        
        return segments

            'she', 'we', 'they', 'what', 'which', 'who', 'whom', 'this', 'that',
            'these', 'those', 'am', 'is', 'are', 'was', 'were', 'be', 'been',
            'being', 'have', 'has', 'had', 'having', 'do', 'does', 'did', 'doing'
        }
    
    def extract_topics(self, messages: List[Dict], top_n: int = 5) -> List[str]:
        """Extract top topics from messages"""
        all_text = ' '.join([msg.get('text', '') for msg in messages])
        
        # Extract words
        words = re.findall(r'\b[a-zA-Z]{3,}\b', all_text.lower())
        
        # Filter stop words
        words = [w for w in words if w not in self.stop_words]
        
        # Count frequency
        word_counts = Counter(words)
        
        # Get top topics
        top_words = word_counts.most_common(top_n)
        
        # Extract domain-specific terms
        domain_terms = self._extract_domain_terms(all_text)
        
        # Combine and return
        topics = [word for word, count in top_words]
        topics.extend(domain_terms)
        
        # Remove duplicates and limit
        seen = set()
        unique_topics = []
        for topic in topics:
            if topic not in seen:
                seen.add(topic)
                unique_topics.append(topic)
                if len(unique_topics) >= top_n:
                    break
        
        return unique_topics
    
    def _extract_domain_terms(self, text: str) -> List[str]:
        """Extract domain-specific terms"""
        terms = []
        
        # Tech/programming
        if re.search(r'(python|javascript|rust|go|java|cpp|react|vue|angular)', text, re.IGNORECASE):
            terms.extend(['programming', 'development'])
        
        # Trading/finance
        if re.search(r'(trading|strategy|backtest|portfolio|market|price|signal)', text, re.IGNORECASE):
            terms.extend(['trading', 'finance'])
        
        # AI/ML
        if re.search(r'(machine learning|neural|model|training|inference|llm)', text, re.IGNORECASE):
            terms.extend(['AI', 'machine learning'])
        
        # Blockchain/crypto
        if re.search(r'(blockchain|crypto|token|defi|nft|smart contract)', text, re.IGNORECASE):
            terms.extend(['blockchain', 'crypto'])
        
        return terms
    
    def get_conversation_summary(self, messages: List[Dict]) -> Dict:
        """Get conversation summary with topics and metadata"""
        topics = self.extract_topics(messages)
        
        # Calculate message stats
        total_messages = len(messages)
        human_messages = len([m for m in messages if m.get('sender') == 'human'])
        assistant_messages = len([m for m in messages if m.get('sender') == 'assistant'])
        
        # Time span
        timestamps = [m.get('created_at') for m in messages if m.get('created_at')]
        
        return {
            'topics': topics,
            'total_messages': total_messages,
            'human_messages': human_messages,
            'assistant_messages': assistant_messages,
            'timestamp': datetime.utcnow().isoformat()
        }

import subprocess

import tempfile
import os
from typing import Dict, List
from datetime import datetime

class LocalExecutor:
    """Execute trading strategies locally and return results"""
    
    def __init__(self):
        self.execution_history = []
    
    def execute_strategy(self, code: str, data: Dict = None) -> Dict:
        """
        Execute strategy code locally
        Returns output, metrics, and any errors
        """
        if data is None:
            data = {}
        
        # Create temporary file with strategy code
        with tempfile.NamedTemporaryFile(mode='w', suffix='.py', delete=False) as f:
            f.write(code)
            temp_file = f.name
        
        try:
            # Execute the code
            result = subprocess.run(
                ['python3', temp_file],
                capture_output=True,
                text=True,
                timeout=30
            )
            
            execution_record = {
                'code': code,
                'stdout': result.stdout,
                'stderr': result.stderr,
                'returncode': result.returncode,
                'success': result.returncode == 0,
                'timestamp': datetime.utcnow().isoformat()
            }
            
            self.execution_history.append(execution_record)
            
            return execution_record
            
        except subprocess.TimeoutExpired:
            return {
                'code': code,
                'stdout': '',
                'stderr': 'Execution timeout',
                'returncode': -1,
                'success': False,
                'timestamp': datetime.utcnow().isoformat()
            }
        finally:
            # Clean up temp file
            try:
                os.unlink(temp_file)
            except:
                pass
    
    def execute_with_data(self, code: str, data: List[Dict]) -> Dict:
        """
        Execute strategy with market data
        Data is passed as JSON to the strategy
        """
        # Create temp file with data
        with tempfile.NamedTemporaryFile(mode='w', suffix='.json', delete=False) as f:
            json.dump(data, f)

        # Modify code to load data
        modified_code = f"""
import json
with open('{data_file}', 'r') as f:
    market_data = json.load(f)

{code}
"""
        
        result = self.execute_strategy(modified_code)
        
        # Clean up data file
        try:
            os.unlink(data_file)
        except:
            pass
        
        return result
    
    def get_execution_history(self) -> List[Dict]:
        """Get recent execution history"""
        return self.execution_history[-10:]  # Last 10 executions

        """Clear execution history"""

"""
Provenance hashing module for data integrity and tracking
"""
import hashlib
import json
from datetime import datetime
from typing import Dict, Any

class ProvenanceTracker:
    """Track data provenance with cryptographic hashing"""
    
    def __init__(self):
        self.chain = []
    
    def hash_data(self, data: Any) -> str:
        """Create SHA-256 hash of data"""
        if isinstance(data, (dict, list)):
            data = json.dumps(data, sort_keys=True)
        return hashlib.sha256(str(data).encode()).hexdigest()
    
    def add_entry(self, data: Any, source: str, metadata: Dict = None) -> str:
        """Add an entry to the provenance chain"""
        timestamp = datetime.utcnow().isoformat()
        data_hash = self.hash_data(data)
        
        previous_hash = self.chain[-1]['hash'] if self.chain else '0' * 64
        
        entry = {
            'timestamp': timestamp,
            'source': source,
            'data_hash': data_hash,
            'previous_hash': previous_hash,
            'metadata': metadata or {}
        }
        
        entry_hash = self.hash_data(entry)
        entry['hash'] = entry_hash
        
        self.chain.append(entry)
        return entry_hash
    
    def verify_chain(self) -> bool:
        """Verify the integrity of the provenance chain"""
        for i, entry in enumerate(self.chain):
            if i > 0:
                if entry['previous_hash'] != self.chain[i-1]['hash']:
                    return False
            if entry['hash'] != self.hash_data(entry):
                return False
        return True
    
    def get_provenance(self, data_hash: str) -> Dict:
        """Get provenance info for a specific data hash"""
        for entry in self.chain:
            if entry['data_hash'] == data_hash:
                return entry
        return None

import time

class KPITracker:
    """Track and calculate live KPIs"""
    
    def __init__(self, window_size: int = 100):
        self.window_size = window_size
        self.metrics = {
            'response_times': deque(maxlen=window_size),
            'token_counts': deque(maxlen=window_size),
            'user_interactions': deque(maxlen=window_size),
            'errors': deque(maxlen=window_size),
            'costs': deque(maxlen=window_size)
        }

        """Record response time"""
        self.metrics['response_times'].append({
            'value': duration,
            'timestamp': datetime.utcnow().isoformat()
        })
    
    def record_token_count(self, count: int):
        """Record token count"""
        self.metrics['token_counts'].append({
            'value': count,
            'timestamp': datetime.utcnow().isoformat()
        })
    
    def record_interaction(self, interaction_type: str):
        """Record user interaction"""
        self.metrics['user_interactions'].append({
            'type': interaction_type,
            'timestamp': datetime.utcnow().isoformat()
        })
    
    def record_error(self, error_type: str):
        """Record error"""
        self.metrics['errors'].append({
            'type': error_type,
            'timestamp': datetime.utcnow().isoformat()
        })
    
    def record_cost(self, cost: float):
        """Record cost"""
        self.metrics['costs'].append({
            'value': cost,
            'timestamp': datetime.utcnow().isoformat()
        })
    
    def get_avg_response_time(self) -> float:
        """Get average response time"""
        if not self.metrics['response_times']:
            return 0.0
        return sum(m['value'] for m in self.metrics['response_times']) / len(self.metrics['response_times'])
    
    def get_total_tokens(self) -> int:
        """Get total tokens"""
        return sum(m['value'] for m in self.metrics['token_counts'])
    
    def get_interaction_rate(self) -> float:
        """Get interactions per minute"""
        if not self.metrics['user_interactions']:
            return 0.0
        duration_minutes = (time.time() - self.start_time) / 60
        return len(self.metrics['user_interactions']) / max(duration_minutes, 1)
    
    def get_error_rate(self) -> float:
        """Get error rate"""
        total = len(self.metrics['user_interactions'])
        errors = len(self.metrics['errors'])
        return (errors / total * 100) if total > 0 else 0.0
    
    def get_total_cost(self) -> float:
        """Get total cost"""
        return sum(m['value'] for m in self.metrics['costs'])
    
    def get_kpi_dashboard(self) -> Dict:
        """Get current KPI dashboard"""
        return {
            'avg_response_time': self.get_avg_response_time(),
            'total_tokens': self.get_total_tokens(),
            'interaction_rate': self.get_interaction_rate(),
            'error_rate': self.get_error_rate(),
            'total_cost': self.get_total_cost(),
            'uptime_seconds': time.time() - self.start_time,
            'total_interactions': len(self.metrics['user_interactions']),
            'total_errors': len(self.metrics['errors']),
            'timestamp': datetime.utcnow().isoformat()
