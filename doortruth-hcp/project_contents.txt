"""Immutable structured audit logging with integrity checks for institutional compliance."""

from __future__ import annotations

import hashlib
import json
import logging
import time
import uuid
from datetime import datetime
from pathlib import Path
from typing import Any

from config import Config
from models import AuditEvent


logger = logging.getLogger("verify.audit")


def _hash_ip(ip: str) -> str:
    """Privacy-preserving IP hash for audit trails."""
    return hashlib.sha256(ip.encode()).hexdigest()[:16]


def _compute_event_checksum(event: dict[str, Any]) -> str:
    """Compute SHA-256 checksum of canonical JSON for tamper detection."""
    canonical = json.dumps(event, sort_keys=True, ensure_ascii=True, default=str)
    return hashlib.sha256(canonical.encode()).hexdigest()


class AuditLogger:
    """Thread-safe append-only audit logger with checksums."""

    def __init__(self, path: Path = Config.AUDIT_LOG) -> None:
        self._path = path
        self._path.parent.mkdir(parents=True, exist_ok=True)

    def _append(self, event: dict[str, Any]) -> None:
        event["_checksum"] = _compute_event_checksum(event)
        event["_written_at"] = datetime.utcnow().isoformat()
        with open(self._path, "a", encoding="utf-8") as fh:
            fh.write(json.dumps(event, ensure_ascii=True, default=str) + "\n")

    def log(
        self,
        event_type: str,
        session_id: str,
        user_id: str = "anonymous",
        ip: str = "",
        record_name: str = "",
        inputs: dict[str, Any] | None = None,
        outputs: dict[str, Any] | None = None,
        duration_ms: int = 0,
        status: str = "OK",
        error_message: str = "",
    ) -> str:
        event_id = uuid.uuid4().hex
        event = AuditEvent(
            event_id=event_id,
            timestamp=datetime.utcnow(),
            event_type=event_type,
            session_id=session_id,
            user_id=user_id,
            ip_hash=_hash_ip(ip) if ip else "",
            record_name=record_name,
            inputs=inputs or {},
            outputs=outputs or {},
            duration_ms=duration_ms,
            status=status,
            error_message=error_message,
        )
        self._append(event.model_dump())
        return event_id

    def log_verification(
        self,
        session_id: str,
        record_name: str,
        result: dict[str, Any],
        duration_ms: int,
        user_id: str = "anonymous",
        ip: str = "",
    ) -> str:
        return self.log(
            event_type="VERIFY_SINGLE",
            session_id=session_id,
            user_id=user_id,
            ip=ip,
            record_name=record_name,
            outputs=result,
            duration_ms=duration_ms,
        )

    def log_batch_start(
        self,
        session_id: str,
        count: int,
        user_id: str = "anonymous",
        ip: str = "",
    ) -> str:
        return self.log(
            event_type="BATCH_START",
            session_id=session_id,
            user_id=user_id,
            ip=ip,
            inputs={"batch_size": count},
        )

    def log_batch_end(
        self,
        session_id: str,
        count: int,
        success_count: int,
        error_count: int,
        duration_ms: int,
        user_id: str = "anonymous",
        ip: str = "",
    ) -> str:
        return self.log(
            event_type="BATCH_END",
            session_id=session_id,
            user_id=user_id,
            ip=ip,
            outputs={"processed": count, "success": success_count, "errors": error_count},
            duration_ms=duration_ms,
        )

    def log_export(
        self,
        session_id: str,
        format: str,
        record_count: int,
        user_id: str = "anonymous",
        ip: str = "",
    ) -> str:
        return self.log(
            event_type="EXPORT",
            session_id=session_id,
            user_id=user_id,
            ip=ip,
            inputs={"format": format, "record_count": record_count},
        )

    def tail(self, n: int = 50) -> list[dict[str, Any]]:
        """Return the last n audit events (for admin dashboard)."""
        if not self._path.exists():
            return []
        lines = self._path.read_text(encoding="utf-8").strip().split("\n")
        events: list[dict[str, Any]] = []
        for line in lines[-n:]:
            try:
                events.append(json.loads(line))
            except json.JSONDecodeError:
                continue
        return events


# Global singleton
AUDIT = AuditLogger()


def get_rate_limit_key(session_id: str) -> str:
    return f"rl:{session_id}:{int(time.time()) // 60}"

class DoctorRecord(BaseModel):
    """Input record for a single doctor verification."""
    model_config = ConfigDict(str_strip_whitespace=True)

    name: str = Field(..., min_length=2, max_length=200, description="Full name of the doctor")
    address: str = Field(..., min_length=5, max_length=300, description="Street address")
    city: str = Field(..., min_length=2, max_length=100, description="City")
    state: str = Field(..., min_length=2, max_length=2, description="Two-letter state code")
    zip_code: str = Field(..., min_length=5, max_length=10, description="ZIP code")
    specialty: str = Field(default="", max_length=200, description="Medical specialty")
    target_flag: str = Field(default="", max_length=50)
    decile: str | int = Field(default="")
    gilead_id: str | int = Field(default="")
    territory: str = Field(default="", max_length=100, description="Rep territory (e.g. Bronx)")
    rep_id: str = Field(default="", max_length=50, description="Sales rep ID")
    product: str = Field(default="", max_length=100, description="Product/therapy segment")

    @field_validator("state")
    @classmethod
    def _uppercase_state(cls, v: str) -> str:
        return v.upper().strip()

    @field_validator("zip_code")
    @classmethod
    def _normalize_zip(cls, v: str) -> str:

        if len(v) == 9 and v.isdigit():

        return v

            "session_id": self.session_id,
            "checksum": self.checksum,

            prefix = src.source.lower().replace("_", "")
            flat[f"{prefix}_status"] = src.status.value
            flat[f"{prefix}_notes"] = src.notes
            if src.raw_address:
                flat[f"{prefix}_address"] = src.raw_address
            if src.url:
                flat[f"{prefix}_url"] = src.url
        return flat

"""Probabilistic record linkage using Splink for HCP deduplication."""

from __future__ import annotations

import logging
from typing import Any

import pandas as pd
from splink.duckdb.linker import DuckDBLinker
from splink.duckdb import DuckDBAPIConnection

logger = logging.getLogger("verify.deduplication")


def deduplicate_providers(df: pd.DataFrame) -> pd.DataFrame:
    """
    Deduplicate healthcare providers using Splink probabilistic linkage.
    
    Blocks on state+zip, compares name+address, exact-matches external_id when present.
    """
    if len(df) <= 1:
        return df
    
    # Ensure required columns exist
    required_cols = ["name", "state"]
    for col in required_cols:
        if col not in df.columns:
            df[col] = ""
    
    # Create a copy for deduplication
    df_dedup = df.copy()
    df_dedup["unique_id"] = range(len(df_dedup))
    
    # Configure Splink settings
    settings = {
        "link_type": "deduplicate_only",
        "proportion_of_matches": 0.01,  # Assume 1% are duplicates
        "blocking_rules_to_generate_predictions": [
            "l.state = r.state",
            "l.zip = r.zip",
        ],
        "comparisons": [
            {
                "comparison_description": "Name",
                "comparison_levels": [
                    {
                        "sql_condition": "name_l = name_r",
                        "label_for_charts": "Exact match",
                        "m_probability": 0.8,
                        "u_probability": 0.01,
                    },
                    {
                        "sql_condition": "levenshtein(name_l, name_r) <= 2",
                        "label_for_charts": "Fuzzy match",
                        "m_probability": 0.15,
                        "u_probability": 0.1,
                    },
                ],
            },
            {
                "comparison_description": "Address",
                "comparison_levels": [
                    {
                        "sql_condition": "address_l = address_r",
                        "label_for_charts": "Exact match",
                        "m_probability": 0.7,
                        "u_probability": 0.01,
                    },
                    {
                        "sql_condition": "levenshtein(address_l, address_r) <= 5",
                        "label_for_charts": "Fuzzy match",
                        "m_probability": 0.2,
                        "u_probability": 0.1,
                    },
                ],
            },
        ],
    }
    
    try:
        # Initialize Splink linker
        db_api = DuckDBAPIConnection()
        linker = DuckDBLinker(df_dedup, settings, db_api=db_api)
        
        # Train model
        linker.estimate_probability_two_random_records_match(["name", "address"])
        
        # Predict matches
        df_predictions = linker.predict()
        
        # Cluster using single linkage
        df_clustered = linker.cluster(df_predictions)
        
        # Get representative record for each cluster
        df_unique = df_clustered.sort_values("match_probability").groupby("cluster_id").first().reset_index()
        
        # Map back to original dataframe
        unique_ids = df_unique["unique_id"].tolist()
        df_result = df.iloc[unique_ids].copy()
        
        logger.info(f"Deduplicated {len(df)} records to {len(df_result)} unique providers")
        return df_result
        
    except Exception as e:
        logger.warning(f"Splink deduplication failed: {e}, falling back to exact matching")
        # Fallback to simple exact matching on name + state + zip
        df_result = df.drop_duplicates(subset=["name", "state", "zip"], keep="first")
        logger.info(f"Fallback deduplication: {len(df)} to {len(df_result)} records")
        return df_result

"""Rate limiter for external API calls to prevent blocking."""

import asyncio
import time
from collections import deque
from typing import Optional


class RateLimiter:
    """Token bucket rate limiter for API calls."""
    
    def __init__(self, rate: float, burst: int = 10):
        """
        Args:
            rate: Requests per second
            burst: Maximum burst size (default 10)
        """
        self.rate = rate
        self.burst = burst
        self.tokens = burst
        self.last_update = time.time()
        self._lock = asyncio.Lock()
    
    async def acquire(self) -> None:
        """Acquire a token, waiting if necessary."""
        async with self._lock:
            now = time.time()
            elapsed = now - self.last_update
            self.tokens = min(self.burst, self.tokens + elapsed * self.rate)
            self.last_update = now
            
            if self.tokens < 1:
                wait_time = (1 - self.tokens) / self.rate
                await asyncio.sleep(wait_time)
                self.tokens = 0
            else:
                self.tokens -= 1


# Global rate limiters for different APIs
_nppes_limiter: Optional[RateLimiter] = None

from __future__ import annotations

import json
import logging
import sqlite3
from datetime import datetime, timedelta
from pathlib import Path
from typing import Any, Optional

logger = logging.getLogger("verify.cache")

CACHE_DB = Path("cache.db")
CACHE_TTL_DAYS = 7  # Cache results for 7 days


def init_cache() -> None:
    """Initialize SQLite cache database."""
    conn = sqlite3.connect(CACHE_DB)
    cursor = conn.cursor()
    
    cursor.execute("""
        CREATE TABLE IF NOT EXISTS cache (
            key TEXT PRIMARY KEY,
            value TEXT,
            created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP
        )
    """)
    
    conn.commit()
    conn.close()
    logger.info("Cache initialized")


def get_cached(key: str) -> Optional[dict[str, Any]]:
    """Retrieve cached result if it exists and is not expired."""
    conn = sqlite3.connect(CACHE_DB)
    cursor = conn.cursor()
    
    cursor.execute("SELECT value, created_at FROM cache WHERE key = ?", (key,))
    row = cursor.fetchone()
    
    conn.close()
    
    if not row:
        return None
    
    value, created_at_str = row
    created_at = datetime.fromisoformat(created_at_str)
    
    # Check if cache is expired
    if datetime.now() - created_at > timedelta(days=CACHE_TTL_DAYS):
        logger.debug(f"Cache expired for key: {key}")
        return None
    
    try:
        return json.loads(value)
    except json.JSONDecodeError:
        logger.warning(f"Failed to parse cached value for key: {key}")
        return None


def set_cached(key: str, value: dict[str, Any]) -> None:
    """Store result in cache."""
    conn = sqlite3.connect(CACHE_DB)
    cursor = conn.cursor()
    
    cursor.execute(
        "INSERT OR REPLACE INTO cache (key, value, created_at) VALUES (?, ?, ?)",
        (key, json.dumps(value), datetime.now().isoformat())
    )
    
    conn.commit()
    conn.close()
    logger.debug(f"Cached result for key: {key}")


def clear_cache() -> None:
    """Clear all cached entries."""
    conn = sqlite3.connect(CACHE_DB)
    cursor = conn.cursor()
    
    cursor.execute("DELETE FROM cache")
    
    conn.commit()
    conn.close()
    logger.info("Cache cleared")


def cleanup_expired() -> None:
    """Remove expired cache entries."""
    conn = sqlite3.connect(CACHE_DB)
    cursor = conn.cursor()
    
    cutoff = (datetime.now() - timedelta(days=CACHE_TTL_DAYS)).isoformat()
    cursor.execute("DELETE FROM cache WHERE created_at < ?", (cutoff,))
    
    deleted = cursor.rowcount
    conn.commit()
    conn.close()
    
    if deleted > 0:
        logger.info(f"Cleaned up {deleted} expired cache entries")

# ---------------------------------------------------------------------------
# SQLite cache
# ---------------------------------------------------------------------------
class PageCache:
    def __init__(self, db_path: Path = Config.CACHE_DB) -> None:
        self._path = db_path
        self._ensure_schema()

    def _conn(self) -> sqlite3.Connection:
        return sqlite3.connect(str(self._path), timeout=10)

    def _ensure_schema(self) -> None:
        with self._conn() as conn:
            conn.execute(
                "CREATE TABLE IF NOT EXISTS page_cache ("
                "url TEXT PRIMARY KEY, doctor TEXT, status TEXT, notes TEXT, created INTEGER"
                ")"
            )
            conn.execute("CREATE INDEX IF NOT EXISTS idx_cache_created ON page_cache(created)")
            conn.commit()

    def get(self, url: str, doctor_name: str) -> SourceResult | None:
        cutoff = int(time.time()) - 86400
        with self._conn() as conn:
            row = conn.execute(
                "SELECT status, notes FROM page_cache WHERE url=? AND doctor=? AND created > ?",
                (url, doctor_name, cutoff),
            ).fetchone()
        if row:
            return SourceResult(source="CACHE", status=VerificationStatus(row[0]), notes=row[1])
        return None

    def set(self, url: str, doctor_name: str, status: str, notes: str) -> None:
        with self._conn() as conn:
            conn.execute(
                "INSERT OR REPLACE INTO page_cache (url, doctor, status, notes, created) VALUES (?, ?, ?, ?, ?)",
                (url, doctor_name, status, notes, int(time.time())),
            )
            conn.commit()

    def size(self) -> int:
        with self._conn() as conn:
            row = conn.execute("SELECT COUNT(*) FROM page_cache").fetchone()
        return row[0] if row else 0

# Phone extraction

_PHONE_RE = re.compile(r"\(?\d{3}\)?[-.\s]?\d{3}[-.\s]?\d{4}")


def extract_phones(text: str) -> list[str]:
    found = _PHONE_RE.findall(text)
    cleaned: set[str] = set()
    for p in found:
        digits = re.sub(r"\D", "", p)
        if len(digits) == 10:
            cleaned.add(f"({digits[:3]}) {digits[3:6]}-{digits[6:]}")
    return sorted(cleaned)

def generate_search_query(record: DoctorRecord) -> str:
    if Config.NO_OLLAMA or not _OLLAMA_AVAILABLE:
        return f'"{record.name}" doctor "{record.address}" {record.city} {record.state}'
    prompt = (
        f"Generate a concise web search query to find whether this doctor practices at this address.\n"
        f"Doctor: {record.name}\n"
        f"Address: {record.address}, {record.city}, {record.state}\n"
        f"Specialty: {record.specialty}\n\n"
        f"Rules:\n"
        f"- Use the doctor's full name in quotes\n"
        f"- Include the street address in quotes\n"
        f"- Keep it under 120 characters\n"
        f"- Do NOT include ZIP code (reduces hits)\n"
        f"- Respond ONLY with the query string, no explanation.\n"
    )
    try:
        response = ollama.chat(
            model=Config.OLLAMA_MODEL,
            messages=[{"role": "user", "content": prompt}],
            options={"temperature": 0.1, "num_predict": 100},
        )
        query = response["message"]["content"].strip().strip('"')
        if query and len(query) > 10:
            return query
    except Exception as exc:
        logger.debug("LLM query generation failed: %s", exc)
    return f'"{record.name}" doctor "{record.address}" {record.city} {record.state}'


# ---------------------------------------------------------------------------
# Web page fetching
# ---------------------------------------------------------------------------
def extract_search_links(html: str, max_links: int = 5) -> list[str]:
    soup = BeautifulSoup(html, "lxml")
    links: list[str] = []
    for a in soup.find_all("a", class_="result__a"):
        href = a.get("href", "")
        if href.startswith("//"):
            href = "https:" + href
        if href.startswith("http") and "duckduckgo.com" not in href:
            links.append(href)
        if len(links) >= max_links:
            break
    return links


def fetch_page_text(url: str) -> str:
    try:
        with httpx.Client(timeout=20, follow_redirects=True) as client:
            resp = client.get(url, headers=Config.HEADERS)
            resp.raise_for_status()
            soup = BeautifulSoup(resp.text, "lxml")
            for tag in soup(["script", "style", "nav", "footer", "header"]):
                tag.decompose()
            text = soup.get_text(separator=" ", strip=True)
            return " ".join(text.split()[:3000])
    except Exception as exc:
        return f"[FETCH_ERROR: {exc}]"

# LLM page analysis
# ---------------------------------------------------------------------------
def analyze_page(page_text: str, record: DoctorRecord) -> dict[str, Any]:
    if Config.NO_OLLAMA or not _OLLAMA_AVAILABLE:
        return {"doctor_found": False, "address_match": False, "confidence": "Low", "reasoning": "Ollama disabled"}
    prompt = (
        "You are a medical-address verification analyst. Analyze the webpage text below.\n\n"
        f"Doctor: {record.name}\n"
        f"Target address: {record.address}, {record.city}, {record.state}\n\n"
        f"Webpage text (first 2000 chars):\n{page_text[:2000]}\n\n"
        "Answer ONLY with valid JSON:\n"
        '{"doctor_found": true/false, "address_match": true/false, "confidence": "High|Medium|Low", "reasoning": "short explanation"}'
    )
    try:
        response = ollama.chat(
            model=Config.OLLAMA_MODEL,
            messages=[{"role": "user", "content": prompt}],
            options={"temperature": 0.05, "num_predict": 300},
        )
        text = response["message"]["content"]
        for delim in ("```json", "```"):
            if delim in text:
                text = text.split(delim)[1].split("```")[0]
                break
        parsed = json.loads(text.strip())
        return {
            "doctor_found": bool(parsed.get("doctor_found", False)),
            "address_match": bool(parsed.get("address_match", False)),
            "confidence": str(parsed.get("confidence", "Low")),
            "reasoning": str(parsed.get("reasoning", "")),
        }
    except Exception as exc:
        logger.warning("LLM page analysis failed for %s: %s", record.name, exc)
        return {"doctor_found": False, "address_match": False, "confidence": "Low", "reasoning": f"LLM error: {exc}"}


# ---------------------------------------------------------------------------
# Full web crawl verifier
# ---------------------------------------------------------------------------
def web_search_verify(record: DoctorRecord, npi_status: str = "") -> SourceResult:
    if npi_status == "CONFIRMED":
        return SourceResult(source="WEB_CRAWL", status=VerificationStatus.SKIPPED, notes="NPI already confirmed")

    query = generate_search_query(record)
    encoded = urllib.parse.quote(query)
    search_url = f"https://html.duckduckgo.com/html/?q={encoded}"
    try:
        with httpx.Client(timeout=20, follow_redirects=True) as client:
            resp = client.get(search_url, headers=Config.HEADERS)
            resp.raise_for_status()
            links = extract_search_links(resp.text, max_links=Config.WEB_CRAWL_MAX_LINKS)
    except Exception as exc:
        return SourceResult(source="WEB_CRAWL", status=VerificationStatus.ERROR, notes=f"DDG search failed: {exc}")

    if not links:
        return SourceResult(source="WEB_CRAWL", status=VerificationStatus.NOT_FOUND, notes="No DDG result links")

    confirmations = 0
    mismatches = 0
    best_url = ""
    best_reasoning = ""
    link_notes: list[str] = []
    all_phones: set[str] = set()

    for link in links:
        cached = _CACHE.get(link, record.name)
        if cached:
            if cached.status == VerificationStatus.CONFIRMED:
                confirmations += 1
            elif cached.status == VerificationStatus.MISMATCH:
                mismatches += 1
            link_notes.append(f"{link[:60]}… [cached {cached.status.value}]")
            continue

        page_text = fetch_page_text(link)
        if page_text.startswith("[FETCH_ERROR"):
            link_notes.append(f"{link[:60]}… fetch error")
            continue

        analysis = analyze_page(page_text, record)
        phones = extract_phones(page_text)
        all_phones.update(phones)
        phone_str = f" Phones: {', '.join(phones)}" if phones else ""

        if analysis["doctor_found"] and analysis["address_match"] and analysis["confidence"] in ("High", "Medium"):
            status = VerificationStatus.CONFIRMED
            confirmations += 1
            best_url = link
            best_reasoning = analysis["reasoning"]
        elif analysis["doctor_found"] and not analysis["address_match"]:
            status = VerificationStatus.MISMATCH
            mismatches += 1
        else:
            status = VerificationStatus.NOT_FOUND

        link_notes.append(f"{link[:60]}… [{status.value}] {analysis['reasoning'][:80]}{phone_str}")
        _CACHE.set(link, record.name, status.value, analysis["reasoning"])

    phone_summary = f" Phones found: {', '.join(sorted(all_phones))}" if all_phones else ""

    if confirmations >= 1 and mismatches == 0:
        return SourceResult(source="WEB_CRAWL", status=VerificationStatus.CONFIRMED, url=best_url, notes=f"{confirmations}/2 links confirm. {best_reasoning}{phone_summary}")
    if confirmations >= 1 and mismatches >= 1:
        return SourceResult(source="WEB_CRAWL", status=VerificationStatus.MISMATCH, url=best_url, notes=f"Mixed: {confirmations} confirm, {mismatches} mismatch. {best_reasoning}")
    if mismatches >= 1:
        return SourceResult(source="WEB_CRAWL", status=VerificationStatus.MISMATCH, notes=f"{mismatches}/2 links found doctor at different address. {' | '.join(link_notes)}")
    return SourceResult(source="WEB_CRAWL", status=VerificationStatus.NOT_FOUND, notes=f"No confirmation. {' | '.join(link_notes)}")

# ---------------------------------------------------------------------------
# Retry decorator
# ---------------------------------------------------------------------------
import random


def _retry_http(max_retries: int = 3, base_delay: float = 1.0):
    def decorator(func):
        if asyncio.iscoroutinefunction(func):
            async def async_wrapper(*args, **kwargs):
                last_exc = None
                for attempt in range(max_retries):
                    try:
                        return await func(*args, **kwargs)
                    except Exception as exc:
                        last_exc = exc
                        delay = base_delay * (2 ** attempt) + random.random()
                        logger.debug("Retry %s attempt %d/%d: %s", func.__name__, attempt + 1, max_retries, exc)
                        await asyncio.sleep(delay)
                logger.warning("%s failed after %d retries: %s", func.__name__, max_retries, last_exc)
                raise last_exc
            return async_wrapper
        else:
            def sync_wrapper(*args, **kwargs):
                last_exc = None
                for attempt in range(max_retries):
                    try:
                        return func(*args, **kwargs)
                    except Exception as exc:
                        last_exc = exc
                        delay = base_delay * (2 ** attempt) + random.random()
                        logger.debug("Retry %s attempt %d/%d: %s", func.__name__, attempt + 1, max_retries, exc)
                        time.sleep(delay)
                logger.warning("%s failed after %d retries: %s", func.__name__, max_retries, last_exc)
                raise last_exc
            return sync_wrapper
    return decorator

import re

_NORMALIZE_RE = re.compile(r"[^\w\s]")


def _normalize_address(addr: str) -> str:
    addr = addr.upper()
    for old, new in (("STREET", "ST"), ("AVENUE", "AVE"), ("BOULEVARD", "BLVD"),
                     ("ROAD", "RD"), ("DRIVE", "DR"), ("SUITE", "STE"),
                     ("APARTMENT", "APT"), ("FLOOR", "FL")):
        addr = addr.replace(old, new)
    addr = _NORMALIZE_RE.sub(" ", addr)
    return " ".join(addr.split())


def _parse_name(full_name: str) -> tuple[str, str]:
    parts = full_name.replace(",", " ").split()
    if not parts:
        return ("", "")
    return (parts[0], parts[1] if len(parts) > 1 else "")

        return {"status": "ERROR", "notes": str(exc), "publication_count": 0, "recent_titles": []}

    if not id_list:
        return {"status": "NOT_FOUND", "notes": "No publications found", "publication_count": 0, "recent_titles": []}
    
    # Fetch summaries for recent publications
    summary_url = "https://eutils.ncbi.nlm.nih.gov/entrez/eutils/esummary.fcgi"

        "db": "pubmed",
        "id": ",".join(id_list[:5]),  # Get up to 5 most recent

            summary_data = resp.json()
    except Exception as exc:

        return {"status": "PARTIAL", "notes": f"Found {len(id_list)} publications but failed to fetch details", "publication_count": len(id_list), "recent_titles": []}
    
    titles = []
    result = summary_data.get("result", {})
    for pmid in id_list[:5]:
        if pmid in result:
            titles.append(result[pmid].get("title", ""))
    
    return {
        "status": "CONFIRMED",
        "notes": f"Found {len(id_list)} publications",
        "publication_count": len(id_list),
        "recent_titles": titles
    }

def compute_consensus(record: DoctorRecord, sources: list[SourceResult]) -> ConsensusResult:
    confirmations = [s for s in sources if s.status == VerificationStatus.CONFIRMED]
    mismatches = [s for s in sources if s.status == VerificationStatus.MISMATCH]
    errors = [s for s in sources if s.status == VerificationStatus.ERROR]

    confirm_score = sum(Config.SOURCE_WEIGHTS.get(s.source, 1) for s in confirmations)
    mismatch_score = sum(Config.SOURCE_WEIGHTS.get(s.source, 1) for s in mismatches)
    total = confirm_score + mismatch_score
    ratio = confirm_score / total if total else 0

    best_addr = ""
    best_src = ""
    if confirmations:
        best = max(confirmations, key=lambda s: Config.SOURCE_WEIGHTS.get(s.source, 1))
        best_addr = best.raw_address
        best_src = best.source

    if confirm_score >= 10 and ratio >= 0.7:
        return ConsensusResult(
            overall=ConsensusLevel.HIGH, best_source=best_src, best_address=best_addr,
            agreement_ratio=ratio, sources_used=len(sources) - len(errors),
        )
    if confirm_score >= 6 and ratio >= 0.5:
        return ConsensusResult(
            overall=ConsensusLevel.MEDIUM, best_source=best_src, best_address=best_addr,
            agreement_ratio=ratio, sources_used=len(sources) - len(errors),
        )
    if confirm_score > 0:
        return ConsensusResult(
            overall=ConsensusLevel.LOW, best_source=best_src, best_address=best_addr,
            agreement_ratio=ratio, sources_used=len(sources) - len(errors),
        )
    if mismatch_score >= 6:
        return ConsensusResult(
            overall=ConsensusLevel.LOW, best_source="", best_address="",
            agreement_ratio=0, sources_used=len(sources) - len(errors),
        )
    return ConsensusResult(
        overall=ConsensusLevel.UNVERIFIED, best_source="", best_address="",
        agreement_ratio=0, sources_used=len(sources) - len(errors),
    )


# ---------------------------------------------------------------------------
# DoorTruth Score
# ---------------------------------------------------------------------------
def compute_doortruth_score(
    record: DoctorRecord,
    npi: NPIResult,
    consensus: ConsensusResult,
    geo: GeoResult,
    sources: list[SourceResult],
) -> DoorTruthScore:
    """Compute 0-100 DoorTruth Score from identity, consensus, address, territory, freshness."""
    penalties: list[str] = []

    # Identity match (0-25)
    identity_match = 0
    if npi.status == "CONFIRMED":
        identity_match = 25
    elif npi.status == "MISMATCH":
        identity_match = 10
        penalties.append("NPI address mismatch")
    elif npi.status == "NOT_FOUND":
        identity_match = 0
        penalties.append("NPI not found")
    else:
        identity_match = 5
        penalties.append("NPI lookup error")

    # Source consensus (0-30)
    source_consensus = 0
    if consensus.overall == ConsensusLevel.HIGH:
        source_consensus = 30
    elif consensus.overall == ConsensusLevel.MEDIUM:
        source_consensus = 18
    elif consensus.overall == ConsensusLevel.LOW:
        source_consensus = 8
    else:
        source_consensus = 0
        penalties.append("No source consensus")

    # Address precision (0-25)
    address_precision = 0
    if geo.found:
        address_precision = 20
        # Bonus for specific location types
        if any(x in geo.display_name.lower() for x in ("hospital", "medical", "clinic", "health")):
            address_precision = 25
    else:
        penalties.append("Address not geocoded")

    # Territory fit (0-10)
    territory_fit = 10
    if record.territory:
        terr = record.territory.lower()
        city_match = terr in record.city.lower()
        state_match = terr == record.state.lower()
        if not city_match and not state_match:
            territory_fit = 0
            penalties.append("Territory mismatch")
        elif city_match:
            territory_fit = 10
        else:
            territory_fit = 5
    else:
        territory_fit = 5  # Neutral when not specified

    # Freshness (0-10) — simplified; in prod would use source timestamps
    freshness = 10
    error_count = sum(1 for s in sources if s.status == VerificationStatus.ERROR)
    if error_count >= 3:
        freshness = 5
        penalties.append("Multiple source errors")

    # Apply mismatch penalties
    mismatch_count = sum(1 for s in sources if s.status == VerificationStatus.MISMATCH)
    if mismatch_count >= 2:
        penalties.append("Multiple source mismatches")

    # Residential penalty
    if any(x in record.address.lower() for x in ("apt", "unit", "suite", "#", "residence", "home")):
        if not any(x in record.address.lower() for x in ("medical", "hospital", "clinic", "center", "plaza", "building")):
            penalties.append("Possible residential address")

    score = identity_match + source_consensus + address_precision + territory_fit + freshness

    # Final clamp
    score = max(0, min(100, score))

    return DoorTruthScore(
        score=score,
        identity_match=identity_match,
        source_consensus=source_consensus,
        address_precision=address_precision,
        territory_fit=territory_fit,
        freshness=freshness,
        penalty_notes=penalties,
    )

# Visit Action
# ---------------------------------------------------------------------------
def compute_visit_action(
    record: DoctorRecord,
    npi: NPIResult,
    doortruth_score: DoorTruthScore,
    consensus: ConsensusResult,
) -> tuple[VisitAction, str]:
    """Return (action, reasoning) for the rep."""
    score = doortruth_score.score

    # Territory reassignment check (highest priority)
    if record.territory:
        terr = record.territory.lower()
        city_match = terr in record.city.lower()
        state_match = terr == record.state.lower()
        if not city_match and not state_match:
            return VisitAction.REASSIGN, f"Provider in {record.city}, {record.state} — outside territory ({record.territory})"

    # GO: strong evidence, high score, NPI confirmed
    if npi.status == "CONFIRMED" and score >= 70:
        return VisitAction.GO, f"NPI confirmed + DoorTruth Score {score}/100 — safe to visit"

    # UPDATE: NPI shows a different but valid address
    if npi.status == "MISMATCH" and score >= 40:
        return VisitAction.UPDATE, f"NPI shows different address ({npi.practice_address}, {npi.practice_city}) — update to corrected address before visit"

    # CALL_FIRST: medium score or conflicting evidence
    if 30 <= score < 70:
        return VisitAction.CALL_FIRST, f"DoorTruth Score {score}/100 — conflicting or incomplete evidence. Call office before visiting."

    # DO_NOT_GO: low score, residential, no evidence
    if score < 30:
        reasons = ", ".join(doortruth_score.penalty_notes) or "Insufficient evidence"
        return VisitAction.DO_NOT_GO, f"DoorTruth Score {score}/100 — {reasons}"

    # Fallback
    return VisitAction.MANUAL_LOOKUP, f"DoorTruth Score {score}/100 — requires manual review"

def ollama_synthesize(record: DoctorRecord, geo: GeoResult, sources: list[SourceResult], web_crawl: SourceResult) -> LLMResult:
    if Config.NO_OLLAMA or not _OLLAMA_AVAILABLE:
        return LLMResult()

    evidence_lines = [f"- {s.source}: {s.status.value} — {s.notes}" for s in sources if s.status in (VerificationStatus.CONFIRMED, VerificationStatus.MISMATCH)]
    web_context = f"\nTop verified web page: {web_crawl.url}\n" if web_crawl.status in (VerificationStatus.CONFIRMED, VerificationStatus.MISMATCH) and web_crawl.url else ""

    prompt = (
        "You are a medical-address verification analyst. Review the evidence below and decide whether "
        f"the doctor most likely still practices at the listed address.\n\n"
        f"Doctor: {record.name}\n"
        f"Specialty: {record.specialty}\n"
        f"Listed address: {record.address}, {record.city}, {record.state} {record.zip_code}\n"
        f"Geocoding: {'FOUND — ' + geo.display_name if geo.found else 'NOT FOUND'}\n\n"
        "Evidence from registries and web verification:\n"
        + "\n".join(evidence_lines[:8])
        + web_context
        + "\n\nInstructions:\n"
        "1. NPI Registry CONFIRMED is the strongest single piece of evidence.\n"
        "2. LLM web crawl CONFIRMED is also strong.\n"
        "3. Google Places or Bing API confirmation adds credibility.\n"
        "4. If sources disagree, note the conflict and downgrade confidence.\n"
        '5. Respond ONLY with JSON: {"verified": bool, "confidence": "High|Medium|Low", "reasoning": "..."}\n'
    )
    try:
        response = ollama.chat(
            model=Config.OLLAMA_MODEL,
            messages=[{"role": "user", "content": prompt}],
            options={"temperature": 0.1, "num_predict": 400},
        )
        text = response["message"]["content"]
        for delim in ("```json", "```"):
            if delim in text:
                text = text.split(delim)[1].split("```")[0]
                break
        parsed = json.loads(text.strip())
        return LLMResult(
            verified=parsed.get("verified", False),
            standardized="",
            confidence=parsed.get("confidence", ""),
            notes=parsed.get("reasoning", parsed.get("notes", "")),
        )
    except Exception as exc:
        logger.warning("Ollama synthesis error: %s", exc)
        return LLMResult()

# Single doctor verification
# ---------------------------------------------------------------------------
def verify_doctor(record: DoctorRecord, session_id: str = "") -> VerificationOutput:
    start = time.time()

    # 1. NPI (must be first — web crawl depends on it)
    npi_dict = npi_lookup(record)
    npi = NPIResult(**npi_dict)

    # 2. Parallel sources
    def _safe_run(func, *args):
        try:
            return func(*args)
        except Exception as exc:
            logger.warning("Source error: %s — %s", func.__name__, exc)
            return SourceResult(source="UNKNOWN", status=VerificationStatus.ERROR, notes=str(exc))

    if npi.status != "CONFIRMED":
        with concurrent.futures.ThreadPoolExecutor(max_workers=Config.MAX_WORKERS) as pool:
            futures = [
                pool.submit(_safe_run, ny_opd_lookup, record),
                pool.submit(_safe_run, bing_verify, record),
                pool.submit(_safe_run, google_places_verify, record),
                pool.submit(_safe_run, samhsa_lookup, record),
                pool.submit(_safe_run, healthgrades_verify, record),
                pool.submit(_safe_run, yelp_verify, record),
                pool.submit(_safe_run, ratemds_verify, record),
                pool.submit(_safe_run, cms_physician_compare, record),
                pool.submit(_safe_run, web_search_verify, record, npi.status),
            ]
            parallel_sources = [f.result() for f in concurrent.futures.as_completed(futures)]
    else:
        with concurrent.futures.ThreadPoolExecutor(max_workers=Config.MAX_WORKERS) as pool:
            futures = [
                pool.submit(_safe_run, ny_opd_lookup, record),
                pool.submit(_safe_run, bing_verify, record),
                pool.submit(_safe_run, google_places_verify, record),
                pool.submit(_safe_run, samhsa_lookup, record),
                pool.submit(_safe_run, healthgrades_verify, record),
                pool.submit(_safe_run, yelp_verify, record),
                pool.submit(_safe_run, ratemds_verify, record),
                pool.submit(_safe_run, cms_physician_compare, record),
            ]
            parallel_sources = [f.result() for f in concurrent.futures.as_completed(futures)]

    # 3. Build all sources
    all_sources = parallel_sources
    npi_as_source = SourceResult(
        source="NPI_REGISTRY",
        status=VerificationStatus(npi.status) if npi.status in ("CONFIRMED", "MISMATCH", "NOT_FOUND", "ERROR") else VerificationStatus.MISMATCH,
        raw_address=f"{npi.practice_address}, {npi.practice_city}, {npi.practice_state} {npi.practice_postal}",
        notes=npi.notes,
    )
    all_sources.insert(0, npi_as_source)

    # 4. Consensus
    consensus = compute_consensus(record, all_sources)

    # 5. Geocoding
    geo = geocode_address(record)

    # 6. DoorTruth Score
    doortruth_score = compute_doortruth_score(record, npi, consensus, geo, all_sources)

    # 7. Visit action
    visit_action, action_reason = compute_visit_action(record, npi, doortruth_score, consensus)

    # 8. Corrected address
    corrected_address = ""
    if npi.status == "MISMATCH" and npi.practice_address:
        corrected_address = f"{npi.practice_address}, {npi.practice_city}, {npi.practice_state} {npi.practice_postal}"
    elif consensus.best_address:
        corrected_address = consensus.best_address

    # 9. LLM synthesis
    web_crawl = next((s for s in all_sources if s.source == "WEB_CRAWL"), SourceResult())
    llm = ollama_synthesize(record, geo, all_sources, web_crawl)

    # 10. Aggregate phones
    all_phones: set[str] = set()
    for src in all_sources:
        all_phones.update(extract_phones(src.notes + " " + src.raw_address))
    phones_found = sorted(all_phones)

    # 11. Evidence links
    evidence_links = collect_evidence_links(all_sources, npi)
    if action_reason:
        evidence_links.append(f"Routing: {action_reason}")

    # 12. Checksum
    payload = json.dumps({"name": record.name, "address": record.address, "action": visit_action.value, "score": doortruth_score.score}, sort_keys=True)
    checksum = hashlib.sha256(payload.encode()).hexdigest()[:16]

    duration = int((time.time() - start) * 1000)
    logger.info("[%s] %s — Action: %s | Score: %d | NPI: %s | Geo: %s | %dms",
                session_id[:8] if session_id else "-", record.name, visit_action.value,
                doortruth_score.score, npi.status, "FOUND" if geo.found else "NOT FOUND", duration)

    return VerificationOutput(
        record=record,
        npi=npi,
        geo=geo,
        llm=llm,
        consensus=consensus,
        doortruth_score=doortruth_score,
        visit_action=visit_action,
        corrected_address=corrected_address,
        sources=all_sources,
        evidence_links=evidence_links,
        phones_found=phones_found,
        processed_at=datetime.utcnow(),
        session_id=session_id,
        checksum=checksum,
    )

def verify_batch(records: list[DoctorRecord], session_id: str = "", progress_callback=None) -> list[VerificationOutput]:
    results: list[VerificationOutput] = []
    for i, record in enumerate(records):
        if progress_callback:
            progress_callback(i + 1, len(records), record.name)

async def _llm_chat(messages: list[dict[str, str]], temperature: float = 0.2, provider: str | None = None) -> str:
    """Model-agnostic LLM chat. Tries requested provider, then default, then any available."""
    providers = []
    if provider and provider in _LLMS:
        providers.append(provider)
    if DEFAULT_LLM_PROVIDER in _LLMS and DEFAULT_LLM_PROVIDER not in providers:
        providers.append(DEFAULT_LLM_PROVIDER)
    for name in _LLMS:
        if name not in providers:
            providers.append(name)

    if not providers:
        return "[No LLM API key configured]"

    last_error = ""
    for p_name in providers:
        cfg = _LLMS[p_name]
        try:
            body: dict[str, Any] = {"model": cfg["model"], "messages": messages, "temperature": temperature, "max_tokens": 1500}
            # Anthropic uses different body shape
            if p_name == "anthropic":
                body = {"model": cfg["model"], "messages": messages, "temperature": temperature, "max_tokens": 1500}
            async with httpx.AsyncClient(timeout=60) as client:
                resp = await client.post(
                    cfg["url"],
                    headers=cfg["headers"](cfg["key"]),
                    json=body,
                )
                resp.raise_for_status()
                data = resp.json()
                return cfg["parse"](data)
        except Exception as exc:
            last_error = f"[{p_name}: {exc}]"
            continue

    return f"[LLM ERROR: all providers failed. Last error: {last_error}]"

    return {"lat": None, "lng": None, "error": "No results"}


def _extract_text_from_html(html: str, max_chars: int = 4000) -> str:
    """Strip HTML tags, scripts, styles and extract readable text."""
    # Remove script/style/nav/footer/header tags and their contents
    for tag in ("script", "style", "nav", "footer", "header", "aside", "form", "iframe"):
        html = re.sub(rf"<{tag}[^>]*>.*?</{tag}>", " ", html, flags=re.S | re.I)
    # Remove all remaining HTML tags
    text = re.sub(r"<[^>]+>", " ", html)
    # Decode common entities
    text = text.replace("&nbsp;", " ").replace("&#x27;", "'").replace("&quot;", '"').replace("&amp;", "&").replace("&lt;", "<").replace("&gt;", ">")
    # Collapse whitespace
    text = " ".join(text.split())
    return text[:max_chars]


async def _fetch_page_text(url: str, max_chars: int = 4000) -> str:
    """Fetch a URL and return extracted text. Handles DuckDuckGo redirects."""
    # Decode DuckDuckGo redirect URLs to real targets
    if "duckduckgo.com/l/?" in url:
        match = re.search(r"uddg=([^&]+)", url)
        if match:
            import urllib.parse
            url = urllib.parse.unquote(match.group(1))
    # Fix protocol-relative URLs
    if url.startswith("//"):
        url = "https:" + url

    try:
        async with httpx.AsyncClient(timeout=15, follow_redirects=True) as client:
            resp = await client.get(
                url,
                headers={
                    "User-Agent": (
                        "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
                        "AppleWebKit/537.36 (KHTML, like Gecko) "
                        "Chrome/120.0.0.0 Safari/537.36"
                    ),
                    "Accept": "text/html,application/xhtml+xml",
                },
            )
            if resp.status_code != 200:
                return ""
            return _extract_text_from_html(resp.text, max_chars)
    except Exception:
        return ""


async def _web_search_verify(name: str, address: str, city: str, state: str, specialty: str) -> dict[str, Any]:
    """Live web search + CRAWL result pages for full LLM analysis."""
    query = f"Dr {name} {specialty or ''} {city} {state} practice address".replace("  ", " ").strip()
    snippets: list[str] = []
    urls: list[str] = []
    pages: list[dict[str, str]] = []

    # Step 1: Search DuckDuckGo HTML (no API key)
    try:
        async with httpx.AsyncClient(timeout=20, follow_redirects=True) as client:
            resp = await client.get(
                "https://duckduckgo.com/html/",
                params={"q": query, "kl": "us-en"},
                headers={
                    "User-Agent": (
                        "Mozilla/5.0 (Windows NT 10.0; Win64; x64) "
                        "AppleWebKit/537.36 (KHTML, like Gecko) "
                        "Chrome/120.0.0.0 Safari/537.36"
                    ),
                    "Accept": "text/html",
                },
            )
            html = resp.text
            result_blocks = re.findall(
                r'<a[^>]+class="result__a"[^>]*href="([^"]+)"[^>]*>(.*?)</a>',
                html,
                re.S | re.I,
            )
            for url, title in result_blocks[:5]:
                clean_url = re.sub(r'<[^>]+>', '', url)
                clean_title = re.sub(r'<[^>]+>', '', title)
                urls.append(clean_url)
                snippets.append(clean_title)

            snippet_blocks = re.findall(
                r'<a[^>]+class="result__snippet"[^>]*>(.*?)</a>',
                html,
                re.S | re.I,
            )
            for snip in snippet_blocks[:5]:
                clean_snip = re.sub(r'<[^>]+>', '', snip).replace("&#x27;", "'").replace("&quot;", '"')
                if clean_snip and clean_snip not in snippets:
                    snippets.append(clean_snip)
    except Exception as exc:
        snippets.append(f"[Web search error: {exc}]")

    # Step 2: CRAWL the actual result pages for full content
    for raw_url in urls[:3]:  # Crawl top 3 pages
        display_url = raw_url
        if "duckduckgo.com/l/?" in raw_url:
            match = re.search(r"uddg=([^&]+)", raw_url)
            if match:
                import urllib.parse
                display_url = urllib.parse.unquote(match.group(1))
        if display_url.startswith("//"):
            display_url = "https:" + display_url
        text = await _fetch_page_text(raw_url, max_chars=4000)
        if text:
            pages.append({"url": display_url, "text": text})

    # Step 3: Run token-level match analysis on combined text
    all_text = " ".join(snippets + [p["text"] for p in pages]).upper()
    name_tokens = set(_normalize_address(name).split())
    addr_tokens = set(_normalize_address(address).split()) if address else set()
    city_token = city.upper() if city else ""
    state_token = state.upper() if state else ""

    name_found = bool(name_tokens & set(all_text.split()))
    city_found = city_token in all_text if city_token else False
    state_found = state_token in all_text if state_token else False
    address_found = bool(addr_tokens & set(all_text.split())) if addr_tokens else False

    confidence = "Low"
    if address_found and name_found:
        confidence = "High"
    elif (city_found or state_found) and name_found:
        confidence = "Medium"
    elif name_found:
        confidence = "Low"

    return {
        "source": "web_search",
        "query": query,
        "confidence": confidence,
        "name_found": name_found,
        "address_found": address_found,
        "city_found": city_found,
        "state_found": state_found,
        "snippets": snippets[:5],
        "urls": urls[:5],
        "pages": pages,
        "crawled_at": _now(),
    }

async def _llm_verify_doctor(name: str, address: str, city: str, state: str, specialty: str, web_data: dict[str, Any] | None = None) -> dict[str, Any]:
    """Use LLM to verify doctor practice location with LIVE CRAWLED page content."""
    web_context = ""
    if web_data:
        pages = web_data.get("pages", [])
        if pages:
            # Build full crawled content context for the LLM
            page_texts = []
            for i, page in enumerate(pages[:3], 1):
                page_texts.append(f"--- PAGE {i} ({page.get('url', '')}) ---\n{page.get('text', '')[:3500]}")
            full_content = "\n\n".join(page_texts)
            web_context = (
                f"\n\nLIVE CRAWLED WEB CONTENT for Dr. {name}:\n"
                f"Search query used: {web_data.get('query', '')}\n\n"
                f"{full_content}\n\n"
                f"--- END CRAWLED CONTENT ---\n"
            )
        else:
            # Fallback to snippets only
            snippets_text = "\n".join(f"- {s}" for s in web_data.get("snippets", [])[:3])
            web_context = (
                f"\n\nLive web search snippets for this doctor:\n"
                f"{snippets_text}\n"
            )

    prompt = (
        f"You are a medical practice verification analyst. You have access to LIVE CRAWLED web content. "
        f"Evaluate whether Dr. {name} practices at {address}, {city}, {state} "
        f"(specialty: {specialty or 'unknown'}).{web_context}\n\n"
        "INSTRUCTIONS:\n"
        "1. Read the crawled web content carefully.\n"
        "2. Look for the doctor's name, practice address, phone, hours, and any notes about accepting visits.\n"
        "3. Compare the crawled address against the provided address.\n"
        "4. Respond ONLY with a single JSON object — no markdown, no code blocks, no extra text.\n"
        "Use exactly these keys and value formats:\n"
        '{\n'
        '  "doctor_found": true or false,\n'
        '  "address_match": true or false,\n'
        '  "confidence": "High" or "Medium" or "Low",\n'
        '  "reasoning": "2-3 sentence explanation citing specific evidence from the crawled pages",\n'
        '  "corrected_address": "corrected full street address if different; otherwise empty string",\n'
        '  "visit_recommendation": "GO" or "CALL_FIRST" or "VERIFY_ADDRESS" or "DO_NOT_GO"\n'
        '}\n\n'
        "Guidelines:\n"
        "- GO = Crawled content confirms the doctor practices here and is accepting visits.\n"
        "- CALL_FIRST = Crawled content suggests valid but call to confirm hours/availability.\n"
        "- VERIFY_ADDRESS = Crawled content is unclear or contradicts the address; verify before visiting.\n"
        "- DO_NOT_GO = Crawled content confirms closed, moved, retired, or wrong location.\n"
    )
    result = await _llm_chat([{"role": "user", "content": prompt}], temperature=0.1)

    # Extract JSON from response
    text = result.strip()
    # Remove markdown code blocks
    if text.startswith("```"):
        text = text.split("\n", 1)[1] if "\n" in text else ""
        if text.endswith("```"):
            text = text.rsplit("\n", 1)[0]
    if text.startswith("```json"):
        text = text[7:]
        if text.endswith("```"):
            text = text[:-3]
    text = text.strip()

    try:
        parsed = json.loads(text)
        rec = str(parsed.get("visit_recommendation", "VERIFY_ADDRESS"))
        if rec not in ("GO", "CALL_FIRST", "VERIFY_ADDRESS", "DO_NOT_GO"):
            rec = "VERIFY_ADDRESS"
        return {
            "doctor_found": bool(parsed.get("doctor_found", False)),
            "address_match": bool(parsed.get("address_match", False)),
            "confidence": str(parsed.get("confidence", "Low")),
            "reasoning": str(parsed.get("reasoning", "")),
            "corrected_address": str(parsed.get("corrected_address", "")),
            "visit_recommendation": rec,
        }
    except Exception:
        # Fallback: try regex to extract JSON object
        match = re.search(r'\{[^{}]*"doctor_found"[^{}]*\}', text)
        if match:
            try:
                parsed = json.loads(match.group())
                return {
                    "doctor_found": bool(parsed.get("doctor_found", False)),
                    "address_match": bool(parsed.get("address_match", False)),
                    "confidence": str(parsed.get("confidence", "Low")),
                    "reasoning": str(parsed.get("reasoning", "")),
                    "corrected_address": str(parsed.get("corrected_address", "")),
                    "visit_recommendation": str(parsed.get("visit_recommendation", "VERIFY_ADDRESS")),
                }
            except Exception:
                pass
        return {"doctor_found": False, "address_match": False, "confidence": "Low", "reasoning": text[:300] or "LLM did not return valid JSON", "corrected_address": "", "visit_recommendation": "VERIFY_ADDRESS"}


def _compute_doortruth_score(npi: dict, llm: dict, web: dict, has_address: bool, has_phone: bool, has_email: bool) -> dict[str, Any]:
    """Compute DoorTruth 0-100 score with live web evidence."""
    score = 0
    identity = 0
    if npi.get("status") == "CONFIRMED":
        identity = 80
        score += 30
    elif npi.get("status") == "NOT_FOUND":
        identity = 15
        score += 5
    else:
        identity = 25
        score += 10

    addr_score = 0
    if has_address:
        addr_score = 30
        score += 20
    if llm.get("address_match"):
        addr_score += 25
        score += 15

    # Web evidence boost
    web_conf = web.get("confidence", "Low") if web else "Low"
    web_score = 0
    if web and web.get("address_found") and web.get("name_found"):
        web_score = 20
        addr_score += 20
    elif web and (web.get("city_found") or web.get("state_found")) and web.get("name_found"):
        web_score = 12
        addr_score += 10
    elif web and web.get("name_found"):
        web_score = 5
    score += web_score

    reachability = 0
    if has_phone:
        reachability += 15
        score += 10
    if has_email:
        reachability += 15
        score += 10

    llm_conf = llm.get("confidence", "Low")
    if llm_conf == "High":
        score += 18
    elif llm_conf == "Medium":
        score += 10
    else:
        score += 4

    score = min(100, max(0, score))

    # Determine action — web evidence heavily weighted
    llm_rec = llm.get("visit_recommendation", "")
    if llm_rec in ("GO", "CALL_FIRST", "VERIFY_ADDRESS", "DO_NOT_GO"):
        action = llm_rec
    elif web and web.get("address_found") and web.get("name_found") and npi.get("status") == "CONFIRMED":
        action = "GO"
    elif score >= 65 and (web.get("name_found") if web else False):
        action = "GO"
    elif score >= 50:
        action = "CALL_FIRST"
    elif score >= 25:
        action = "VERIFY_ADDRESS"
    elif score >= 10:
        action = "UPDATE"
    else:
        action = "DO_NOT_GO"

    return {
        "score": score,
        "identity_match": identity,
        "address_precision": addr_score,
        "reachability": reachability,
        "llm_confidence": llm_conf,
        "web_confidence": web_conf,
        "web_score": web_score,
        "action": action,
        "reasoning": llm.get("reasoning", ""),
    }

async def _verify_single_row(idx: int, df: pd.DataFrame, col_map: dict[str, str]) -> dict[str, Any]:
    """Verify a single row — used for parallel batch processing."""
    def _get(row: pd.Series, field: str) -> str:
        col = col_map.get(field)
        if col and col in row.index:
            val = row[col]
            return str(val) if pd.notna(val) else ""
        return ""

    row = df.iloc[idx]
    name = _get(row, "name")
    if not name:
        fn = _get(row, "first_name")
        ln = _get(row, "last_name")
        name = f"{fn} {ln}".strip()

    address = _get(row, "address")
    city = _get(row, "city")
    state = _get(row, "state")
    specialty = _get(row, "specialty")
    npi_str = _get(row, "npi")
    phone = _get(row, "phone")
    email = _get(row, "email")

    # Run NPI lookup
    npi_result = await _npi_lookup(name, state, city) if name else {"status": "NOT_FOUND", "notes": "No name"}

    # If NPI provided, try direct lookup
    if npi_str and npi_str.isdigit():
        direct = await _npi_lookup(npi_str)
        if direct.get("status") == "CONFIRMED":
            npi_result = direct

    # Live web search + scrape
    web_result = await _web_search_verify(name, address, city, state, specialty)

    # LLM verification with live web evidence
    llm_result = await _llm_verify_doctor(name, address, city, state, specialty, web_result)

    # Geocode for map accuracy
    geo = {"lat": None, "lng": None}
    if address and city and state:
        try:
            geo = await _geocode_address(address, city, state)
        except Exception:
            pass

    # Score
    score_data = _compute_doortruth_score(
        npi_result, llm_result, web_result, bool(address and city and state), bool(phone), bool(email)
    )

    return {
        "index": idx,
        "name": name,
        "address": address,
        "city": city,
        "state": state,
        "specialty": specialty,
        "npi": npi_str,
        "phone": phone,
        "email": email,
        "lat": geo.get("lat"),
        "lng": geo.get("lng"),
        "npi_result": npi_result,
        "llm_result": llm_result,
        "web_result": web_result,
        "score": score_data,
        "verified_at": _now(),
    }


@app.post("/api/verify")
async def api_verify(session_id: str = Form(...), row_indices: str = Form("")):
    sess = _sessions.get(session_id)
    if not sess:
        return JSONResponse({"error": "Session not found"}, status_code=404)

    df = sess["df"]
    mapping = sess["mapping"]
    col_map: dict[str, str] = {v: k for k, v in mapping.items()}

    indices = []
    if row_indices:
        indices = [int(x) for x in row_indices.split(",") if x.strip().isdigit()]
    if not indices:
        indices = list(range(len(df)))  # Process all rows by default

    # Process in parallel batches of 10 for speed
    results: list[dict[str, Any]] = []
    batch_size = 10
    for i in range(0, len(indices), batch_size):
        batch = indices[i:i + batch_size]
        batch_results = await asyncio.gather(*[_verify_single_row(idx, df, col_map) for idx in batch])
        results.extend(batch_results)

    # Merge with existing results
    existing = {r["index"]: r for r in sess.get("results", [])}
    for r in results:
        existing[r["index"]] = r
    sess["results"] = list(existing.values())

    if "history" not in sess:
        sess["history"] = []
    sess["history"].append({"type": "VERIFY", "timestamp": _now(), "detail": f"Verified {len(results)} rows. Total verified: {len(sess['results'])} / {len(df)}"})

    return JSONResponse({
        "results": results,
        "count": len(sess["results"]),
        "total_rows": len(df),
        "session_id": session_id,
    })


@app.post("/api/verify-all")
async def api_verify_all(session_id: str = Form(...)):
    """Verify all rows in the dataset with batch processing for large datasets."""
    sess = _sessions.get(session_id)
    if not sess:
        return JSONResponse({"error": "Session not found"}, status_code=404)

    df = sess["df"]
    mapping = sess["mapping"]
    col_map: dict[str, str] = {v: k for k, v in mapping.items()}
    
    # Enforce max rows limit
    total_rows = len(df)
    if total_rows > Config.MAX_ROWS:
        return JSONResponse({"error": f"Dataset exceeds maximum of {Config.MAX_ROWS} rows"}, status_code=400)
    
    indices = list(range(total_rows))

    # Process in parallel batches using configured batch size
    results: list[dict[str, Any]] = []
    batch_size = Config.BATCH_SIZE
    total_batches = (len(indices) + batch_size - 1) // batch_size
    
    for i in range(0, len(indices), batch_size):
        batch = indices[i:i + batch_size]
        batch_num = i // batch_size + 1
        logger.info(f"Processing batch {batch_num}/{total_batches} ({len(batch)} rows)")
        batch_results = await asyncio.gather(*[_verify_single_row(idx, df, col_map) for idx in batch])
        results.extend(batch_results)
        
        # Update session with partial results for progress tracking
        sess["results"] = results
        sess["progress"] = {"completed": len(results), "total": total_rows, "batch": batch_num, "total_batches": total_batches}

    sess["results"] = results
    if "history" not in sess:
        sess["history"] = []
    sess["history"].append({"type": "VERIFY_ALL", "timestamp": _now(), "detail": f"Verified all {len(results)} rows in dataset"})

    return JSONResponse({
        "results": results,
        "count": len(results),
        "total_rows": total_rows,
        "session_id": session_id,
    })

@app.post("/api/route")
async def api_route(session_id: str = Form(...), indices: str = Form(...), start_lat: float = Form(0.0), start_lng: float = Form(0.0)):
    """Build an optimized route from selected doctors."""
    sess = _sessions.get(session_id)
    if not sess:
        return JSONResponse({"error": "Session not found"}, status_code=404)

    results = sess.get("results", [])
    selected = [int(x) for x in indices.split(",") if x.strip().isdigit()]
    doctors = [r for r in results if r.get("index") in selected]

    # Simple nearest-neighbor TSP for route optimization
    route = doctors[:]
    if start_lat and start_lng:
        # Sort by distance from start
        route.sort(key=lambda d: abs((d.get("lat") or 0) - start_lat) + abs((d.get("lng") or 0) - start_lng))

    # Calculate totals
    total_time = len(route) * 12  # 12 min per visit avg
    total_distance = len(route) * 2.5  # rough miles
    high_value = sum(1 for d in route if d.get("score", {}).get("score", 0) >= 75)

    return JSONResponse({
        "route": route,
        "total_stops": len(route),
        "estimated_time_min": total_time,
        "estimated_distance_mi": round(total_distance, 1),
        "high_value_stops": high_value,
    })


@app.post("/api/sheet-autopsy")
async def api_sheet_autopsy(session_id: str = Form(...), territory_city: str = Form(""), territory_state: str = Form("")):
    """Run Sheet Autopsy: find duplicates, stale addresses, territory mismatches, campus traps, residential flags."""
    sess = _sessions.get(session_id)
    if not sess:
        return JSONResponse({"error": "Session not found"}, status_code=404)

    df = sess["df"]
    mapping = sess["mapping"]
    col_map: dict[str, str] = {v: k for k, v in mapping.items()}
    results = sess.get("results", [])

    def _get(row: pd.Series, field: str) -> str:
        col = col_map.get(field)
        if col and col in row.index:
            val = row[col]
            return str(val) if pd.notna(val) else ""
        return ""

    issues: list[dict[str, Any]] = []
    name_counts: dict[str, list[int]] = {}
    phone_counts: dict[str, list[int]] = {}

    for idx in range(len(df)):
        row = df.iloc[idx]
        name = _get(row, "name") or f"{_get(row, 'first_name')} {_get(row, 'last_name')}".strip()
        address = _get(row, "address")
        city = _get(row, "city")
        state = _get(row, "state")
        specialty = _get(row, "specialty")
        npi = _get(row, "npi")
        phone = _get(row, "phone")
        decile = _get(row, "decile")

        # Duplicate tracking
        norm_name = re.sub(r"[^a-z]", "", name.lower())
        if norm_name:
            name_counts.setdefault(norm_name, []).append(idx)
        if phone:
            phone_counts.setdefault(phone, []).append(idx)

        # Stale / missing address
        if not address or not city or not state:
            issues.append({"index": idx, "name": name, "type": "MISSING_ADDRESS", "severity": "HIGH", "detail": "Missing address, city, or state"})

        # Territory mismatch
        if territory_city and territory_state:
            tc = territory_city.upper()
            ts = territory_state.upper()
            if city.upper() != tc or state.upper() != ts:
                issues.append({"index": idx, "name": name, "type": "TERRITORY_MISMATCH", "severity": "HIGH", "detail": f"Row city={city}, state={state} vs territory={territory_city}, {territory_state}"})

        # Campus / hospital trap
        addr_upper = address.upper()
        campus_keywords = ["HOSPITAL", "MEDICAL CENTER", "CAMPUS", "UNIVERSITY", "CLINIC CENTER", "HEALTH SYSTEM", "MEDICAL GROUP"]
        if any(k in addr_upper for k in campus_keywords):
            issues.append({"index": idx, "name": name, "type": "CAMPUS_TRAP", "severity": "MEDIUM", "detail": f"Address may be a hospital/campus: {address}"})

        # Residential flag
        resi_keywords = ["APT", "APARTMENT", "UNIT", "SUITE", "RESIDENCE", "HOME", "HOUSE", "CONDO", "BLDG"]
        if any(k in addr_upper for k in resi_keywords):
            issues.append({"index": idx, "name": name, "type": "RESIDENTIAL_FLAG", "severity": "MEDIUM", "detail": f"Address may be residential: {address}"})

        # Missing NPI
        if not npi or not npi.isdigit():
            issues.append({"index": idx, "name": name, "type": "MISSING_NPI", "severity": "LOW", "detail": "NPI missing or invalid"})

        # Missing specialty
        if not specialty:
            issues.append({"index": idx, "name": name, "type": "MISSING_SPECIALTY", "severity": "LOW", "detail": "Specialty not provided"})

    # Report duplicates
    for norm_name, indices in name_counts.items():
        if len(indices) > 1:
            first_name = str(df.iloc[indices[0]].get(col_map.get("name", col_map.get("first_name", "")), ""))
            issues.append({"index": indices[0], "name": first_name, "type": "DUPLICATE_NAME", "severity": "HIGH", "detail": f"Duplicate name found {len(indices)} times at rows {indices}"})

    for phone, indices in phone_counts.items():
        if len(indices) > 1:
            first_name = str(df.iloc[indices[0]].get(col_map.get("name", col_map.get("first_name", "")), ""))
            issues.append({"index": indices[0], "name": first_name, "type": "DUPLICATE_PHONE", "severity": "MEDIUM", "detail": f"Duplicate phone {phone} found {len(indices)} times"})

    summary = {
        "total_rows": len(df),
        "issue_count": len(issues),
        "by_type": {},
        "by_severity": {"HIGH": 0, "MEDIUM": 0, "LOW": 0},
    }
    for issue in issues:
        summary["by_type"][issue["type"]] = summary["by_type"].get(issue["type"], 0) + 1
        summary["by_severity"][issue["severity"]] = summary["by_severity"].get(issue["severity"], 0) + 1

    return JSONResponse({"issues": issues, "summary": summary})

async def api_export_crm(session_id: str, format: str = "veeva"):
    """Export in CRM-specific formats."""
    sess = _sessions.get(session_id)
    if not sess:
        return JSONResponse({"error": "Session not found"}, status_code=404)

    results = sess.get("results", [])
    if not results:
        return JSONResponse({"error": "No verified results. Run verification first."}, status_code=400)

    for r in results:
        s = r.get("score", {})
        base = {
            "name": r.get("name", ""),
            "address": r.get("address", ""),
            "city": r.get("city", ""),
            "state": r.get("state", ""),
            "zip": r.get("zip", ""),
            "specialty": r.get("specialty", ""),
            "npi": r.get("npi", ""),
            "phone": r.get("phone", ""),
            "email": r.get("email", ""),
            "doortruth_score": s.get("score", 0),
            "action": s.get("action", ""),
            "llm_reasoning": s.get("reasoning", ""),
            "verified_at": r.get("verified_at", ""),
        }

        if format == "veeva":
            base["veeva_account_status"] = "Active" if s.get("action") == "GO" else "Pending Verification"
            base["veeva_account_type"] = "HCP"
            base["veeva_territory"] = r.get("city", "") + " " + r.get("state", "")
            base["veeva_call_priority"] = "High" if s.get("score", 0) >= 75 else "Medium" if s.get("score", 0) >= 45 else "Low"
        elif format == "salesforce":
            base["sf_account_name"] = r.get("name", "")
            base["sf_account_type"] = "HCP"
            base["sf_industry"] = "Healthcare"
            base["sf_rating"] = "Hot" if s.get("score", 0) >= 75 else "Warm" if s.get("score", 0) >= 45 else "Cold"
            base["sf_next_step"] = s.get("action", "").replace("_", " ")
        elif format == "hubspot":
            base["hs_company_name"] = r.get("name", "") + " — " + r.get("specialty", "")
            base["hs_lifecycle_stage"] = "lead"
            base["hs_lead_status"] = "NEW" if s.get("action") == "GO" else "UNQUALIFIED"
            base["hs_hs_lead_source"] = "DoorTruth HCP"

        rows.append(base)

        "format": format,
        "row_count": len(rows),
    })

@app.post("/api/history/summarize")
async def api_history_summarize(session_id: str = Form(...)):
    """Use LLM to summarize session history."""
    sess = _sessions.get(session_id)
    if not sess:
        return JSONResponse({"error": "Session not found"}, status_code=404)

    history = sess.get("history", [])
    if not history:
        return JSONResponse({"summary": "No history to summarize.", "key_facts": []})

    # Build a text representation
    lines = []
    for h in history:
        lines.append(f"[{h.get('timestamp', '')}] {h.get('type', '')}: {h.get('detail', '')}")
    history_text = "\n".join(lines[:100])  # Cap at 100 entries

    prompt = (
        "You are a session summarizer. Summarize the following session activity log into 3-5 bullet points. "
        "Then extract 3-5 key facts that should be remembered for future sessions (territory insights, doctor moves, "
        "verified addresses, call outcomes, etc.).\n\n"
        f"Activity Log:\n{history_text}\n\n"
        "Respond ONLY with JSON: {\"summary\": \"...\", \"key_facts\": [\"fact1\", \"fact2\"]}"
    )
    result = await _llm_chat([{"role": "user", "content": prompt}], temperature=0.2)

    # Try to parse JSON
    summary = "Summary unavailable."
    key_facts: list[str] = []
    try:
        # Strip markdown if present
        text = result.strip()
        if text.startswith("```"):
            text = text.split("\n", 1)[1] if "\n" in text else ""
            if text.endswith("```"):
                text = text.rsplit("\n", 1)[0]
        parsed = json.loads(text)
        summary = parsed.get("summary", result)
        key_facts = parsed.get("key_facts", [])
    except Exception:
        summary = result

    # Store key facts in long-term memory
    if key_facts:
        if "extracted_memory" not in sess:
            sess["extracted_memory"] = []
        for fact in key_facts:
            sess["extracted_memory"].append({"fact": fact, "timestamp": _now(), "session_id": session_id})
        # Also store globally per user if we had user IDs; for now use session as proxy
        _lt_memory[session_id] = sess["extracted_memory"]

    return JSONResponse({"summary": summary, "key_facts": key_facts, "history_count": len(history)})

async function handleFile(file) {
  uploadProgress.classList.remove('hidden');
  progressFill.style.width = '30%';
  progressText.textContent = 'Parsing ' + file.name + '...';

  const formData = new FormData();
  formData.append('file', file);

  try {
    const res = await fetch('/api/upload', { method: 'POST', body: formData });
    const data = await res.json();

    if (data.error) {
      alert('Error: ' + data.error);
      uploadProgress.classList.add('hidden');
      return;
    }

    sessionId = data.session_id;
    currentMapping = data.mapping;
    allColumns = data.columns;

    progressFill.style.width = '100%';
    progressText.textContent = 'Done!';

    previewFilename.textContent = file.name;
    previewCount.textContent = data.row_count;
    renderPreview(data.preview, data.columns);
    uploadPreview.classList.remove('hidden');
    buildMappingUI(data.columns, data.mapping);

  } catch (err) {
    alert('Upload failed: ' + err.message);
  }
}

function renderResults(results) {
  const counts = { GO: 0, CALL_FIRST: 0, VERIFY_ADDRESS: 0, UPDATE: 0, DO_NOT_GO: 0, total: results.length };
  results.forEach(r => {
    const a = r.score?.action;
    if (a === 'GO') counts.GO++;
    else if (a === 'CALL_FIRST') counts.CALL_FIRST++;
    else if (a === 'VERIFY_ADDRESS') counts.VERIFY_ADDRESS++;
    else if (a === 'UPDATE') counts.UPDATE++;
    else if (a === 'DO_NOT_GO') counts.DO_NOT_GO++;
  });

  document.getElementById('count-go').textContent = counts.GO;
  document.getElementById('count-call').textContent = counts.CALL_FIRST;
  document.getElementById('count-verify').textContent = counts.VERIFY_ADDRESS;
  document.getElementById('count-update').textContent = counts.UPDATE;
  document.getElementById('count-dng').textContent = counts.DO_NOT_GO;
  document.getElementById('count-total').textContent = counts.total;

  resultsTable.innerHTML = results.map(r => {
    const s = r.score || {};
    const action = s.action || 'VERIFY_ADDRESS';
    const badgeClass = action === 'GO' ? 'badge-go' : action === 'CALL_FIRST' ? 'badge-call' : action === 'VERIFY_ADDRESS' ? 'badge-verify' : action === 'UPDATE' ? 'badge-update' : 'badge-dng';
    const badgeText = action.replace('_', ' ');
    const scoreColor = s.score >= 75 ? 'high' : s.score >= 45 ? 'medium' : 'low';

    return `
      <tr>
        <td><div style="font-weight:600">${escapeHtml(r.name || 'Unknown')}</div><div style="font-size:0.75rem;color:var(--text-muted)">${escapeHtml(r.specialty || '')}</div></td>
        <td><div>${escapeHtml(r.address || '')}</div><div style="font-size:0.75rem;color:var(--text-muted)">${escapeHtml(r.city || '')}, ${escapeHtml(r.state || '')}</div></td>
        <td><span style="font-family:monospace;font-size:0.8rem;color:var(--text-muted)">${escapeHtml(r.npi || '—')}</span></td>
        <td><div style="display:flex;align-items:center;gap:8px"><div class="score-bar" style="width:60px"><div class="score-fill ${scoreColor}" style="width:${s.score || 0}%"></div></div><span style="font-size:0.75rem;font-weight:600">${s.score || 0}</span></div></td>
        <td><div style="font-size:0.7rem;color:var(--text-muted)">${s.identity_match || 0}%</div></td>
        <td><div style="font-size:0.7rem;color:var(--text-muted)">${s.address_precision || 0}%</div></td>
        <td><div style="font-size:0.7rem;color:var(--text-muted)">${s.reachability || 0}%</div></td>
        <td><span class="badge ${badgeClass}">${badgeText}</span></td>
        <td><button class="btn btn-secondary" style="padding:4px 10px;font-size:0.75rem" onclick="showDoctorProfile(${r.index})">Profile</button></td>
      </tr>
    `;
  }).join('');
}

// Escape HTML
function escapeHtml(text) {
  const div = document.createElement('div');
  div.textContent = text;
  return div.innerHTML;

// Map
function initMap() {
  if (leafletMap) { leafletMap.remove(); leafletMap = null; }
  const container = document.getElementById('map-container');
  if (!container) return;

  leafletMap = L.map('map-container').setView([39.8283, -98.5795], 4);
  L.tileLayer('https://{s}.tile.openstreetmap.org/{z}/{x}/{y}.png', {
    attribution: '&copy; OpenStreetMap',
    maxZoom: 18,
  }).addTo(leafletMap);

  const filter = document.getElementById('map-filter')?.value || 'ALL';
  const doctors = verifyResults.filter(d => filter === 'ALL' || d.score?.action === filter);

  const bounds = [];
  doctors.forEach(d => {
    const action = d.score?.action || 'VERIFY_ADDRESS';
    const color = action === 'GO' ? '#22c55e' : action === 'CALL_FIRST' ? '#eab308' : action === 'VERIFY_ADDRESS' ? '#3b82f6' : '#ef4444';
    const lat = d.lat;
    const lng = d.lng;
    if (!lat || !lng) return;
    bounds.push([lat, lng]);

    const marker = L.circleMarker([lat, lng], {
      radius: 8, fillColor: color, color: color, weight: 2, opacity: 1, fillOpacity: 0.7
    }).addTo(leafletMap);

    marker.bindPopup(`
      <div style="font-family:Inter,sans-serif;font-size:0.875rem">
        <strong>${escapeHtml(d.name || '')}</strong><br>
        <span style="color:#666">${escapeHtml(d.specialty || '')}</span><br>
        <span style="color:#666">${escapeHtml(d.address || '')}, ${escapeHtml(d.city || '')}, ${escapeHtml(d.state || '')}</span><br>
        <span style="color:${color};font-weight:600">${(action).replace('_', ' ')}</span> — Score: ${d.score?.score || 0}
      </div>
    `);
  });

  if (bounds.length > 1) {
    leafletMap.fitBounds(bounds, { padding: [30, 30] });
  } else if (bounds.length === 1) {
    leafletMap.setView(bounds[0], 13);
  }
}

document.getElementById('map-filter')?.addEventListener('change', initMap);

function initRouteView() {
  const list = document.getElementById('route-doctor-list');
  if (!list) return;
  list.innerHTML = verifyResults.map(d => {
    const action = d.score?.action || 'VERIFY';
    const isGo = action === 'GO';
    return `
      <div class="route-doctor-item ${isGo ? 'selected' : ''}" data-idx="${d.index}" onclick="toggleRouteDoctor(${d.index})">
        <input type="checkbox" class="checkbox" ${isGo ? 'checked' : ''} onchange="toggleRouteDoctor(${d.index})" onclick="event.stopPropagation()">
        <div class="route-doctor-info">
          <div class="name">${escapeHtml(d.name || '')}</div>
          <div class="addr">${escapeHtml(d.city || '')}, ${escapeHtml(d.state || '')} — ${escapeHtml(d.specialty || '')}</div>
        </div>
        <div class="route-doctor-score">${d.score?.score || 0}</div>
      </div>
    `;
  }).join('');

  // Pre-select GO doctors
  selectedRouteDoctors = new Set(verifyResults.filter(d => d.score?.action === 'GO').map(d => d.index));
}

window.toggleRouteDoctor = function (idx) {
  if (selectedRouteDoctors.has(idx)) selectedRouteDoctors.delete(idx);
  else selectedRouteDoctors.add(idx);

  document.querySelectorAll('.route-doctor-item').forEach(el => {
    const elIdx = parseInt(el.dataset.idx);
    el.classList.toggle('selected', selectedRouteDoctors.has(elIdx));
    const cb = el.querySelector('.checkbox');
    if (cb) cb.checked = selectedRouteDoctors.has(elIdx);
  });
};

let mediaRecorder = null;
let recordedChunks = [];

document.getElementById('btn-record-start')?.addEventListener('click', async () => {
  if (!sessionId) { alert('Upload a file first'); return; }
  try {
    const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
    mediaRecorder = new MediaRecorder(stream);
    recordedChunks = [];
    mediaRecorder.ondataavailable = (e) => { if (e.data.size > 0) recordedChunks.push(e.data); };
    mediaRecorder.onstop = async () => {
      const blob = new Blob(recordedChunks, { type: 'audio/webm' });
      const formData = new FormData();
      formData.append('session_id', sessionId);
      formData.append('file', blob, 'memo.webm');
      formData.append('notes', document.getElementById('voice-notes').value || '');
      await fetch('/api/voice-memo', { method: 'POST', body: formData });
      document.getElementById('record-status').textContent = 'Saved!';
      document.getElementById('voice-notes').value = '';
      loadVoiceMemos();
    };
    mediaRecorder.start();
    document.getElementById('btn-record-start').disabled = true;
    document.getElementById('btn-record-stop').disabled = false;
    document.getElementById('record-status').textContent = 'Recording...';
  } catch (err) {
    alert('Microphone access denied or unavailable: ' + err.message);
  }
});

document.getElementById('btn-record-stop')?.addEventListener('click', () => {
  if (mediaRecorder && mediaRecorder.state !== 'inactive') {
    mediaRecorder.stop();
    mediaRecorder.stream.getTracks().forEach(t => t.stop());
    document.getElementById('btn-record-start').disabled = false;
    document.getElementById('btn-record-stop').disabled = true;
  }
});
