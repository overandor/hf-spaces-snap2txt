FROM node:20-slim

# Install build dependencies
RUN apt-get update && apt-get install -y \
    git \
    python3 \
    cmake \
    build-essential \
    curl \
    wget \
    ca-certificates \
    xz-utils \
    && rm -rf /var/lib/apt/lists/*

# Set toolchain paths
ENV INSTALL_BASE=/opt/wasm-toolchains
ENV EMSDK_PATH=/opt/wasm-toolchains/emsdk
ENV SWIFTWASM_PATH=/opt/wasm-toolchains/swiftwasm

# --- Install Emscripten ---
RUN mkdir -p ${INSTALL_BASE} && \
    git clone https://github.com/emscripten-core/emsdk.git ${EMSDK_PATH} && \
    cd ${EMSDK_PATH} && \
    ./emsdk install latest && \
    ./emsdk activate latest

# --- Install SwiftWasm (Linux x86_64) ---
RUN mkdir -p ${SWIFTWASM_PATH} && \
    SWIFTWASM_VERSION="swift-wasm-5.10-SNAPSHOT-2023-11-02-a" && \
    SWIFTWASM_TARBALL="${SWIFTWASM_VERSION}-ubuntu22.04_x86_64.tar.gz" && \
    SWIFTWASM_URL="https://github.com/swiftwasm/swift/releases/download/${SWIFTWASM_VERSION}/${SWIFTWASM_TARBALL}" && \
    cd /tmp && \
    curl -L -o "${SWIFTWASM_TARBALL}" "${SWIFTWASM_URL}" && \
    tar xzf "${SWIFTWASM_TARBALL}" -C ${SWIFTWASM_PATH} --strip-components=1 && \
    rm -f "${SWIFTWASM_TARBALL}"

# Set up app
WORKDIR /app
COPY package*.json ./
RUN npm ci

COPY . .
RUN npm run build

# HF Space default port is 7860, but we accept PORT env
EXPOSE 7860
ENV PORT=7860

CMD ["node", "dist/server.js"]

```

# WASM Compiler Bridge

A web-based compiler bridge that compiles **C++** and **Swift** source code to **WebAssembly (WASM)** and runs it in the browser.

## Features

- **C++ -> WASM** via Emscripten (single-file JS+WASM glue, runs directly in browser)
- **Swift -> WASM** via SwiftWasm (generates WASI-target WASM binary)
- Monaco code editor with error/warning squiggles
- Examples dropdown with built-in code snippets
- Compiler flags input (e.g. `-O3 -s WASM=1`)
- Export compiled output (`.js` for C++, `.wasm` for Swift)
- WASM Inspector panel (exports, imports, memory)
- Resizable editor/output panels
- Settings modal (theme, font size, auto-compile)
- URL hash sharing + localStorage persistence
- Keyboard shortcuts (`Ctrl+Enter` compile, `Ctrl+R` run, etc.)

## Architecture

```
Browser (Monaco Editor)  <--HTTP-->  Express API  -->  emcc / swiftc
                                               (temp files)
```

## Prerequisites (Local Development)

- Node.js 18+ (for the bridge server)
- macOS or Linux with ~3GB free disk space (for toolchains)

## Quick Start (Local)

### 1. Install dependencies

```bash
cd wasm-compiler-bridge
npm install
npm run build   # compiles TypeScript
```

### 2. Install compiler toolchains

```bash
./scripts/install-toolchains.sh
```

This installs:
- **Emscripten** to `~/wasm-toolchains/emsdk`
- **SwiftWasm** to `~/wasm-toolchains/swiftwasm`

> **Note:** This downloads ~1-2GB.

### 3. Start the bridge

```bash
npm start
```

Open `http://localhost:3456` in your browser.

## Deploy to Hugging Face Space

The project includes a `Dockerfile` with both toolchains pre-installed.

1. Create a new HF Space with **Docker** as the SDK
2. Push this repo to the Space (or use the Dockerfile)
3. The container installs Emscripten + SwiftWasm at build time
4. The server listens on port `7860` (HF default)

```bash
# Clone your HF Space repo
git clone https://huggingface.co/spaces/YOURNAME/wasm-compiler-bridge
cd wasm-compiler-bridge

# Copy project files
cp -r /path/to/wasm-compiler-bridge/* .
git add .
git commit -m "Initial deploy"
git push
```

> **Note:** First build will take 10-20 minutes as it downloads and installs Emscripten + SwiftWasm inside the container.

## API Endpoints

| Endpoint | Method | Description |
|----------|--------|-------------|
| `GET /api/status` | GET | Check toolchain availability |
| `GET /api/examples` | GET | Built-in code examples for C++ and Swift |
| `POST /api/compile/cpp` | POST | Compile C++ code to WASM (accepts `{ code, flags }`) |
| `POST /api/compile/swift` | POST | Compile Swift code to WASM (accepts `{ code, flags }`) |
| `POST /api/inspect` | POST | Inspect a WASM binary (`{ wasmBase64 }`) |

## Environment Variables

| Variable | Default | Description |
|----------|---------|-------------|
| `PORT` | `3456` | Server port |
| `EMSDK_PATH` | `~/wasm-toolchains/emsdk` | Path to Emscripten SDK |
| `SWIFTWASM_PATH` | `~/wasm-toolchains/swiftwasm` | Path to SwiftWasm toolchain |

## C++ Example

```cpp
#include <emscripten.h>
#include <stdio.h>

extern "C" {
  EMSCRIPTEN_KEEPALIVE
  int add(int a, int b) { return a + b; }
}

int main() {
  printf("Hello from C++ WASM\n");
  return 0;
}
```

## Swift Example

```swift
@_cdecl("add")
public func add(a: Int32, b: Int32) -> Int32 {
    return a + b
}

print("Swift WASM module compiled")
```

## Limitations

- **Swift browser execution** requires a full WASI runtime (the bridge shows module info/exports). For full browser execution, use SwiftWasm's JavaScriptKit.
- **Security**: This is a dev tool. Do not expose to untrusted networks without sandboxing.

(function () {
  const API_BASE = '';
  const LS_KEY = 'wasm-bridge-v1';

  let currentLang = 'cpp';
  let editor = null;
  let compiledJs = null;
  let compiledWasm = null;
  let examplesData = { cpp: [], swift: [] };
  let diagnosticsDecorations = [];
  let autoCompileTimer = null;

  const settings = {
    theme: localStorage.getItem('wb-theme') || 'vs-dark',
    fontSize: parseInt(localStorage.getItem('wb-fontSize') || '13', 10),
    autoCompile: localStorage.getItem('wb-autoCompile') === 'true',
    autoCompileDelay: parseInt(localStorage.getItem('wb-autoCompileDelay') || '1500', 10),
  };

  const $ = (s) => document.querySelector(s);
  const $$ = (s) => document.querySelectorAll(s);

  /* ---------- UTILS ---------- */
  function toast(message, type = 'success', duration = 3000) {
    const container = $('#toast-container');
    const el = document.createElement('div');
    el.className = 'toast ' + type;
    el.textContent = message;
    container.appendChild(el);
    setTimeout(() => el.remove(), duration);
  }

  function encodeHash(code, lang) {
    try {
      const payload = JSON.stringify({ lang, code });
      const compressed = btoa(unescape(encodeURIComponent(payload)));
      location.hash = 'code=' + compressed;
    } catch { /* ignore */ }
  }

  function decodeHash() {
    const hash = location.hash;
    if (!hash.startsWith('#code=')) return null;
    try {
      const payload = decodeURIComponent(escape(atob(hash.slice(6))));
      return JSON.parse(payload);
    } catch { return null; }
  }

  function saveToStorage() {
    const data = { cpp: getSavedCode('cpp'), swift: getSavedCode('swift'), lastLang: currentLang };
    localStorage.setItem(LS_KEY, JSON.stringify(data));
  }

  function loadFromStorage() {
    try { return JSON.parse(localStorage.getItem(LS_KEY)); } catch { return null; }
  }

  function getSavedCode(lang) {
    const model = editor ? editor.getModel() : null;
    if (model && currentLang === lang) return model.getValue();
    const stored = loadFromStorage();
    return stored ? stored[lang] || '' : '';
  }

  function setSavedCode(lang, code) {
    const stored = loadFromStorage() || {};
    stored[lang] = code; stored.lastLang = lang;
    localStorage.setItem(LS_KEY, JSON.stringify(stored));
  }

  /* ---------- OUTPUT ---------- */
  function logOutput(text, type = 'info') {
    const out = $('#compiler-output');
    const span = document.createElement('span');
    span.className = type === 'error' ? 'error-line' : type === 'success' ? 'success-line' : type === 'warn' ? 'warn-line' : '';
    span.textContent = text + '\n';
    out.appendChild(span);
    out.scrollTop = out.scrollHeight;
  }

  function logProgram(text) {
    const out = $('#program-output');
    out.textContent += text + '\n';
    out.scrollTop = out.scrollHeight;
  }

  function clearCompilerOutput() { $('#compiler-output').textContent = ''; }
  function clearProgramOutput() { $('#program-output').textContent = ''; }

  function setBadge(id, ok, text) {
    const el = $(id);
    el.className = 'badge ' + (ok ? 'ready' : 'missing');
    el.textContent = text;
  }

  /* ---------- TOOLCHAINS ---------- */
  async function checkToolchains() {
    try {
      const res = await fetch(`${API_BASE}/api/status`);
      const data = await res.json();
      setBadge('#emcc-badge', data.emccAvailable, data.emccAvailable ? 'emcc: ready' : 'emcc: missing');
      setBadge('#swiftc-badge', data.swiftcAvailable, data.swiftcAvailable ? 'swiftc: ready' : 'swiftc: missing');
    } catch (e) {
      setBadge('#emcc-badge', false, 'emcc: offline');
      setBadge('#swiftc-badge', false, 'swiftc: offline');
    }
  }

  /* ---------- EXAMPLES ---------- */
  async function loadExamples() {
    try {
      const res = await fetch(`${API_BASE}/api/examples`);
      examplesData = await res.json();
      populateExamples();
    } catch { /* ignore */ }
  }

  function populateExamples() { /* dropdown removed in Xcode UI */ }

  function loadExample(code) {
    if (!editor || !code) return;
    editor.setValue(code);
    setSavedCode(currentLang, code);
    encodeHash(code, currentLang);
    clearCompilerOutput();
    clearProgramOutput();
    clearDiagnostics();
    toast('Example loaded', 'success');
  }

  /* ---------- DIAGNOSTICS ---------- */
  function clearDiagnostics() {
    if (!editor) return;
    diagnosticsDecorations = editor.deltaDecorations(diagnosticsDecorations, []);
  }

  function showDiagnostics(diagnostics) {
    clearDiagnostics();
    if (!diagnostics || !diagnostics.length) return;
    const decos = diagnostics.map((d) => ({
      range: new monaco.Range(d.line || 1, d.column || 1, d.line || 1, (d.column || 1) + 1),
      options: {
        isWholeLine: true,
        className: d.severity === 'error' ? 'monaco-error-line' : d.severity === 'warning' ? 'monaco-warn-line' : undefined,
        glyphMarginClassName: d.severity === 'error' ? 'monaco-error-glyph' : 'monaco-warn-glyph',
        hoverMessage: { value: `**${d.severity.toUpperCase()}**: ${d.message}` },
        overviewRuler: { color: d.severity === 'error' ? '#f85149' : '#d29922', position: monaco.editor.OverviewRulerLane.Right },
      },
    }));
    diagnosticsDecorations = editor.deltaDecorations(diagnosticsDecorations, decos);
  }

  /* ---------- INSPECTOR ---------- */
  function renderInspector(wasmInfo) {
    const container = $('#inspector-content');
    if (!wasmInfo) {
      container.innerHTML = '<div class="inspector-placeholder">Compile to see module info</div>';
      return;
    }
    let html = '';
    if (wasmInfo.exports?.length) {
      html += '<div class="inspector-section"><div class="inspector-section-title">Exports (' + wasmInfo.exports.length + ')</div>';
      for (const e of wasmInfo.exports) {
        html += `<div class="inspector-item"><span>${escapeHtml(e.name)}</span><span class="inspector-kind">${kindName(e.kind)}</span></div>`;
      }
      html += '</div>';
    }
    if (wasmInfo.imports?.length) {
      html += '<div class="inspector-section"><div class="inspector-section-title">Imports (' + wasmInfo.imports.length + ')</div>';
      for (const imp of wasmInfo.imports) {
        html += `<div class="inspector-item"><span>${escapeHtml(imp.module)}.${escapeHtml(imp.name)}</span><span class="inspector-kind">${kindName(imp.kind)}</span></div>`;
      }
      html += '</div>';
    }
    if (wasmInfo.memories?.length) {
      html += '<div class="inspector-section"><div class="inspector-section-title">Memory</div>';
      for (const m of wasmInfo.memories) {
        html += `<div class="inspector-item"><span>min pages</span><span class="inspector-kind">${m.min} (${m.min * 64} KB)</span></div>`;
      }
      html += '</div>';
    }
    container.innerHTML = html || '<div class="inspector-placeholder">No module info available</div>';
  }

  function kindName(k) {
    const map = { 0: 'func', 1: 'table', 2: 'mem', 3: 'global' };
    return map[k] ?? ('kind:' + k);
  }

  function escapeHtml(s) {
    return s.replace(/&/g, '&amp;').replace(/</g, '&lt;').replace(/>/g, '&gt;');
  }

  function defaultCpp() {
    return `#include <emscripten.h>
#include <stdio.h>

extern "C" {
  EMSCRIPTEN_KEEPALIVE
  int add(int a, int b) { return a + b; }

  EMSCRIPTEN_KEEPALIVE
  int factorial(int n) {
    if (n <= 1) return 1;
    return n * factorial(n - 1);
  }
}

int main() {
  printf("C++ WASM ready\\n");
  printf("5 + 3 = %d\\n", add(5, 3));
  printf("factorial(5) = %d\\n", factorial(5));
  return 0;
}
`;
  }

  function defaultSwift() {
    return `import Foundation

@_cdecl("add")
public func add(a: Int32, b: Int32) -> Int32 {
    return a + b
}

@_cdecl("factorial")
public func factorial(n: Int32) -> Int32 {
    if n <= 1 { return 1 }
    return n * factorial(n: n - 1)
}

print("Swift WASM module compiled successfully")
`;
  }

  function initEditor() {
    require.config({ paths: { vs: 'https://cdnjs.cloudflare.com/ajax/libs/monaco-editor/0.44.0/min/vs' } });
    require(['vs/editor/editor.main'], () => {
      const style = document.createElement('style');
      style.textContent = `
        .monaco-error-line { background: rgba(248,81,73,0.1); }
        .monaco-warn-line { background: rgba(210,153,34,0.1); }
        .monaco-error-glyph { background: #f85149; width: 5px !important; margin-left: 3px; }
        .monaco-warn-glyph { background: #d29922; width: 5px !important; margin-left: 3px; }
      `;
      document.head.appendChild(style);

      let initialCode, initialLang;
      const fromHash = decodeHash();
      if (fromHash) {
        initialLang = fromHash.lang;
        initialCode = fromHash.code;
      } else {
        const stored = loadFromStorage();
        initialLang = stored?.lastLang || 'cpp';
        initialCode = stored?.[initialLang] || (initialLang === 'cpp' ? defaultCpp() : defaultSwift());
      }
      currentLang = initialLang;
      $('#lang-select').value = currentLang;

      editor = monaco.editor.create($('#editor'), {
        value: initialCode,
        language: initialLang === 'cpp' ? 'cpp' : 'swift',
        theme: settings.theme,
        automaticLayout: true,
        minimap: { enabled: false },
        fontSize: settings.fontSize,
        tabSize: 2,
        scrollBeyondLastLine: false,
        glyphMargin: true,
      });

      editor.onDidChangeModelContent(() => {
        const code = editor.getValue();
        setSavedCode(currentLang, code);
        encodeHash(code, currentLang);
        if (settings.autoCompile) {
          clearTimeout(autoCompileTimer);
          autoCompileTimer = setTimeout(() => compile(true), settings.autoCompileDelay);
        }
      });
    });
  }

  function switchLang(lang) {

    setSavedCode(currentLang, editor.getValue());

    $('#nav-main-cpp').classList.toggle('active', lang === 'cpp');
    // Update editor tab
    const tabLabel = $('.tab-label');
    const tabIcon = $('.tab-icon');
    if (tabLabel) tabLabel.textContent = lang === 'cpp' ? 'main.cpp' : 'main.swift';
    if (tabIcon) tabIcon.textContent = lang === 'cpp' ? '⚙️' : '🦅';

  async function compile(silent = false) {
    const code = editor ? editor.getValue() : '';
    if (!code.trim()) return;
    const flags = $('#flags-input').value.trim();

    if (!silent) { clearCompilerOutput(); clearProgramOutput(); }

        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ code, flags }),
      });
      const data = await res.json();

      if (data.stdout) logOutput(data.stdout, 'info');
      if (data.stderr) logOutput(data.stderr, 'warn');
      if (data.diagnostics?.length) showDiagnostics(data.diagnostics);
      else clearDiagnostics();

      if (data.success) {

        if (!silent) toast('Compilation failed', 'error');
      }
    } catch (err) {
      logOutput('Network error: ' + err.message, 'error');
      if (!silent) toast('Network error', 'error');

      $('#compile-btn').disabled = false;
      $('#compile-btn').innerHTML = '<span class="icon-play"></span> Build';

  async function runCpp() {
    if (!compiledJs) return;
    clearProgramOutput();
    try {
      const blob = new Blob([compiledJs], { type: 'application/javascript' });
      const url = URL.createObjectURL(blob);
      const originalLog = console.log;
      const originalError = console.error;
      console.log = (...args) => { logProgram(args.map(String).join(' ')); originalLog.apply(console, args); };
      console.error = (...args) => { logProgram('ERR: ' + args.map(String).join(' ')); originalError.apply(console, args); };

      const Module = await import(url);
      const factory = Module.default || Module;
      if (typeof factory === 'function') {
        await factory({ print: (text) => logProgram(text), printErr: (text) => logProgram('ERR: ' + text) });
      }
      console.log = originalLog;
      console.error = originalError;
      URL.revokeObjectURL(url);
    } catch (err) {
      logProgram('Runtime error: ' + err.message);
    }
  }

  async function runSwift() {
    if (!compiledWasm) return;
    clearProgramOutput();
    try {
      const binary = Uint8Array.from(atob(compiledWasm), (c) => c.charCodeAt(0));
      const wasm = await WebAssembly.compile(binary);
      const imports = WebAssembly.Module.imports(wasm);
      const exports = WebAssembly.Module.exports(wasm);

      logProgram('=== Swift WASM Module Info ===');
      logProgram('Imports (' + imports.length + '):');
      for (const imp of imports) logProgram('  ' + imp.kind + ' ' + imp.module + '.' + imp.name);
      logProgram('Exports (' + exports.length + '):');
      for (const exp of exports) logProgram('  ' + exp.kind + ' ' + exp.name);

      const wasiImports = {};
      for (const imp of imports) {
        if (imp.module === 'wasi_snapshot_preview1') {
          wasiImports[imp.name] = () => { throw new Error('WASI not supported'); };
        }
      }
      try {
        const instance = await WebAssembly.instantiate(wasm, { wasi_snapshot_preview1: wasiImports });
        logProgram('Module instantiated (WASI stubs).');
        if (instance.exports._start) logProgram('Entry point _start found but requires full WASI.');
      } catch (e) {
        logProgram('Instantiation requires full WASI runtime: ' + e.message);
      }
    } catch (err) {
      logProgram('Runtime error: ' + err.message);
    }
  }

function parseDiagnostics(stderr: string, lang: 'cpp' | 'swift'): Diagnostic[] {
  const diagnostics: Diagnostic[] = [];
  if (!stderr) return diagnostics;
  const lines = stderr.split('\n');

  if (lang === 'cpp') {
    // emcc/clang: file.cpp:12:34: error: message
    const re = /^(.+?):(\d+):(\d+):\s*(error|warning|note):\s*(.+)$/;
    for (const line of lines) {
      const m = line.match(re);
      if (m) {
        diagnostics.push({
          file: m[1],
          line: parseInt(m[2], 10),
          column: parseInt(m[3], 10),
          severity: m[4] as Diagnostic['severity'],
          message: m[5],
        });
      }
    }
  } else {
    // swiftc: file.swift:12:34: error: message
    const re = /^(.+?):(\d+):(\d+):\s*(error|warning|note):\s*(.+)$/;
    for (const line of lines) {
      const m = line.match(re);
      if (m) {
        diagnostics.push({
          file: m[1],
          line: parseInt(m[2], 10),
          column: parseInt(m[3], 10),
          severity: m[4] as Diagnostic['severity'],
          message: m[5],
        });
      }
    }
  }
  return diagnostics;
}

function parseWasmExportsImports(wasmBytes: Buffer): { exports: any[]; imports: any[]; memories: any[] } {
  const result = { exports: [] as any[], imports: [] as any[], memories: [] as any[] };
  let pos = 8; // skip magic + version
  while (pos < wasmBytes.length) {
    const id = wasmBytes[pos++];
    if (id > 11) break;
    let len = 0;
    let shift = 0;
    while (true) {
      const b = wasmBytes[pos++];
      len |= (b & 0x7f) << shift;
      shift += 7;
      if ((b & 0x80) === 0) break;
    }
    const end = pos + len;
    if (id === 1) { // Type section - skip
    } else if (id === 2) { // Import section
      let count = readLeb(wasmBytes, pos);
      for (let i = 0; i < count.value; i++) {
        const mod = readString(wasmBytes, pos + count.bytes);
        const name = readString(wasmBytes, pos + count.bytes + mod.bytes);
        const kind = wasmBytes[pos + count.bytes + mod.bytes + name.bytes];
        result.imports.push({ module: mod.str, name: name.str, kind });
      }
    } else if (id === 7) { // Export section
      let count = readLeb(wasmBytes, pos);
      for (let i = 0; i < count.value; i++) {
        const name = readString(wasmBytes, pos + count.bytes);
        const kind = wasmBytes[pos + count.bytes + name.bytes];
        const idx = readLeb(wasmBytes, pos + count.bytes + name.bytes + 1);
        result.exports.push({ name: name.str, kind, index: idx.value });
      }
    } else if (id === 5) { // Memory section
      let count = readLeb(wasmBytes, pos);
      for (let i = 0; i < count.value; i++) {
        const flags = wasmBytes[pos + count.bytes];
        const min = readLeb(wasmBytes, pos + count.bytes + 1);
        result.memories.push({ flags, min: min.value });
      }
    }
    pos = end;
  }
  return result;
}

function readLeb(buf: Buffer, offset: number): { value: number; bytes: number } {
  let value = 0, shift = 0, bytes = 0;
  while (true) {
    const b = buf[offset + bytes++];
    value |= (b & 0x7f) << shift;
    shift += 7;
    if ((b & 0x80) === 0) break;
  }
  return { value, bytes };
}

function readString(buf: Buffer, offset: number): { str: string; bytes: number } {
  const len = readLeb(buf, offset);
  const str = buf.slice(offset + len.bytes, offset + len.bytes + len.value).toString('utf8');
  return { str, bytes: len.bytes + len.value };
}

app.post('/api/compile/cpp', async (req: Request, res: Response) => {
  const { code, flags } = req.body as CompileRequest;
  if (!code || typeof code !== 'string') {
    res.status(400).json({ success: false, error: 'Missing code field' });
    return;
  }

  const workDir = await fs.mkdtemp(join(tmpdir(), 'cpp-wasm-'));
  try {
    const emcc = await findInPathsOnly('emcc', [`${EMSDK_PATH}/upstream/emscripten`]);
    if (!emcc) {
      res.status(503).json({ success: false, error: 'emcc not found. Run: ./scripts/install-toolchains.sh' });
      return;
    }

    await fs.writeFile(join(workDir, 'main.cpp'), code);

    const extraFlags = flags ? ` ${flags}` : '';
    const cmd = `bash -c '${getEmsdkEnvCommand()} && emcc "${workDir}/main.cpp" -o "${workDir}/main.js" -s EXPORTED_FUNCTIONS="['_main']" -s EXPORTED_RUNTIME_METHODS="['ccall','cwrap','UTF8ToString']" -s ALLOW_MEMORY_GROWTH=1 -s INITIAL_MEMORY=16MB -s STACK_SIZE=1MB -s SINGLE_FILE=1 -O2${extraFlags}'`;

    const { stdout, stderr } = await execAsync(cmd, {
      timeout: 60000,
      env: { ...process.env, HOME: process.env.HOME || '' }
    });

    const jsFile = await fs.readFile(join(workDir, 'main.js'), 'utf-8');
    const diagnostics = parseDiagnostics(stderr, 'cpp');

    res.json({
      success: true,
      js: jsFile,
      stdout,
      stderr,
      diagnostics
    });
  } catch (err: any) {
    const diagnostics = parseDiagnostics(err.stderr || '', 'cpp');
    res.json({
      success: false,
      error: err.stderr || err.message,
      stdout: err.stdout || '',
      diagnostics
    });
  } finally {
    await fs.rm(workDir, { recursive: true, force: true }).catch(() => { });
  }
});

      return;
    }

    await fs.writeFile(join(workDir, 'main.swift'), code);

    const extraFlags = flags ? ` ${flags}` : '';
    const cmd = `PATH="${getSwiftPath()}:$PATH" swiftc -target wasm32-unknown-wasi -o "${workDir}/main.wasm" "${workDir}/main.swift"${extraFlags}`;

    const { stdout, stderr } = await execAsync(cmd, {
      timeout: 60000,
      env: { ...process.env, HOME: process.env.HOME || '' }
    });

    const wasmBytes = await fs.readFile(join(workDir, 'main.wasm'));
    const wasmInfo = parseWasmExportsImports(wasmBytes);
    const diagnostics = parseDiagnostics(stderr, 'swift');

    res.json({
      success: true,
      wasm: wasmBytes.toString('base64'),
      stdout,
      stderr,
      diagnostics,
      wasmInfo
    });
  } catch (err: any) {
    const diagnostics = parseDiagnostics(err.stderr || '', 'swift');
    res.json({
      success: false,
      error: err.stderr || err.message,
      stdout: err.stdout || '',
      diagnostics
    });
  } finally {
    await fs.rm(workDir, { recursive: true, force: true }).catch(() => { });
  }
});

app.post('/api/inspect', async (req: Request, res: Response) => {
  const { wasmBase64 } = req.body;
  if (!wasmBase64 || typeof wasmBase64 !== 'string') {
    res.status(400).json({ success: false, error: 'Missing wasmBase64 field' });
    return;
  }
  try {
    const wasmBytes = Buffer.from(wasmBase64, 'base64');
    const info = parseWasmExportsImports(wasmBytes);
    res.json({ success: true, info });
  } catch (err: any) {
    res.status(500).json({ success: false, error: err.message });
  }
});
