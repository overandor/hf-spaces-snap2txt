```
from ollama_bridge.phone.relay import connect_worker_websocket, disconnect_worker, handle_worker_message
from ollama_bridge.phone.sessions import verify_worker_join_token

app = FastAPI(title="Ollama Bridge", version="0.1.0")
import os
_static_dir = os.path.join(os.path.dirname(__file__), "static")
os.makedirs(_static_dir, exist_ok=True)

app.include_router(chat_router)

    ensure_storage_dirs()


@app.get("/", response_class=HTMLResponse)
async def landing(request: Request):
    return render_landing()


@app.websocket("/ws/worker/{session_id}/{worker_id}")
async def worker_ws(ws: WebSocket, session_id: str, worker_id: str):
    if not verify_worker_join_token(session_id, ws.query_params.get("token", "")):
        await ws.close(code=4001, reason="Invalid token")
        return
    await connect_worker_websocket(session_id, worker_id, ws)
    try:
        while True:
            msg = await ws.receive_json()
            await handle_worker_message(session_id, worker_id, msg)
    except WebSocketDisconnect:
        await disconnect_worker(session_id, worker_id)
```

```
import time
from typing import Dict, List, Optional
from ollama_bridge.core.models import MeshNode, MeshNodeStatus, NodeType, WorkerRuntimeType


class MeshRegistry:
    _nodes: Dict[str, MeshNode] = {}

    @classmethod
    def register_node(cls, node: MeshNode) -> None:
        cls._nodes[node.node_id] = node

    @classmethod
    def update_node_heartbeat(cls, node_id: str, metrics: dict) -> None:
        node = cls._nodes.get(node_id)
        if node:
            node.last_heartbeat = time.time()
            node.tokens_per_second = metrics.get("tokens_per_second", 0.0)
            node.ram_used_mb = metrics.get("ram_used_mb", 0.0)
            node.ram_total_mb = metrics.get("ram_total_mb", 0.0)
            node.cpu_usage = metrics.get("cpu_usage", 0.0)
            node.thermal_state = metrics.get("thermal_state", "nominal")
            node.battery_level = metrics.get("battery_level", 100.0)
            node.status = MeshNodeStatus.BUSY if metrics.get("is_generating") else MeshNodeStatus.ONLINE
            if node.current_job_id and not metrics.get("is_generating"):
                node.current_job_id = None

    @classmethod
    def set_ollama_url(cls, node_id: str, url: str) -> None:
        node = cls._nodes.get(node_id)
        if node:
            node.ollama_public_url = url

    @classmethod
    def get_node(cls, node_id: str) -> Optional[MeshNode]:
        return cls._nodes.get(node_id)

    @classmethod
    def list_nodes(cls) -> List[MeshNode]:
        cls._prune_stale()
        return list(cls._nodes.values())

    @classmethod
    def get_any_ollama_url(cls) -> Optional[str]:
        """Return the first available public Ollama URL from any connected node."""
        cls._prune_stale()
        for node in cls._nodes.values():
            if node.ollama_public_url and node.status != MeshNodeStatus.OFFLINE:
                return node.ollama_public_url
        return None

    @classmethod
    def get_combined_stats(cls) -> dict:
        nodes = cls.list_nodes()
        return {
            "nodes_online": len([n for n in nodes if n.status != MeshNodeStatus.OFFLINE]),
            "nodes_busy": len([n for n in nodes if n.status == MeshNodeStatus.BUSY]),
            "total_nodes": len(nodes),
            "combined_tokens_per_second": round(sum(n.tokens_per_second for n in nodes), 1),
            "combined_ram_used_mb": round(sum(n.ram_used_mb for n in nodes), 0),
            "combined_ram_total_mb": round(sum(n.ram_total_mb for n in nodes), 0),
        }

    @classmethod
    def _prune_stale(cls, timeout: float = 60.0) -> None:
        now = time.time()
        for nid, n in list(cls._nodes.items()):
            if now - n.last_heartbeat > timeout:
                n.status = MeshNodeStatus.OFFLINE
```

```
from typing import Optional, Dict
from ollama_bridge.core.models import PhoneSession


_sessions: Dict[str, PhoneSession] = {}


def create_phone_session(owner_label: str = "") -> PhoneSession:
    sid = f"sess_{uuid.uuid4().hex[:12]}"
    secret = f"tok_{uuid.uuid4().hex[:16]}"
    session = PhoneSession(
        session_id=sid,
        secret=secret,
        owner_label=owner_label,
        expires_at=datetime.datetime.utcnow() + datetime.timedelta(hours=24),
    )
    _sessions[sid] = session
    return session


def get_session(session_id: str) -> Optional[PhoneSession]:
    return _sessions.get(session_id)


def verify_worker_join_token(session_id: str, token: str) -> bool:
    s = _sessions.get(session_id)
    return s is not None and s.secret == token if s else False


def list_all_sessions():
    return list(_sessions.values())


def make_worker_join_url(session_id: str, secret: str) -> str:
    import os
    public_url = os.getenv("SPACE_PUBLIC_URL", "")
    if not public_url:
        public_url = "https://josephrw-ollama-bridge.hf.space"
    return f"{public_url}/phone?session={session_id}&token={secret}"
```

relay.py:
```
from typing import Dict, Any
from fastapi import WebSocket

from ollama_bridge.phone.sessions import get_session
from ollama_bridge.phone.workers import register_worker, get_worker, update_worker_heartbeat, register_capabilities
from ollama_bridge.core.models import WorkerRuntimeType, MeshNode, NodeType
from ollama_bridge.core.mesh import MeshRegistry


_session_sockets: Dict[str, Dict[str, WebSocket]] = {}


async def connect_worker_websocket(session_id: str, worker_id: str, ws: WebSocket) -> None:
    if session_id not in _session_sockets:
        _session_sockets[session_id] = {}
    _session_sockets[session_id][worker_id] = ws
    await ws.accept()


async def disconnect_worker(session_id: str, worker_id: str) -> None:
    if session_id in _session_sockets and worker_id in _session_sockets[session_id]:
        del _session_sockets[session_id][worker_id]


async def send_to_worker(session_id: str, worker_id: str, message: dict) -> None:
    ws = _session_sockets.get(session_id, {}).get(worker_id)
    if ws:
        await ws.send_json(message)


async def broadcast_session_update(session_id: str) -> None:
    workers = list(_session_sockets.get(session_id, {}).values())
    for ws in workers:
        try:
            await ws.send_json({"op": "session_update"})
        except Exception:
            pass


async def handle_worker_message(session_id: str, worker_id: str, message: dict) -> None:
    op = message.get("op")
    if op == "worker_hello":
        runtime_str = message.get("runtime_type", "safari_wasm")
        try:
            runtime = WorkerRuntimeType(runtime_str)
        except ValueError:
            runtime = WorkerRuntimeType.SAFARI_WASM
        # Ollama-only space: reject non-Mac workers
        if runtime != WorkerRuntimeType.OLLAMA:
            await send_to_worker(session_id, worker_id, {
                "op": "worker_rejected",
                "reason": "This Space only accepts Mac Agent (Ollama) compute nodes."
            })
            return
        w = register_worker(session_id, worker_id, runtime, message.get("device_public_key"))
        MeshRegistry.register_node(MeshNode(
            node_id=worker_id,
            node_type=NodeType.MAC_AGENT,
            runtime_type=runtime,
            capabilities=[],
        ))
        if w:
            w.device_model = message.get("device_model")
            w.device_imei = message.get("device_imei")
        await send_to_worker(session_id, worker_id, {"op": "worker_welcome", "worker_id": worker_id})

    elif op == "capabilities":
        caps = message.get("capabilities", [])
        register_capabilities(session_id, worker_id, caps)
        node = MeshRegistry.get_node(worker_id)
        if node:
            node.capabilities = [c.get("capability_name", "") for c in caps]
        await broadcast_session_update(session_id)

    elif op == "heartbeat":
        update_worker_heartbeat(
            session_id, worker_id,
            message.get("battery_level"),
            message.get("thermal_state"),
            message.get("cpu_usage"),
            message.get("ram_used_mb"),
            message.get("ram_total_mb"),
            message.get("tokens_per_second")
        )
        MeshRegistry.update_node_heartbeat(worker_id, {
            "tokens_per_second": message.get("tokens_per_second", 0.0),
            "ram_used_mb": message.get("ram_used_mb", 0.0),
            "ram_total_mb": message.get("ram_total_mb", 0.0),
            "cpu_usage": message.get("cpu_usage", 0.0),
            "thermal_state": message.get("thermal_state", "nominal"),
            "battery_level": message.get("battery_level", 100.0),
            "is_generating": message.get("is_generating", False),
        })

    elif op == "ollama_url":
        url = message.get("public_url")
        if url:
            MeshRegistry.set_ollama_url(worker_id, url)
        await send_to_worker(session_id, worker_id, {"op": "ollama_url_ack", "received": True})
```

```
from typing import Dict, Any, List
from ollama_bridge.core.models import WorkerRuntimeType


class Worker:
    def __init__(self, worker_id: str, runtime_type: WorkerRuntimeType, public_key: str = None):
        self.worker_id = worker_id
        self.runtime_type = runtime_type
        self.public_key = public_key
        self.is_connected = True
        self.tokens_per_second = 0.0
        self.capabilities: List[Dict[str, Any]] = []
        self.device_model = None
        self.device_imei = None


_workers_by_session: Dict[str, Dict[str, Worker]] = {}


def register_worker(session_id: str, worker_id: str, runtime_type: WorkerRuntimeType, public_key: str = None) -> Worker:
    if session_id not in _workers_by_session:
        _workers_by_session[session_id] = {}
    w = Worker(worker_id, runtime_type, public_key)
    _workers_by_session[session_id][worker_id] = w
    return w


def get_worker(session_id: str, worker_id: str) -> Worker:
    return _workers_by_session.get(session_id, {}).get(worker_id)


def list_workers_for_session(session_id: str) -> List[Worker]:
    return list(_workers_by_session.get(session_id, {}).values())


def update_worker_heartbeat(session_id: str, worker_id: str, battery: float, thermal: str, cpu: float, ram_used: float, ram_total: float, tps: float):
    w = get_worker(session_id, worker_id)
    if w:
        w.tokens_per_second = tps or 0.0


def register_capabilities(session_id: str, worker_id: str, caps: List[dict]):
    w = get_worker(session_id, worker_id)
    if w:
        w.capabilities = caps
```

```
import uuid
from fastapi import APIRouter, HTTPException
from pydantic import BaseModel
from ollama_bridge.phone.sessions import create_phone_session, get_session, make_worker_join_url
from ollama_bridge.ui import make_worker_qr_code

router = APIRouter(prefix="/api/gbridge")

_ollama_public_urls: dict = {}


class CreateSessionRequest(BaseModel):
    owner_label: str = ""


@router.post("/session")
async def create_session_route(req: CreateSessionRequest):
    session = create_phone_session(owner_label=req.owner_label)
    worker_url = make_worker_join_url(session.session_id, session.secret)
    qr_b64 = make_worker_qr_code(worker_url)
    return {
        "session_id": session.session_id,
        "join_token": session.secret,
        "space_url": worker_url,
        "qr_code_base64": qr_b64,
        "expires_at": session.expires_at.isoformat() if session.expires_at else None,
    }


@router.get("/session/{session_id}")
async def get_session_route(session_id: str):
    s = get_session(session_id)
    if not s:
        raise HTTPException(status_code=404, detail="Session not found")
    from ollama_bridge.phone.workers import list_workers_for_session
    return {
        "session_id": s.session_id,
        "status": s.status,
        "worker_count": len(list_workers_for_session(session_id)),
        "created_at": s.created_at.isoformat(),
    }


@router.post("/session/{session_id}/ollama-url")
async def set_ollama_url(session_id: str, payload: dict):
    s = get_session(session_id)
    if not s:
        raise HTTPException(status_code=404, detail="Session not found")
    url = payload.get("ollama_public_url")
    if not url:
        raise HTTPException(status_code=400, detail="Missing ollama_public_url")
    _ollama_public_urls[session_id] = url
    return {"status": "ok", "ollama_public_url": url}


@router.get("/session/{session_id}/ollama-url")
async def get_ollama_url(session_id: str):
    return {"ollama_public_url": _ollama_public_urls.get(session_id)}
```

```
    max_tokens: int = 1024
    stream: bool = True


def _build_ollama_payload(request: ChatRequest) -> Dict[str, Any]:
    return {
        "model": request.model or DEFAULT_MODEL,
        "messages": [{"role": m.role, "content": m.content} for m in request.messages],
        "stream": request.stream,
        "options": {
            "temperature": request.temperature,
            "num_predict": request.max_tokens,
        },
    }


async def _stream_ollama(payload: Dict[str, Any], base_url: str) -> AsyncGenerator[str, None]:
    url = f"{base_url.rstrip('/')}/api/chat"
    headers = {"Content-Type": "application/json"}
    async with httpx.AsyncClient(timeout=120.0) as client:
        async with client.stream("POST", url, json=payload, headers=headers) as resp:
            if resp.status_code != 200:
                body = await resp.aread()
                yield f"data: {json.dumps({'error': f'Ollama error {resp.status_code}'})}\n\n"
                return
            async for line in resp.aiter_lines():
                if not line:
                    continue
                try:
                    parsed = json.loads(line)
                    content = parsed.get("message", {}).get("content", "")
                    done = parsed.get("done", False)
                    if content:
                        yield f"data: {json.dumps({'content': content})}\n\n"
                    if done:
                        yield f"data: {json.dumps({'done': True})}\n\n"
                        break
                except json.JSONDecodeError:
                    continue


async def _chat_nonstream(payload: Dict[str, Any], base_url: str) -> Dict[str, Any]:
    url = f"{base_url.rstrip('/')}/api/chat"
    headers = {"Content-Type": "application/json"}
    async with httpx.AsyncClient(timeout=120.0) as client:
        resp = await client.post(url, json=payload, headers=headers)
        if resp.status_code != 200:
            raise HTTPException(status_code=resp.status_code, detail=resp.text)
        return resp.json()
```
